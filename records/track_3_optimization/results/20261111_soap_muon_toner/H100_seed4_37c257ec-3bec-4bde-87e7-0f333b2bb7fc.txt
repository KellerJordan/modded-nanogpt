"""
train_gpt_toner.py: Track 3 record #46 (SOAP-Muon, PR #328) with Toner's input geometry on the hidden matrices.

Changes from #46 (everything else is #46's: model, init, data, schedules, SOAP-Muon with the attn.proj trust gate,
12-step Newton-Schulz, radial scaling, RowFloor, radius pin, cautious weight decay, aux Adam, AdamW on the embedding
and head, EMA-Nesterov lookahead, tail-EMA readout):

1. Input statistic. For each hidden matrix, C = E[x x^T] of its input x (attention input for q, k, v; attention output
   for attn.proj; MLP input for mlp.fc; MLP hidden activation for mlp.proj), from the training forward pass (rows at
   positions 0..31, then every 16th; all ranks; FP32; EMA 0.938 per step). No extra forward or backward pass.
2. Toner's roots, every step: outer root R = (C / mean eig(C) + 1e-3 I)^(-1/8), inner root R^4 = (...)^(-1/2).
3. The SOAP-Muon step runs in the inner root's coordinates: the Nesterov update U and the gradient G enter as U R^4
   and G R^4 (SOAP's statistics from G R^4); SOAP preconditioning, the trust gate, the blend and Newton-Schulz act
   there; the result O leaves through the outer root as O R, rescaled to O's Frobenius norm. #46's radial scaling,
   RowFloor, radius pin and cautious decay then act in the original coordinates, unchanged.
4. Shape decay: after #46's post-update, W -= lr * 0.08 * (W R~ minus its radial part), R~ = R^2 / mean diag(R^2),
   while the EMA-Nesterov lookahead is active (steps < 1950, #46's lookahead end). It moves weight mass from
   low-variance toward high-variance input directions; the radius is unchanged to first order.

Implementation (also an efficient re-implementation of #46 for 8 GPUs; rounding-level differences from #46's code):
SOAP-Muon runs on the matrices' owners, batched by shape (one in-place reduce-scatter of a flat gradient buffer, one
all-gather of the flat parameter buffer); SOAP's basis update runs as a CUDA graph on side streams after the step,
used by the next step as in #46. The roots come from an eigenbasis of the same statistic a few steps old (host FP32
eigh refreshes, staggered) and a preconditioned fixed-point iteration for B^(-1/4), B = X^T (C/mu + eps I) X
(FP32-accurate FP16 hi/lo products), one square-root stage for the 1/8 root, and two FP32 squarings for the 1/2 root;
a safeguard falls back to an exact GPU eigh. Statistics run on a side stream during the backward.

Descends from the NanoGPT speedrun (https://github.com/KellerJordan/modded-nanogpt), record #46.
"""

def _load_data_shard(file: Path):
    header = torch.from_file(str(file), False, 256, dtype=torch.int32) # header is 256 int32
    assert header[0] == 20240520, "magic number mismatch in the data .bin file"
    assert header[1] == 1, "unsupported version"
    num_tokens = int(header[2]) # number of tokens (claimed)
    with file.open("rb", buffering=0) as f:
        tokens = torch.empty(num_tokens, dtype=torch.uint16, pin_memory=True)
        f.seek(256 * 4)
        nbytes = f.readinto(tokens.numpy()) # avoid bytes->array copy
        assert nbytes == 2 * num_tokens, "number of tokens read does not match header"
    return tokens

def distributed_data_generator(filename_pattern: str, batch_size: int, seq_len=1024):
    world_size = dist.get_world_size()
    rank = dist.get_rank()
    files = sorted(Path.cwd().glob(filename_pattern))
    assert batch_size % world_size == 0
    local_batch_size = batch_size // world_size
    file_iter = iter(files)
    tokens, pos = _load_data_shard(next(file_iter)), 0
    while True:
        if pos + batch_size + 1 >= len(tokens):
            tokens, pos = _load_data_shard(next(file_iter)), 0
        buf = tokens[pos + rank * local_batch_size:][:local_batch_size + 1]
        inputs = buf[:-1].to(device="cuda", dtype=torch.int32, non_blocking=True)
        targets = buf[1:].to(device="cuda", dtype=torch.int64, non_blocking=True)
        pos += batch_size
        yield inputs.view(-1, seq_len), targets.view(-1, seq_len)

# #46's model; with PD, the sampled rows of each hidden matrix's input are copied into the module's rows buffer
# (training forwards only; a side output that changes nothing in the forward).

def norm(x: Tensor):
    return F.rms_norm(x, (x.size(-1),))

def pd_capture(mod, name, t: Tensor):
    """Rows pd_idx of every sequence of t (B, T, D) into buffer `name` of mod at rows mod.pd_at (this micro-batch's)."""
    buf = getattr(mod, name)
    rows = t.detach().index_select(1, mod.pd_idx).reshape(-1, t.size(-1))
    if rows.size(0) == buf.size(0):
        buf.copy_(rows)
    else:
        buf.index_copy_(0, mod.pd_at, rows)

def pd_buffers(mod, names_dims, rows, at):
    """PD rows buffers on the module that fills them (bf16, one allocation each: syrk reads them through a pointer
    table), the sampled positions and this micro-batch's row slots (shared tensors)."""
    dev = next(mod.parameters()).device
    for name, d in names_dims:
        mod.register_buffer(name, torch.zeros(rows, d, dtype=torch.bfloat16, device=dev), persistent=False)
    mod.register_buffer("pd_idx", torch.tensor(PD_POS, device=dev), persistent=False)
    mod.register_buffer("pd_at", at, persistent=False)
    mod.pd_on = True

class RMSNorm(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.gains = nn.Parameter(torch.ones(dim))

    def forward(self, x):
        return (norm(x.float()) * self.gains).type_as(x)

class Linear(nn.Linear):
    def __init__(self, in_features, out_features):
        super().__init__(in_features, out_features, bias=True)

    def forward(self, x):
        return F.linear(x, self.weight.type_as(x), self.bias.type_as(x))

class Rotary(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        # half-truncate RoPE (w/ base freq tuning)
        angular_freq = (1 / 1024) ** torch.linspace(0, 1, steps=dim//4, dtype=torch.float32)
        self.register_buffer("angular_freq", torch.cat([angular_freq, angular_freq.new_zeros(dim//4)]))

    def forward(self, x_BTHD: Tensor):
        pos = torch.arange(x_BTHD.size(1), dtype=torch.float32, device=x_BTHD.device)
        theta = torch.outer(pos, self.angular_freq)[None, :, None, :]
        cos, sin = theta.cos(), theta.sin()
        x1, x2 = x_BTHD.to(dtype=torch.float32).chunk(2, dim=-1)
        y1 = x1 * cos + x2 * sin
        y2 = x1 * (-sin) + x2 * cos
        return torch.cat((y1, y2), 3).type_as(x_BTHD)

class CausalSelfAttention(nn.Module):
    def __init__(self, dim: int, head_dim=128):
        super().__init__()
        self.num_heads = dim // head_dim
        self.head_dim = head_dim
        hdim = self.num_heads * self.head_dim
        self.q = Linear(dim, hdim)
        self.k = Linear(dim, hdim)
        self.v = Linear(dim, hdim)
        self.proj = Linear(hdim, dim)
        self.rotary = Rotary(head_dim)
        self.pd_on = False

    def forward(self, x: Tensor):
        B, T = x.size(0), x.size(1)
        q = self.q(x).view(B, T, self.num_heads, self.head_dim)
        k = self.k(x).view(B, T, self.num_heads, self.head_dim)
        v = self.v(x).view(B, T, self.num_heads, self.head_dim)
        q, k = norm(q), norm(k)
        q, k = self.rotary(q), self.rotary(k)
        y = F.scaled_dot_product_attention(q.transpose(1, 2), k.transpose(1, 2),
                                           v.transpose(1, 2), scale=0.12, is_causal=True).transpose(1, 2)
        y = y.contiguous().view(B, T, self.num_heads * self.head_dim)
        if self.pd_on and self.training:
            pd_capture(self, "pdr_o", y)
        y = self.proj(y)
        return y

class MLP(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        hdim = 4 * dim
        self.fc = Linear(dim, hdim)
        self.proj = Linear(hdim, dim)
        self.pd_on = False

    def forward(self, x: Tensor):
        if self.pd_on and self.training:
            pd_capture(self, "pdr_mlp", x)
        x = self.fc(x)
        x = x.relu().square()
        if self.pd_on and self.training:
            pd_capture(self, "pdr_hidden", x)
        x = self.proj(x)
        return x

class Block(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        self.attn = CausalSelfAttention(dim)
        self.mlp = MLP(dim)
        self.norm1 = RMSNorm(dim)
        self.norm2 = RMSNorm(dim)
        self.pd_on = False

    def enable_pd(self, rows, at, dim, hdim):
        pd_buffers(self, (("pdr_qkv", dim),), rows, at)
        pd_buffers(self.attn, (("pdr_o", dim),), rows, at)
        pd_buffers(self.mlp, (("pdr_mlp", dim), ("pdr_hidden", hdim)), rows, at)

    def pd_rows(self, kind):
        return {"qkv": self.pdr_qkv, "o": self.attn.pdr_o, "mlp": self.mlp.pdr_mlp, "hidden": self.mlp.pdr_hidden}[kind]

    def forward(self, x: Tensor):
        h = self.norm1(x)
        if self.pd_on and self.training:
            pd_capture(self, "pdr_qkv", h)
        x = x + self.attn(h)
        x = x + self.mlp(self.norm2(x))
        return x

class GPT(nn.Module):
    def __init__(self, vocab_size: int, num_layers: int, model_dim: int):
        super().__init__()
        self.embed = nn.Embedding(vocab_size, model_dim).bfloat16()
        self.blocks = nn.ModuleList([Block(model_dim) for _ in range(num_layers)])
        self.proj = Linear(model_dim, vocab_size)
        self.norm1 = RMSNorm(model_dim)
        self.norm2 = RMSNorm(model_dim)

    def forward(self, inputs: Tensor, targets: Tensor):
        x = self.norm1(self.embed(inputs))
        for block in self.blocks:
            x = block(x)
        logits = self.proj(self.norm2(x)).float()
        logits = 15 * logits * (logits.square() + 15**2).rsqrt()
        return F.cross_entropy(logits.view(targets.numel(), -1), targets.view(-1), reduction="sum")

# #46's functions, written for stacks (k, m, n) of same-shape matrices: per-matrix reductions over dims (-2, -1)
# instead of whole-tensor ones, the same operations in the same order otherwise.

def gram_frobenius_norm_estimate(G: Tensor, keepdim: bool = False, eps: float = 1e-10) -> Tensor:
    X = G.float()
    gram = X.mT @ X if X.size(-2) > X.size(-1) else X @ X.mT
    return gram.norm(dim=(-2, -1), keepdim=keepdim).sqrt().clamp_min(eps)

def _ns_inner(X: Tensor) -> Tensor:
    a, b, c = 2, -1.5, 0.5
    for _ in range(12):
        A = X @ X.mT
        B = b * A + c * A @ A
        X = a * X + B @ X
    return X

def zeropower_via_newtonschulz5(G: Tensor) -> Tensor:
    assert G.ndim >= 2
    X = G.bfloat16()
    if G.size(-2) > G.size(-1):
        X = X.mT
    X = X / gram_frobenius_norm_estimate(X, keepdim=True, eps=1e-7).to(X.dtype)
    X = _ns_inner(X)
    if G.size(-2) > G.size(-1):
        X = X.mT
    return X

def muon_update(update):
    update = zeropower_via_newtonschulz5(update)
    update *= max(1, update.size(-2) / update.size(-1))**0.5
    return update

def tensor_cosine(a: Tensor, b: Tensor, eps: float = 1e-8) -> Tensor:
    a_f, b_f = a.float(), b.float()
    return (a_f * b_f).sum((-2, -1)) / (a_f.norm(dim=(-2, -1)) * b_f.norm(dim=(-2, -1))).clamp_min(eps)

def trust_gate(raw: Tensor, soap: Tensor, grad: Tensor, eps: float = 1e-8) -> Tensor:
    raw_grad = tensor_cosine(raw, grad, eps)
    soap_grad = tensor_cosine(soap, grad, eps)
    soap_raw = tensor_cosine(soap, raw, eps)
    agree_gate = ((soap_raw - ATTN_TRUST_MIN_AGREE) / (1 - ATTN_TRUST_MIN_AGREE)).clamp(0, 1)
    denom = (raw_grad - ATTN_TRUST_MIN_GRAD_ALIGN).clamp_min(eps)
    grad_gate = ((soap_grad - ATTN_TRUST_MIN_GRAD_ALIGN) / denom).clamp(0, 1)
    gate = (agree_gate * grad_gate).clamp(0, 1)
    return gate

def early_trust_floor_for_step(step: int) -> float:
    if step < ATTN_TRUST_FLOOR_END_STEP:
        return ATTN_EARLY_TRUST_FLOOR
    if step >= ATTN_TRUST_FLOOR_FADE_END_STEP:
        return 0.0
    return ATTN_EARLY_TRUST_FLOOR * (
        ATTN_TRUST_FLOOR_FADE_END_STEP - step
    ) / (ATTN_TRUST_FLOOR_FADE_END_STEP - ATTN_TRUST_FLOOR_END_STEP)

def bounded_trust_gate(gate: Tensor, step: int) -> Tensor:
    floor = early_trust_floor_for_step(step)
    cap = ATTN_EARLY_TRUST_CAP if step < ATTN_TRUST_FLOOR_FADE_END_STEP else 1.0
    return gate.clamp(min=floor, max=cap)

def norm_preserving_blend(raw: Tensor, soap: Tensor, gate: Tensor, eps: float = 1e-8) -> Tensor:
    blended = raw + (soap - raw) * gate.to(raw.dtype)[:, None, None]
    raw_norm = gram_frobenius_norm_estimate(raw, eps=eps)
    blended_norm = gram_frobenius_norm_estimate(blended, eps=eps)
    return (blended * (raw_norm / blended_norm).to(blended.dtype)[:, None, None]).to(raw.dtype)

def soap_precondition_momentum(update, gr, beta2=SOAP_BETA2, eps=1e-8):
    """#46's SOAP preconditioning of a stack; gr holds the stacked state (q_row, q_col, exp_avg_sq)."""
    update_f = update.float()
    if not gr.soap_ready:
        return update
    projected = gr.q_row.mT @ update_f @ gr.q_col
    gr.exp_avg_sq.mul_(beta2).add_(projected.square(), alpha=1 - beta2)
    denom = gr.exp_avg_sq.clamp_min(eps * eps).pow(SOAP_DENOM_POWER)
    precond = gr.q_row @ (projected / denom) @ gr.q_col.mT
    precond.mul_((gram_frobenius_norm_estimate(update_f, eps=eps) / gram_frobenius_norm_estimate(precond, eps=eps))
                 [:, None, None])
    return precond.to(update.dtype)

def soap_eigenbasis(mat: Tensor) -> Tensor:
    try:
        _, q = torch.linalg.eigh(mat + 1e-30 * torch.eye(mat.size(0), device=mat.device))
    except RuntimeError:
        _, q = torch.linalg.eigh(mat.double() + 1e-30 * torch.eye(mat.size(0), device=mat.device))
        q = q.float()
    return torch.flip(q, [1])

def _cqr_pass(X, shift=0.0):
    W = X.mT @ X
    if shift:
        W = W + shift * torch.eye(W.size(-1), device=W.device, dtype=W.dtype)
    L, _ = torch.linalg.cholesky_ex(W)
    return torch.linalg.solve_triangular(L, X.mT, upper=False).mT  # X L^-T


def scqr3(A):
    """Q of A = QR for a stack (k, n, n): shifted Cholesky-QR3 in FP64 (Fukaya et al. 2020) on column-scaled input (Q
    is invariant to positive column scaling). The same Q as Householder up to column signs for full-rank A (SOAP's
    update is invariant to column signs); orthonormal also for rank-deficient A (the shift). Returns Q (FP32) and its
    orthogonality defect max |Q^T Q - I| (device scalar)."""
    X = A.double()
    X = X / X.norm(dim=-2, keepdim=True).clamp_min(1e-300)
    m, n = X.shape[-2:]
    Q = _cqr_pass(X, shift=11 * (m * n + n * (n + 1)) * 2.0 ** -53 * n)  # ||X||_2^2 <= n after the scaling
    Q = _cqr_pass(_cqr_pass(Q)).float()
    defect = (Q.mT @ Q - torch.eye(n, device=Q.device)).abs().amax()
    return Q, torch.nan_to_num(defect, nan=float("inf"))


def post_update_(P: Tensor, update: Tensor, lr: float, target_uw: float = SOAP_TARGET_UW):
    """#46's step after Newton-Schulz for a stack: radial scaling, RowFloor, target radius, update, radius pin, CWD."""
    # scale_radial_update
    update_f = update.float()
    param_f = P.float()
    denom = (param_f * param_f).sum((-2, -1), keepdim=True).clamp_min(1e-12)
    coeff = (update_f * param_f).sum((-2, -1), keepdim=True) / denom
    radial = coeff * param_f
    tangential = update_f - radial
    radial_scale = torch.where(coeff < 0, update_f.new_tensor(RADIAL_OUTWARD_SCALE),
                               update_f.new_tensor(RADIAL_INWARD_SCALE))
    update = (tangential + radial_scale * radial).to(update.dtype)
    # (B) RowFloor
    r_row = P.float().norm(dim=-1, keepdim=True).clamp_min(1e-8)
    s_row = update.float().norm(dim=-1, keepdim=True).clamp_min(1e-8)
    f_row = torch.clamp(target_uw * r_row / s_row, min=1.0).pow(ROWFLOOR_RHO)
    update = (update.float() * f_row).to(update.dtype)
    # target_radius_after_update
    before_norm = P.float().norm(dim=(-2, -1), keepdim=True).clamp_min(1e-8)
    radial_delta = -lr * (update.float() * P.float()).sum((-2, -1), keepdim=True) / before_norm
    target_radius = (before_norm + radial_delta).clamp_min(1e-8)
    # (C) cautious weight decay mask, update, radius pin, CWD
    cwd_mask = (update.float() * P.float() > 0).to(P.dtype)
    P.add_(update, alpha=-lr)
    after_norm = P.float().norm(dim=(-2, -1), keepdim=True).clamp_min(1e-8)
    P.mul_((target_radius / after_norm).to(P.dtype))
    P.mul_(1.0 - (lr * CWD) * cwd_mask)


def inner_root(Rs: Tensor) -> Tensor:
    """Toner's inner root R^4 = (C / mean eig + eps I)^(-1/2) from the outer root R = (...)^(-1/8): two squarings
    in true FP32 (this script never enables TF32), the same damping."""
    Ri = Rs.float()  # true FP32: this trainer never enables TF32 (PyTorch's default for FP32 matmul is off)
    for _ in range(2):
        Ri = Ri @ Ri
    return Ri


def pd_shape_decay_(P: Tensor, Rs: Tensor, rate: float):
    """--pd-shape-decay for a stack: P -= rate (P R~ - radial part), R~ = R^2 / mean diag(R^2); zero at alpha 0."""
    Rf = Rs.float()
    Rt = Rf @ Rf
    Rt = Rt / Rt.diagonal(dim1=-2, dim2=-1).mean(-1)[:, None, None]
    Pf = P.float()
    Wd = Pf @ Rt
    Wd = Wd - ((Wd * Pf).sum((-2, -1), keepdim=True) / Pf.square().sum((-2, -1), keepdim=True).clamp_min(1e-30)) * Pf
    P.sub_(Wd.to(P.dtype), alpha=rate)

# Packed symmetric rank-k Grams (syrk.py), fused unpack + EMA, symmetric products (symm16.py), FP16 hi/lo splits and
# the near-diagonal fourth root (pd_roots.py). HOPPER selects the larger tiles tuned on H100.
HOPPER = torch.cuda.is_available() and torch.cuda.get_device_capability()[0] == 9


@triton.jit
def _syrk_packed_kernel(X0, OFFS, S, OIDX, R: tl.constexpr, N: tl.constexpr, BM: tl.constexpr, BK: tl.constexpr):
    nt = tl.cdiv(N, BM)
    ntt = nt * (nt + 1) // 2
    b = tl.program_id(0) // ntt
    t = tl.program_id(0) % ntt
    i = 0
    rem = t
    while rem >= nt - i:
        rem -= nt - i
        i += 1
    j = i + rem
    m = i * BM + tl.arange(0, BM)
    n = j * BM + tl.arange(0, BM)
    k = tl.arange(0, BK)
    base = X0 + tl.multiple_of(tl.load(OFFS + b), 256)
    acc = tl.zeros((BM, BM), tl.float32)
    for r in range(tl.cdiv(R, BK)):
        rr = r * BK + k
        a = tl.load(base + rr[:, None] * N + m[None, :], (rr[:, None] < R) & (m[None, :] < N), other=0)
        c = tl.load(base + rr[:, None] * N + n[None, :], (rr[:, None] < R) & (n[None, :] < N), other=0)
        acc = tl.dot(tl.trans(a), c, acc)
    off = m[:, None] * N - m[:, None] * (m[:, None] + 1) // 2 + n[None, :]
    keep = (m[:, None] <= n[None, :]) & (n[None, :] < N)
    tl.store(S + tl.load(OIDX + b) * (N * (N + 1) // 2) + off, acc, keep)


@triton.jit
def _syrk_packed_rect_kernel(X0, OFFS, S, OIDX, R: tl.constexpr, N: tl.constexpr, NT: tl.constexpr,
                             BM: tl.constexpr, BN: tl.constexpr, BK: tl.constexpr):
    b = tl.program_id(0) // NT
    t = tl.program_id(0) % NT
    ntn = N // BN
    i = 0
    rem = t
    cnt = ntn
    while rem >= cnt:
        rem -= cnt
        i += 1
        cnt = ntn - (i * BM) // BN
    j = (i * BM) // BN + rem
    m = i * BM + tl.arange(0, BM)
    n = j * BN + tl.arange(0, BN)
    k = tl.arange(0, BK)
    base = X0 + tl.multiple_of(tl.load(OFFS + b), 256)
    acc = tl.zeros((BM, BN), tl.float32)
    for r in range(tl.cdiv(R, BK)):
        rr = r * BK + k
        a = tl.load(base + rr[:, None] * N + m[None, :], rr[:, None] < R, other=0)
        c = tl.load(base + rr[:, None] * N + n[None, :], rr[:, None] < R, other=0)
        acc = tl.dot(tl.trans(a), c, acc)
    off = m[:, None] * N - m[:, None] * (m[:, None] + 1) // 2 + n[None, :]
    tl.store(S + tl.load(OIDX + b) * (N * (N + 1) // 2) + off, acc, m[:, None] <= n[None, :])


def list_offsets(xs):
    p0 = xs[0].data_ptr()
    offs = [(x.data_ptr() - p0) for x in xs]
    assert all(o % 512 == 0 for o in offs), "rows buffers must be 512-byte aligned"
    return torch.tensor([o // xs[0].element_size() for o in offs], dtype=torch.int64, device=xs[0].device)


def syrk_list_packed(xs, offs, out, oidx):
    """out[oidx[b]] (rows of n (n + 1) / 2, FP32) = packed upper triangle of xs[b]^T xs[b] (xs: k (R, n) bf16 buffers);
    out's other rows (padding slots) are left as they are."""
    k = len(xs)
    R, n = xs[0].shape
    assert out.shape[1:] == (n * (n + 1) // 2,) and out.dtype == torch.float32 and out.is_contiguous()
    if HOPPER and n % 256 == 0:
        BM, BN, BK, st = (256, 128, 64, 3) if n >= 3072 else (128, 256, 32, 5)
        nt = sum(n // BN - (i * BM) // BN for i in range(n // BM))
        _syrk_packed_rect_kernel[(k * nt,)](xs[0], offs, out, oidx, R, n, nt, BM, BN, BK, num_warps=8, num_stages=st)
        return out
    BM = 128
    assert n % BM == 0
    nt = n // BM
    _syrk_packed_kernel[(k * nt * (nt + 1) // 2,)](xs[0], offs, out, oidx, R, n, BM, 64, num_warps=4, num_stages=3)
    return out


@triton.jit
def _unpack_ema_kernel(P, C, N: tl.constexpr, BETA: tl.constexpr, GAMMA: tl.constexpr, BLOCK: tl.constexpr):
    b = tl.program_id(1)
    x = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    i = x // N
    j = x % N
    lo = tl.minimum(i, j)
    hi = tl.maximum(i, j)
    val = tl.load(P + b * (N * (N + 1) // 2) + lo * N - lo * (lo + 1) // 2 + hi, x < N * N, other=0)
    old = tl.load(C + b * N * N + x, x < N * N, other=0) * BETA
    tl.store(C + b * N * N + x, tl.fma(val, GAMMA, old), x < N * N)


def unpack_ema_(packed, C, beta, gamma):
    """In place: C[b] = beta C[b] + gamma S[b], S[b] symmetric given by its packed upper triangle."""
    n = C.size(-1)
    assert C.is_contiguous() and packed.is_contiguous() and C.shape == (packed.size(0), n, n)
    _unpack_ema_kernel[(triton.cdiv(n * n, 256), C.size(0))](packed, C, n, float(beta), float(gamma), 256,
                                                             enable_fp_fusion=False)
    return C


LO = 4096.0  # the lo halves of the FP16 splits are stored times LO (no FP16 underflow)


@triton.jit
def _sym_kernel(A, B, C, N: tl.constexpr, K: tl.constexpr, NT: tl.constexpr, sab, sam, sak, sbb, sbk, sbn,
                SPLIT: tl.constexpr, A_LO_FIRST: tl.constexpr, BM: tl.constexpr, BN: tl.constexpr, BK: tl.constexpr,
                INV_LO: tl.constexpr):
    b = tl.program_id(0) // NT
    t = tl.program_id(0) % NT
    ntn = N // BN
    i = 0
    rem = t
    cnt = ntn
    while rem >= cnt:
        rem -= cnt
        i += 1
        cnt = ntn - (i * BM) // BN
    j = (i * BM) // BN + rem
    m = i * BM + tl.arange(0, BM)
    n = j * BN + tl.arange(0, BN)
    k = tl.arange(0, BK)
    a_b = A + b * sab
    b_b = B + b * sbb
    acc = tl.zeros((BM, BN), tl.float32)
    if SPLIT:
        for kk in range(0, 2 * K, BK):
            a = tl.load(a_b + m[:, None] * sam + (kk + k)[None, :] * sak)
            c = tl.load(b_b + (kk + k)[:, None] * sbk + n[None, :] * sbn)
            acc = tl.dot(a, c, acc)
        acc = acc * INV_LO
    a_hi: tl.constexpr = K if SPLIT and A_LO_FIRST else 0
    b_hi: tl.constexpr = K if SPLIT and not A_LO_FIRST else 0
    for kk in range(0, K, BK):
        a = tl.load(a_b + m[:, None] * sam + (a_hi + kk + k)[None, :] * sak)
        c = tl.load(b_b + (b_hi + kk + k)[:, None] * sbk + n[None, :] * sbn)
        acc = tl.dot(a, c, acc)
    up = m[:, None] <= n[None, :]
    cb = C + b * N * N
    tl.store(cb + m[:, None] * N + n[None, :], acc, up)
    tl.store(cb + n[None, :] * N + m[:, None], acc, up)


def _sym_cfg(n):
    return (128, 256, 32, 8, 4) if HOPPER and n % 256 == 0 else (128, 128, 32, 8, 3)


def _sym_launch(A, B, C, n, k, split, a_lo_first):
    BM, BN, BK, w, st = _sym_cfg(n)
    assert n % BN == 0 and n % BM == 0 and k % BK == 0
    nt = sum(n // BN - (i * BM) // BN for i in range(n // BM))
    _sym_kernel[(A.size(0) * nt,)](A, B, C, n, k, nt, A.stride(0), A.stride(1), A.stride(2), B.stride(0), B.stride(1),
                                   B.stride(2), split, bool(a_lo_first), BM, BN, BK, 1.0 / LO, num_warps=w,
                                   num_stages=st)
    return C


def mm16_sym(A2, B2, a_lo_first):
    """mm16 for a result known to be symmetric (k, n, n), from its upper-triangle tiles."""
    k, n, n2 = A2.shape
    assert n2 == 2 * n and B2.shape == (k, 2 * n, n)
    return _sym_launch(A2, B2, torch.empty(k, n, n, device=A2.device), n, n, True, a_lo_first)


def mm_sym(A, B):
    """A @ B, FP16 inputs, FP32 accumulation, for a result known to be symmetric (k, n, n)."""
    k, n, kk = A.shape
    return _sym_launch(A, B, torch.empty(k, n, n, device=A.device), n, kk, False, False)


EPS_PD = 1e-3  # damping relative to the mean eigenvalue
U32 = 2.0 ** -24
LOWP = torch.float16


def _hi_lo(x):
    h = x.half()
    return h, ((x - h.float()) * LO).half()


def _cols_hi(x):
    h, lo = _hi_lo(x)
    return torch.cat([h, lo], -1)


def _cols_lo(x):
    h, lo = _hi_lo(x)
    return torch.cat([lo, h], -1)


def _rows_hi(x):
    h, lo = _hi_lo(x)
    return torch.cat([h, lo], -2)


def _rows_lo(x):
    h, lo = _hi_lo(x)
    return torch.cat([lo, h], -2)


def _cols_hi_scaled(x):
    a = x.abs().amax((-2, -1), keepdim=True).clamp_min(1e-30)
    s = torch.exp2(13.0 - torch.floor(torch.log2(a)))
    h, lo = _hi_lo(x * s)
    return torch.cat([h, lo], -1), 1.0 / s


_EXACT = {"emulate_precision_casts": True}  # lo = x - fp16(x) must not be fused away
torch._dynamo.config.recompile_limit = max(torch._dynamo.config.recompile_limit, 64)
torch._dynamo.config.cache_size_limit = max(torch._dynamo.config.cache_size_limit, 64)
cols_hi = torch.compile(_cols_hi, dynamic=False, options=_EXACT)
cols_lo = torch.compile(_cols_lo, dynamic=False, options=_EXACT)
rows_hi = torch.compile(_rows_hi, dynamic=False, options=_EXACT)
rows_lo = torch.compile(_rows_lo, dynamic=False, options=_EXACT)
cols_hi_scaled = torch.compile(_cols_hi_scaled, dynamic=False, options=_EXACT)


def mm16(A2, B2, a_lo_first):
    """FP32-accurate A @ B from FP16 split operands: A2 = [lo | hi], B2 = [hi ; lo] (a_lo_first) or [hi | lo], [lo ; hi]."""
    k = A2.size(-1) // 2
    Ah, Bh = (A2[..., k:], B2[..., :k, :]) if a_lo_first else (A2[..., :k], B2[..., k:, :])
    if A2.dim() == 3:
        C = torch.bmm(Ah, Bh, out_dtype=torch.float32)
        return torch.baddbmm(C, A2, B2, alpha=1.0 / LO, out_dtype=torch.float32)
    C = torch.mm(Ah, Bh, out_dtype=torch.float32)
    return torch.addmm(C, A2, B2, alpha=1.0 / LO, out_dtype=torch.float32)


def apply16(A, R2):
    """A @ R, FP32-accurate for any scale of A (k, m, n); R2 = rows_lo(R) (k, 2n, n)."""
    A2, inv = cols_hi_scaled(A.contiguous())
    return mm16(A2, R2, False) * inv


class Basis:
    """Orthonormal bases X (k, n, n) kept as FP16 splits: SB = [lo ; hi] (rows), S2 = [hi | lo] (cols)."""

    def __init__(self, X):
        self.SB = rows_lo(X.contiguous())
        self.S2 = cols_hi(X.contiguous())

    @classmethod
    def of_splits(cls, S2):
        b = cls.__new__(cls)
        b.S2 = S2
        return b

    def install(self, j, X):
        X = X.contiguous().unsqueeze(0)
        self.SB[j].copy_(rows_lo(X)[0])
        self.S2[j].copy_(cols_hi(X)[0])

    def install_many(self, X, mask):
        install_masked(self.SB, self.S2, X, mask)


def _install_masked(SB, S2, X, mask):
    m = mask[:, None, None]
    SB.copy_(torch.where(m, _rows_lo(X), SB))
    S2.copy_(torch.where(m, _cols_hi(X), S2))


install_masked = torch.compile(_install_masked, dynamic=False, options=_EXACT)


def _scaled_cols_hi(C):
    n = C.size(-1)
    mu = C.diagonal(dim1=-2, dim2=-1).sum(-1) / n
    return _cols_hi(C / mu[..., None, None]), mu


scaled_cols_hi = torch.compile(_scaled_cols_hi, dynamic=False, options=_EXACT)


def basis_transform(C, basis):
    """B = X^T (C / mu) X (FP32-accurate; EPS_PD is added in the root), mu = tr(C) / n."""
    A2, mu = scaled_cols_hi(C)
    Y = mm16(A2, basis.SB, False)
    if C.size(-1) % 256 == 0:
        return mm16_sym(basis.SB.mT, rows_hi(Y), True), mu
    return mm16(basis.SB.mT, rows_hi(Y), True), mu


def form(basis, lam, O):
    """R = X (diag(lam) + O) X^T, FP32-accurate, symmetric."""
    XY = mm16(basis.S2, rows_lo(torch.diag_embed(lam) + O), False)
    if lam.size(-1) % 256 == 0:
        return mm16_sym(cols_lo(XY), basis.S2.mT, True)
    R = mm16(cols_lo(XY), basis.S2.mT, True)
    return 0.5 * (R + R.mT)


def _start(B):
    """Symmetrize B, add EPS_PD; first-order (Daleckii-Krein) start of Y = (B + eps I)^(-1/4) from B's diagonal."""
    B = 0.5 * (B + B.mT)
    dB = B.diagonal(dim1=-2, dim2=-1)
    d = dB + EPS_PD
    E = B - torch.diag_embed(dB)
    rel = E.abs() / (d[..., :, None] + d[..., None, :])
    rho = rel.amax((-2, -1))
    lam = d.clamp_min(EPS_PD * 1e-3).pow(-0.25)
    dd = d[..., :, None] - d[..., None, :]
    mid = 0.5 * (d[..., :, None] + d[..., None, :])
    L = torch.where(dd.abs() > 1e-6 * mid, (lam[..., :, None] - lam[..., None, :]) / torch.where(dd == 0, 1.0, dd),
                    -0.25 * mid.pow(-1.25))
    O = L * E
    return d, E, E.to(LOWP), lam, O, O.to(LOWP), d.sqrt(), rho, O.square().sum((-2, -1)).sqrt()


def _iter_mid(lam, O, OO):
    P = (lam[..., :, None] + lam[..., None, :]) * O + OO
    return P, P.to(LOWP)


def _iter_w(P, d, EP):
    return (d[..., :, None] * P + EP).to(LOWP)


def _iter_post(lam, O, d, E, P, EP, PW, lam0, sq, pw_sym: bool = False):
    l2 = lam.square()
    F_ = (l2[..., :, None] * E * l2[..., None, :]
          + (l2 * d)[..., :, None] * P + P * (d * l2)[..., None, :]
          + l2[..., :, None] * EP + EP.mT * l2[..., None, :]
          + (PW if pw_sym else 0.5 * (PW + PW.mT)))
    G = torch.diag_embed(1.0 - l2 * d * l2) - F_
    pre = (lam0[..., :, None] + lam0[..., None, :]) * (sq[..., :, None] + sq[..., None, :])
    Dl = G / pre
    dg = Dl.diagonal(dim1=-2, dim2=-1)
    O = O + Dl - torch.diag_embed(dg)
    return lam + dg, O, O.to(LOWP), Dl.square().sum((-2, -1)).sqrt()


def _sqrt_start(lam, O):
    """Square-root stage start: Z = Y^(1/2) = diag(z) + Q for near-diagonal Y = diag(lam) + O (O zero-diagonal)."""
    z = lam.clamp_min(1e-30).sqrt()
    Q = O / (z[..., :, None] + z[..., None, :])
    return z, Q, Q.to(LOWP), Q.square().sum((-2, -1)).sqrt()


def _sqrt_post(lam, O, z, Q, QQ):
    """One fixed-point update of Z^2 = Y: z_i^2 + (Q^2)_ii = lam_i, (z_i + z_j) Q_ij + (Q^2)_ij = O_ij (i != j)."""
    z2 = (lam - QQ.diagonal(dim1=-2, dim2=-1)).clamp_min(1e-30).sqrt()
    Qn = (O - QQ) / (z2[..., :, None] + z2[..., None, :])
    Qn = Qn - torch.diag_embed(Qn.diagonal(dim1=-2, dim2=-1))
    upd = ((Qn - Q).square().sum((-2, -1)) + (z2 - z).square().sum(-1)).sqrt()
    return z2, Qn, Qn.to(LOWP), upd


start = torch.compile(_start, dynamic=False)
iter_mid = torch.compile(_iter_mid, dynamic=False)
iter_w = torch.compile(_iter_w, dynamic=False)
iter_post = torch.compile(_iter_post, dynamic=False)
sqrt_start = torch.compile(_sqrt_start, dynamic=False)
sqrt_post = torch.compile(_sqrt_post, dynamic=False)


def _bmm32(a, b):
    return torch.bmm(a, b, out_dtype=torch.float32) if a.dim() == 3 else torch.mm(a, b, out_dtype=torch.float32)


def _iterate(d, E, Eb, lam, O, Ob, sq, lam0, iters):
    sym = E.size(-1) % 256 == 0 and E.dim() == 3
    dn = []
    for _ in range(iters):
        P, Pb = iter_mid(lam, O, mm_sym(Ob, Ob) if sym else _bmm32(Ob, Ob))
        EP = _bmm32(Eb, Pb)
        W = iter_w(P, d, EP)
        PW = mm_sym(Pb, W) if sym else _bmm32(Pb, W)
        lam, O, Ob, upd = iter_post(lam, O, d, E, P, EP, PW, lam0, sq, sym)
        dn.append(upd)
    return lam, O, Ob, dn


def _estimate(lam, dprev, dlast):
    """Relative Frobenius error after linear convergence at rate r = dlast / dprev (capped at 0.95)."""
    r = (dlast / dprev.clamp_min(1e-30)).clamp_max(0.95)
    return r / (1 - r) * dlast / lam.square().sum(-1).sqrt()


def froot_state(B, iters):
    """Y = (B + eps I)^(-1/4) = diag(lam) + O: lam, O, rho, est (relative error of Y) and the continuation state."""
    d, E, Eb, lam, O, Ob, sq, rho, dn0 = start(B)
    lam0 = lam
    lam, O, Ob, dn = _iterate(d, E, Eb, lam, O, Ob, sq, lam0, iters)
    dn = [dn0] + dn
    est = _estimate(lam, dn[-2], dn[-1]) if iters >= 1 else torch.full_like(rho, float("inf"))
    return lam, O, rho, est, (lam, O, dn[-1])


def froot_more(B, state, iters):
    lam, O, dlast = state
    d, E, Eb, _, _, _, sq, rho, _ = start(B)
    lam0 = d.clamp_min(EPS_PD * 1e-3).pow(-0.25)
    lam, O, Ob, dn = _iterate(d, E, Eb, lam, O, O.to(LOWP), sq, lam0, iters)
    dn = [dlast] + dn
    return lam, O, rho, _estimate(lam, dn[-2], dn[-1]), dn[-1]


def _scaled_split(Q):
    """[hi | lo] and [lo ; hi] FP16 splits of Q 2^k (max |Q 2^k| in [2^13, 2^14): no FP16 underflow), and 2^-2k."""
    a = Q.abs().amax((-2, -1), keepdim=True).clamp_min(1e-30)
    s = torch.exp2(13.0 - torch.floor(torch.log2(a)))
    Qs = Q * s
    return _cols_hi(Qs), _rows_lo(Qs), 1.0 / (s * s)


scaled_split = torch.compile(_scaled_split, dynamic=False, options=_EXACT)


def _qq(Q, Qb):
    """Q Q for the square-root stage: FP16 inputs (--pd-sqrt-prec fp16) or FP32-accurate splits (split)."""
    sym = Q.size(-1) % 256 == 0 and Q.dim() == 3
    A2, B2, inv = scaled_split(Q)
    return (mm16_sym(A2, B2, False) if sym else mm16(A2, B2, False)) * inv


def sqrt_stages(lam, O, est, stages, iters):
    """alpha 1/8 (1 stage) or 1/16 (2): Z = Y^(1/2) (repeated). est becomes the relative error of the result: halved
    per stage (dZ/Z ~ dY/(2Y)) plus each stage's own convergence estimate. Also returns the stages' own estimates."""
    own = []
    for _ in range(stages):
        z, Q, Qb, d0 = sqrt_start(lam, O)
        dn = [d0]
        for _ in range(iters):
            z, Q, Qb, upd = sqrt_post(lam, O, z, Q, _qq(Q, Qb))
            dn.append(upd)
        e = _estimate(z, dn[-2], dn[-1])
        own.append(e)
        est = 0.5 * est + e
        lam, O = z, Q
    return lam, O, est, own


def floor_base(lam, alpha):
    """The exact rule's FP32 floor scale for root power alpha from eigenvalues lam (..., n) (pd_roots.floor_base)."""
    x = lam.double().clamp_min(0)
    x = x / x.mean(-1, keepdim=True).clamp_min(1e-30)
    f = (x + EPS_PD).pow(-alpha)
    fp = alpha * (x + EPS_PD).pow(-alpha - 1)
    return (U32 * x.amax(-1) * fp.norm(dim=-1) / f.norm(dim=-1)).float()


def exact_eigh(C, alpha):
    """The exact rule (FP32 eigh): eigenvectors, root (C / mean eig + eps)^-alpha, eigenvalues."""
    try:
        lam, V = torch.linalg.eigh(0.5 * (C + C.mT))
    except RuntimeError:  # FP32 eigh can fail to converge: FP64, as the exact reference
        lam, V = torch.linalg.eigh(0.5 * (C + C.mT).double())
        lam, V = lam.float(), V.float()
    x = lam.clamp_min(0)
    x = x / x.mean(-1, keepdim=True)
    f = (x + EPS_PD).pow(-alpha)
    if C.size(-1) % 256 == 0 and C.dim() == 3:
        R = mm16_sym(cols_hi(V * f[..., None, :]), rows_lo(V.mT.contiguous()), False)
    else:
        R = mm16(cols_hi(V * f[..., None, :]), rows_lo(V.mT.contiguous()), False)
        R = 0.5 * (R + R.mT)
    return V, R, lam

# Statistic kinds (a hidden matrix's input): qkv = attention input (q, k, v), o = attention output (attn.proj),
# mlp = MLP input (mlp.fc), hidden = relu^2 activation (mlp.proj). A matrix is owned by its statistic's owner.
KIND_MATS = {"qkv": ("attn.q", "attn.k", "attn.v"), "o": ("attn.proj",), "mlp": ("mlp.fc",), "hidden": ("mlp.proj",)}
GROUPS = (("fc", "mlp"), ("proj", "hidden"), ("qkv", "qkv"), ("o", "o"))  # (SOAP-Muon group, statistic kind)


def stat_owner(layer, kind, world, layers=12):
    """Balanced owners. world <= 4: contiguous blocks of layers. world 8: ranks b and b + 4 share layers 3b..3b+2;
    each rank gets 3 MLP and 6 attention matrices (equal parameter counts); rank b owns 2 MLP-hidden statistics."""
    if world <= 4:
        return layer // (layers // world)
    assert world == 8 and layers == 12
    b, j = divmod(layer, 3)
    if kind == "hidden":
        return b if j < 2 else b + 4
    if kind == "mlp":
        return b if j == 2 else b + 4
    if kind == "qkv":
        return b if j == 0 else b + 4
    return b


class MatGroup:
    """One owned SOAP-Muon group: k same-shape matrices as views of the flat parameter / gradient buffers."""

    def __init__(self, name, kind, mats, P, G):
        self.name, self.kind, self.mats = name, kind, mats  # mats: [(layer, param name)], in buffer order
        self.P, self.G = P, G  # (k, m, n) views
        k, m, n = P.shape
        dev = P.device
        self.mom = torch.zeros(k, m, n, device=dev)
        self.exp_avg_sq = torch.zeros(k, m, n, device=dev)
        self.row_gg = torch.zeros(k, m, m, device=dev)
        self.col_gg = torch.zeros(k, n, n, device=dev)
        self.q_row = torch.zeros(k, m, m, device=dev)
        self.q_col = torch.zeros(k, n, n, device=dev)
        self.soap_ready = False
        self.Gw = torch.zeros(k, m, n, device=dev)  # the gradient in the inner root's coordinates (SOAP's input)


class FastSoapMuon:
    """#46's Muon (SOAP-Muon) on the owners, batched by group; with PD, in PD's whitened coordinates."""

    def __init__(self, model, world, rank, device):
        self.world, self.rank, self.device = world, rank, device
        L = len(model.blocks)
        named = dict(model.blocks.named_parameters())
        # per rank: owned matrices in buffer order (groups fc, proj, qkv, o; layers ascending; q, k, v)
        self.layout = []  # per rank: [(group name, kind, [(layer, name)])]
        for r in range(world):
            gl = []
            for gname, kind in GROUPS:
                mats = [(l, f"{l}.{m}.weight") for l in range(L) if stat_owner(l, kind, world, L) == r
                        for m in KIND_MATS[kind]]
                gl.append((gname, kind, mats))
            self.layout.append(gl)
        sizes = [sum(named[n].numel() for _, _, mats in gl for _, n in mats) for gl in self.layout]
        assert len(set(sizes)) == 1, f"unbalanced owners {sizes}"
        self.S = sizes[0]
        self.flat_p = torch.empty(world * self.S, device=device)
        self.flat_g = torch.zeros(world * self.S, device=device)
        self.params, self.grad_views, self.groups = [], [], []
        for r, gl in enumerate(self.layout):
            off = r * self.S
            for gname, kind, mats in gl:
                if not mats:
                    continue
                m, n = named[mats[0][1]].shape
                k = len(mats)
                Pv = self.flat_p[off:off + k * m * n].view(k, m, n)
                Gv = self.flat_g[off:off + k * m * n].view(k, m, n)
                for j, (_, name) in enumerate(mats):
                    p = named[name]
                    Pv[j].copy_(p.data)
                    p.data = Pv[j]
                    self.params.append(p)
                    self.grad_views.append(Gv[j])
                if r == rank:
                    self.groups.append(MatGroup(gname, kind, mats, Pv, Gv))
                off += k * m * n
        self.param_groups = [dict(params=self.params, lr=MUON_LR, mu=MU)]
        self.step_count = 0
        self.soap_graphs = {}  # by QR method
        self.soap_warm = set()
        self.qr_method = "householder"
        self.cqr_defect = torch.zeros((), device=device)  # max orthogonality defect of Cholesky-QR bases so far
        self.soap_stream = torch.cuda.Stream(device=device)
        self.qr_streams = [torch.cuda.Stream(device=device) for _ in range(QR_STREAMS)]
        self.soap_done = None  # event: last step's basis update finished

    # -------------------------------------------------------------- the step (owners)
    @torch.no_grad()
    def step(self, roots):
        """roots: Toner's outer root R = (C / mean eig + eps I)^(-1/8) per (layer, kind) owned by this rank."""
        lr, mu = self.param_groups[0]["lr"], self.param_groups[0]["mu"]
        if self.soap_done is not None:
            torch.cuda.current_stream().wait_event(self.soap_done)  # the bases of this step (see soap_update)
        shape_decay = SHAPE_DECAY_RULE(self.step_count, lr / self.param_groups[0]["initial_lr"])
        rf = torch.profiler.record_function
        for gr in self.groups:
            G = gr.G
            gr.mom.lerp_(G, 1 - mu)
            momentum_update = G.lerp(gr.mom, mu)
            with rf("toner_in"):  # SOAP-Muon in the inner root's coordinates: U R^4, G R^4
                Rs = torch.stack([roots[(l, gr.kind)] for l, _ in gr.mats])
                R2 = rows_lo(Rs)
                R2i = rows_lo(inner_root(Rs))
                momentum_update = apply16(momentum_update, R2i)
                gr.Gw.copy_(apply16(G, R2i))
                grad = gr.Gw
            with rf("soap_precondition"):
                if gr.name in ("qkv", "o"):
                    soap_update = soap_precondition_momentum(momentum_update, gr)
                    if gr.name == "o":
                        gate = bounded_trust_gate(trust_gate(momentum_update, soap_update, grad), self.step_count)
                    else:
                        gate = torch.ones(len(gr.mats), dtype=torch.float32, device=G.device)
                    momentum_update = norm_preserving_blend(momentum_update, soap_update, gate)
                else:
                    momentum_update = soap_precondition_momentum(momentum_update, gr)
            with rf("newton_schulz"):
                update = muon_update(momentum_update)
            with rf("toner_out"):  # back through the outer root, at the update's Frobenius norm: O R
                Of = update.float()
                OR = apply16(Of, R2)
                update = (OR * (Of.norm(dim=(-2, -1), keepdim=True)
                                / OR.norm(dim=(-2, -1), keepdim=True).clamp_min(1e-30))).to(update.dtype)
            with rf("post_update"):
                post_update_(gr.P, update, lr)
                if shape_decay:
                    pd_shape_decay_(gr.P, Rs, lr * shape_decay)
        self.step_count += 1

    def gather_params(self):
        """Every rank's updated matrices to all ranks: one in-place all-gather of the flat FP32 buffer."""
        if self.world > 1:
            dist.all_gather_into_tensor(self.flat_p, self.flat_p[self.rank * self.S:(self.rank + 1) * self.S])

    # -------------------------------------------------------------- SOAP basis update (after the step)
    def _soap_stats(self):
        for gr in self.groups:
            g = gr.Gw if gr.Gw is not None else gr.G
            gr.row_gg.lerp_(g @ g.mT, 1 - SOAP_BETA2)
            gr.col_gg.lerp_(g.mT @ g, 1 - SOAP_BETA2)

    def _soap_qr_work(self):
        """The sorted QR step of every owned basis (#46's soap_basis_qr, batched; the Rayleigh quotients come from the
        same product G Q that feeds the QR). QRs spread over side streams, largest first."""
        jobs = []
        main = torch.cuda.current_stream()
        for gr in self.groups:
            for side in ("row", "col"):
                gg, q = (gr.row_gg, gr.q_row) if side == "row" else (gr.col_gg, gr.q_col)
                k, n, _ = gg.shape
                GQ = gg @ q  # #46's sort keys diag(Q^T gg Q) and gg Q[:, order], from one product
                eig = (q * GQ).sum(-2)
                order = torch.argsort(eig, dim=-1, descending=True)
                A = torch.gather(GQ, -1, order[:, None, :].expand(k, n, n))
                if side == "row":
                    gr.exp_avg_sq.copy_(torch.gather(gr.exp_avg_sq, -2, order[:, :, None].expand_as(gr.exp_avg_sq)))
                else:
                    gr.exp_avg_sq.copy_(torch.gather(gr.exp_avg_sq, -1, order[:, None, :].expand_as(gr.exp_avg_sq)))
                for j in range(k):
                    jobs.append((n, A[j], q[j]))
        jobs.sort(key=lambda x: -x[0])
        if self.qr_method == "cqr":  # the small QRs as one batched FP64 shifted Cholesky-QR3 (GEMM-like kernels)
            small = [(A, q) for n, A, q in jobs if n <= CQR_MAX_N]
            jobs = [j for j in jobs if j[0] > CQR_MAX_N]
            if small:
                Qs, defect = scqr3(torch.stack([A for A, _ in small]))
                torch._foreach_copy_([q for _, q in small], list(Qs.unbind(0)))
                torch.maximum(self.cqr_defect, defect, out=self.cqr_defect)
        ev = torch.cuda.Event()
        ev.record(main)
        for s in self.qr_streams:
            s.wait_event(ev)
        for i, (_, A, q) in enumerate(jobs):
            s = self.qr_streams[i % len(self.qr_streams)]
            with torch.cuda.stream(s):
                q.copy_(torch.linalg.qr(A)[0])
        for s in self.qr_streams:
            main.wait_stream(s)

    @torch.no_grad()
    def soap_update(self):
        """After the step: SOAP statistics and basis update for the next step, on a side stream (CUDA graph from the
        third step on). Step 0: exact eigh bases (#46's first step)."""
        main = torch.cuda.current_stream()
        if not all(gr.soap_ready for gr in self.groups):
            self._soap_stats()
            for gr in self.groups:  # #46: the first basis is an eigh of the first statistics
                for j in range(len(gr.mats)):
                    gr.q_row[j].copy_(soap_eigenbasis(gr.row_gg[j]))
                    gr.q_col[j].copy_(soap_eigenbasis(gr.col_gg[j]))
                gr.soap_ready = True
            self.soap_done = None
            return
        # Householder (cuSOLVER, as #46) for the first steps, where the statistics are rank-deficient; then Cholesky-QR
        self.qr_method = "cqr" if self.step_count > CQR_START else "householder"
        self.soap_stream.wait_stream(main)
        with torch.cuda.stream(self.soap_stream):
            if self.qr_method not in self.soap_warm:
                self._soap_stats()
                self._soap_qr_work()
                self.soap_warm.add(self.qr_method)
            else:
                if self.qr_method not in self.soap_graphs:
                    g = torch.cuda.CUDAGraph()  # (thread_local: PD's host-refresh threads make CUDA calls)
                    with torch.cuda.graph(g, stream=self.soap_stream, capture_error_mode="thread_local"):
                        self._soap_stats()
                        self._soap_qr_work()
                    self.soap_graphs[self.qr_method] = g
                self.soap_graphs[self.qr_method].replay()
            self.soap_done = torch.cuda.Event(enable_timing=True)
            self.soap_done.record(self.soap_stream)

    def check_cqr(self):
        """Raise if a Cholesky-QR basis lost orthogonality (checked at validation: no per-step host sync)."""
        d = self.cqr_defect.item()
        if not d < CQR_TOL:
            raise RuntimeError(f"Cholesky-QR basis orthogonality defect {d:.3e}")
        return d

class StatGroup:
    """Statistics of one size, ordered by owner (each rank's owned ones contiguous). The partial Grams are laid out in
    equal per-rank slots (padded with zero Grams where a rank owns fewer), so the reduce-scatter is one even
    reduce_scatter_tensor: the uneven form runs as coalesced per-rank reduces with NCCL's low-latency protocol, which
    held SMs ~14 ms per call on 8 H100s (profile, 10-14)."""

    def __init__(self, keys, n, bufs, counts, rank, device):
        self.keys, self.n, self.bufs, self.counts = keys, n, bufs, counts
        self.per = counts[rank]
        lo = sum(counts[:rank])
        self.own = keys[lo:lo + self.per]
        self.offs = None
        m = n * (n + 1) // 2
        self.slots = max(counts)
        slot = [r * self.slots + i for r, c in enumerate(counts) for i in range(c)]  # keys' order is by owner
        self.oidx = torch.tensor(slot, dtype=torch.int64, device=device)
        self.partial = torch.zeros(len(counts) * self.slots, m, device=device)  # padding slots stay zero
        self.summed = torch.empty(self.slots, m, device=device)
        self.C = torch.zeros(self.per, n, n, device=device)
        self.basis = None
        self.basis_step = [-1] * self.per
        self.pending = [None] * self.per
        self.host_in = torch.empty(self.per, n, n, pin_memory=True)
        self.host_out = torch.empty(self.per, n, n, pin_memory=True)
        self.host_tol = torch.empty(self.per, pin_memory=True)
        self.tol = torch.full((self.per,), 1e-4, device=device)
        self.stage = torch.empty(self.per, n, n, device=device)
        self.graph = None
        self.warm = False


class FastPD:
    """PD statistics and every-step roots (pd_fast.py's design), for the S∘PD step of FastSoapMuon."""

    def __init__(self, blocks, world, rank, device, rows_total):
        self.world, self.rank, self.device, self.rows_total = world, rank, device, rows_total
        self.alpha = PD_ALPHA
        self.stages = 1  # one square-root stage: B^(-1/4) -> the 1/8 root
        L = len(blocks)
        self.groups = {}
        for gname, kinds, n in (("d", ("qkv", "o", "mlp"), blocks[0].pd_rows("qkv").size(1)),
                                ("f", ("hidden",), blocks[0].pd_rows("hidden").size(1))):
            keys = sorted([(l, k) for l in range(L) for k in kinds], key=lambda x: (stat_owner(x[0], x[1], world, L),) + x)
            counts = [sum(1 for (l, k) in keys if stat_owner(l, k, world, L) == r) for r in range(world)]
            bufs = [blocks[l].pd_rows(k) for l, k in keys]
            self.groups[gname] = StatGroup(keys, n, bufs, counts, rank, device)
        self.weight = 0.0
        self.steps = 0
        self.fallbacks = 0
        self.extra = 0
        self.main_stream = torch.cuda.current_stream(device)
        self.side = torch.cuda.Stream(device=device, priority=-1)
        self.graph_pool = torch.cuda.graph_pool_handle()
        self.pg = dist.new_group(backend="nccl") if world > 1 else None
        self.flags_host = torch.zeros(sum(g.per for g in self.groups.values()), dtype=torch.uint8, pin_memory=True)
        self.cand = None
        self.ev_flags = self.ev_done = None

        def init_worker():
            torch.cuda.set_device(device)
        self.pool = concurrent.futures.ThreadPoolExecutor(max_workers=PD_HOST_WORKERS,
                                                          initializer=init_worker)

    def _tolerance(self, lam):
        return (PD_TOL_FRAC * floor_base(lam, self.alpha)).clamp_min(PD_TOL_MIN)

    # ------------------------------------------------------------------ statistics
    def _stats(self):
        works = []
        for g in self.groups.values():
            if g.offs is None:
                g.offs = list_offsets(g.bufs)
            syrk_list_packed(g.bufs, g.offs, g.partial, g.oidx)
            if self.world == 1:
                g.summed.copy_(g.partial)
            else:
                works.append(dist.reduce_scatter_tensor(g.summed, g.partial, group=self.pg, async_op=True))
        for w in works:
            w.wait()
        b = PD_DECAY
        for g in self.groups.values():
            if g.per:
                unpack_ema_(g.summed[:g.per], g.C, b, (1 - b) / self.rows_total)
        self.weight = b * self.weight + (1 - b)

    # ------------------------------------------------------------------ bases (host)
    def _host_refresh(self, x_in, x_out, tol_out, copied):
        torch.set_num_threads(PD_HOST_THREADS)
        copied.synchronize()
        lam, V = torch.linalg.eigh(0.5 * (x_in + x_in.mT))
        x_out.copy_(V)
        tol_out.copy_(self._tolerance(lam))

    def _submit_refresh(self, g, j):
        g.host_in[j].copy_(g.C[j], non_blocking=True)
        copied = torch.cuda.Event(blocking=True)
        copied.record(self.side)
        g.pending[j] = (self.pool.submit(self._host_refresh, g.host_in[j], g.host_out[j], g.host_tol[j], copied),
                        self.steps)

    def _install_ready(self):
        for g in self.groups.values():
            if not g.per:
                continue
            ready = [j for j in range(g.per) if g.pending[j] is not None and g.pending[j][0].done()]
            if not ready:
                continue
            mask = [False] * g.per
            for j in ready:
                g.pending[j][0].result()
                if g.pending[j][1] >= g.basis_step[j]:
                    g.stage[j].copy_(g.host_out[j], non_blocking=True)
                    g.tol[j].copy_(g.host_tol[j], non_blocking=True)
                    mask[j] = True
                    g.basis_step[j] = g.pending[j][1]
                g.pending[j] = None
            if any(mask):
                g.basis.install_many(g.stage, torch.tensor(mask).pin_memory().to(self.device, non_blocking=True))

    # ------------------------------------------------------------------ roots
    def _roots(self, g):
        """Candidates R, B, flags, the 1/4 iteration's state and diagnostics (rho, est of Y, est of R, each square-root
        stage's own estimate)."""
        B, _ = basis_transform(g.C, g.basis)
        lam, O, rho, est, state = froot_state(B, PD_ITERS)
        lam_r, O_r, est_r, own = sqrt_stages(lam, O, est, self.stages, PD_SQRT_ITERS)
        bad = (rho > PD_RHO_MAX) | ~(est_r <= g.tol) | ~torch.isfinite(lam_r).all(-1)
        return form(g.basis, lam_r, O_r), B, bad, state, torch.stack([rho, est, est_r] + own)

    def _roots_graphed(self, g):
        if not g.warm:
            g.warm = True
            return self._roots(g)
        if g.graph is None:
            graph = torch.cuda.CUDAGraph()
            with torch.cuda.graph(graph, pool=self.graph_pool, stream=self.side, capture_error_mode="thread_local"):
                out = self._roots(g)
            g.graph = (graph,) + out
        g.graph[0].replay()
        return g.graph[1:]

    @torch.no_grad()
    def begin(self):
        """Right after the step's last forward: statistics and root candidates on the side stream (overlapping the
        backward)."""
        ev_fwd = torch.cuda.Event()
        ev_fwd.record(self.main_stream)
        self.cand, flags = {}, []
        with torch.cuda.stream(self.side):
            self.side.wait_event(ev_fwd)
            if all(g.basis is not None for g in self.groups.values() if g.per):
                self._install_ready()
            self._stats()
            for name, g in self.groups.items():
                if not g.per:
                    self.cand[name] = (torch.empty(0, g.n, g.n, device=self.device), None,
                                       torch.zeros(0, dtype=torch.bool, device=self.device), None, None)
                elif g.basis is None or self.steps < PD_INLINE:
                    V, R, lam_x = exact_eigh(g.C, self.alpha)
                    g.tol.copy_(self._tolerance(lam_x))
                    if g.basis is None:
                        g.basis = Basis(V)
                    else:
                        for j in range(g.per):
                            g.basis.install(j, V[j])
                    g.basis_step = [self.steps] * g.per
                    g.pending = [None] * g.per
                    self.cand[name] = (R, None, torch.zeros(g.per, dtype=torch.bool, device=self.device), None, None)
                else:
                    self.cand[name] = self._roots_graphed(g)
                flags.append(self.cand[name][2])
            self.flags_host.copy_(torch.cat(flags).to(torch.uint8), non_blocking=True)
            self.ev_flags = torch.cuda.Event()
            self.ev_flags.record(self.side)
            for g in self.groups.values():  # host refreshes due this step (staggered)
                for j in range(g.per):
                    if g.pending[j] is None and (self.steps + j) % PD_REFRESH == 0 \
                            and g.basis_step[j] != self.steps:
                        self._submit_refresh(g, j)
            self.ev_done = torch.cuda.Event()
            self.ev_done.record(self.side)
        for name, (R, B, _, _, _) in self.cand.items():
            if self.groups[name].graph is None:
                R.record_stream(self.main_stream)
                if B is not None:
                    B.record_stream(self.main_stream)

    @torch.no_grad()
    def finish(self):
        """Flagged roots: more iterations, then the exact fallback. Returns {(layer, kind): R} for the owned
        statistics; the main stream then waits for them."""
        self.ev_flags.synchronize()
        flags = self.flags_host.tolist()
        roots, o, todo = {}, 0, []
        for name, g in self.groups.items():
            R, B, _, state, diag = self.cand[name]
            bad = [j for j in range(g.per) if flags[o + j]]
            o += g.per
            if bad:
                todo.append((g, R, B, state, bad))
            for j, key in enumerate(g.own):
                roots[key] = R[j]
        if todo:
            with torch.cuda.stream(self.side):
                for g, R, B, state, still in todo:
                    flagged = list(still)
                    cur, hopeless = None, []
                    for _ in range(PD_REPAIR_ROUNDS if state is not None else 0):
                        k = 1 << (len(still) - 1).bit_length()
                        pad = still + still[-1:] * (k - len(still))
                        idx = torch.tensor(pad, device=self.device)
                        if cur is None:
                            st = tuple(t.index_select(0, idx) for t in state)
                        else:
                            pos = torch.tensor([cur[0].index(j) for j in pad], device=self.device)
                            st = tuple(t.index_select(0, pos) for t in cur[1])
                        Bi = B.index_select(0, idx)
                        lam, O, rho, est, dl = froot_more(Bi, st, 2)
                        lam_r, O_r, est_r, _ = sqrt_stages(lam, O, est, self.stages, PD_SQRT_ITERS)
                        R2 = form(Basis.of_splits(g.basis.S2.index_select(0, idx)), lam_r, O_r)
                        okl = ((rho <= PD_RHO_MAX) & (est_r <= g.tol.index_select(0, idx))
                               & torch.isfinite(lam_r).all(-1)).tolist()
                        rhok = (rho <= PD_RHO_MAX).tolist()
                        for i, j in enumerate(still):
                            if okl[i]:
                                R[j].copy_(R2[i])
                        self.extra += sum(okl[:len(still)])
                        hopeless += [j for i, j in enumerate(still) if not okl[i] and not rhok[i]]
                        cur = (pad, (lam, O, dl))
                        still = [j for i, j in enumerate(still) if not okl[i] and rhok[i]]
                        if not still:
                            break
                    still = still + hopeless
                    for j in still:
                        self.fallbacks += 1
                        V, Rx, lam_x = exact_eigh(g.C[j:j + 1], self.alpha)
                        g.tol[j:j + 1].copy_(self._tolerance(lam_x))
                        R[j].copy_(Rx[0])
                        g.basis.install(j, V[0])
                        g.basis_step[j] = self.steps
                    for j in flagged:  # converged from an old basis: refresh that basis now
                        if j not in still and g.pending[j] is None:
                            self._submit_refresh(g, j)
            self.main_stream.wait_stream(self.side)
        else:
            self.main_stream.wait_event(self.ev_done)
        self.steps += 1
        return roots

class Adam:
    """#46's bias-correction-free Adam for the aux (<2D) groups, with multi-tensor kernels."""

    def __init__(self, groups, eps=1e-10):
        self.param_groups = groups
        self.eps = eps
        for g in groups:
            g["exp_avg"] = [torch.zeros_like(p) for p in g["params"]]
            g["exp_avg_sq"] = [torch.zeros_like(p, dtype=torch.float32) for p in g["params"]]

    @torch.no_grad()
    def step(self):
        for g in self.param_groups:
            beta1, beta2 = g["betas"]
            grads = [p.grad for p in g["params"]]
            torch._foreach_lerp_(g["exp_avg"], grads, 1 - beta1)
            torch._foreach_lerp_(g["exp_avg_sq"], torch._foreach_mul(grads, grads), 1 - beta2)
            den = torch._foreach_sqrt(g["exp_avg_sq"])
            torch._foreach_add_(den, self.eps)
            torch._foreach_add_(g["params"], torch._foreach_div(g["exp_avg"], den), alpha=-g["lr"])


class EMANesterov:
    """#46's EMA_Nesterov outer wrapper, with multi-tensor kernels."""

    def __init__(self, params, lr_source, lookahead_stepsize=0.3, lookahead_ema=0.99, prefill_steps=0, rest_steps=0):
        self.lr_source = lr_source  # AdamW's embedding group: lookahead scale lr / initial_lr
        self.lookahead_stepsize, self.lookahead_ema = lookahead_stepsize, lookahead_ema
        self.prefill_steps, self.rest_steps = prefill_steps, rest_steps
        self.it = 0
        by_dtype = {}  # one list per dtype: the multi-tensor kernels' fast path needs a single dtype
        for p in params:
            by_dtype.setdefault(p.dtype, []).append(p)
        self.lists = [(ps, [p.detach().clone() for p in ps], [torch.zeros_like(p) for p in ps])
                      for ps in by_dtype.values()]

    @torch.no_grad()
    def nesterov_step(self):
        if self.it + 1 > self.prefill_steps and self.it < self.rest_steps:
            s = self.lookahead_stepsize * (self.lr_source["lr"] / self.lr_source["initial_lr"])
            for ps, _, buf in self.lists:
                torch._foreach_add_(ps, buf, alpha=s)

    @torch.no_grad()
    def accum_lookahead(self):
        for ps, prev, buf in self.lists:
            look = torch._foreach_add(ps, prev, alpha=-1)
            torch._foreach_lerp_(buf, look, 1 - self.lookahead_ema)
            torch._foreach_copy_(prev, ps)
        self.it += 1

def headmean_antithetic_pair(shape, *, device, layer_idx: int, seed: int, head_dim: int = HEAD_DIM):
    width = shape[-1]
    if width % head_dim != 0:
        s0 = (torch.randint(0, 2, shape, device=device, dtype=torch.float32) * 2 - 1)
        return s0, -s0
    gen = torch.Generator(device=device)
    gen.manual_seed(0xA9170000 + seed * 1009 + layer_idx * 9176)
    heads0 = []
    heads1 = []
    for _ in range(width // head_dim):
        h0 = (torch.randint(0, 2, (head_dim,), device=device, generator=gen, dtype=torch.int64) * 2 - 1).float()
        plus0 = int((h0 > 0).sum().item())
        plus0 = int(round(head_dim / 2 + _CGI_HEAD_MEAN_SHRINK * (plus0 - head_dim / 2)))
        plus0 = max(0, min(head_dim, plus0))
        h0 = torch.cat([
            torch.ones(plus0, device=device, dtype=torch.float32),
            -torch.ones(head_dim - plus0, device=device, dtype=torch.float32),
        ])
        h0 = h0[torch.randperm(head_dim, device=device, generator=gen)]
        plus1 = head_dim - plus0
        h1 = torch.cat([
            torch.ones(plus1, device=device, dtype=torch.float32),
            -torch.ones(head_dim - plus1, device=device, dtype=torch.float32),
        ])
        h1 = h1[torch.randperm(head_dim, device=device, generator=gen)]
        heads0.append(h0)
        heads1.append(h1)
    return torch.cat(heads0).reshape(shape), torch.cat(heads1).reshape(shape)

def _lr(step, initial_lr, power_c, power=1.0):
    t_end = FINAL_SCHEDULE_STEPS
    flat_lr = initial_lr
    downward_lr = power_c * max(0.0, t_end - step) ** power
    return min(flat_lr, downward_lr)

_MU_MIN = 0.85
_MU_MAX = 0.95
_MU_WARMUP_STEPS = 300
_MU_COOLDOWN_STEPS = 200

def _muon_mu_at_step(step, train_steps):
    cd_start = train_steps - _MU_COOLDOWN_STEPS
    if step < _MU_WARMUP_STEPS:
        frac = step / max(_MU_WARMUP_STEPS, 1)
        return _MU_MIN + frac * (_MU_MAX - _MU_MIN)
    elif step > cd_start:
        frac = (step - cd_start) / max(_MU_COOLDOWN_STEPS, 1)
        return _MU_MAX - frac * (_MU_MAX - _MU_MIN)
    else:
        return _MU_MAX

def set_hparams(step):
    progress = step / FINAL_SCHEDULE_STEPS
    assert 0 <= progress
    mu = _muon_mu_at_step(step, FINAL_TRAIN_STEPS)
    for opt in (optimizer1, optimizer2, optimizer3):
        for group in opt.param_groups:
            group["lr"] = _lr(step, group["initial_lr"], group["power_c"], FINAL_LR_POWER)
    for group in optimizer2.param_groups:
        group["mu"] = mu
