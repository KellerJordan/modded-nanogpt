"""
train_gpt_cwd_SOTA.py

This file descends from the [NanoGPT speedrun](https://github.com/KellerJordan/modded-nanogpt).

SOTA optimizer for a PR. It is the clean SOAP-Muon base from PR #321 (the 2750/2755
"aux-b2 + SOAP-f1 clean" record) with THREE additional levers stacked on top, each on a
different axis chosen to survive the radius pin (which re-pins global Frobenius magnitude
every step, so only directional / shape / readout changes persist):

  (A) Tail-EMA eval readout (PR #325): an eval-time-only weight blend. Over the cooldown
      tail [2400, 2900] maintain a slow EMA  ema += (w - ema) / TAU  (TAU=150) of every
      parameter except the token embedding; at validation, evaluate the partial blend
      w_eval = (1-L)*w + L*ema  (L=0.6). Training is untouched; only the weights used for
      the val forward change. The partial (not full) blend cancels the late cross-valley
      oscillation while still tracking the descending floor.

  (B) RowFloor (per-output-row u/w-floor): replace the SCALAR u/w-floor with a per-row
      floor on the orthogonalized update -- each output row whose update norm is below
      TARGET_UW * ||row|| is lifted to that target (RHO=1.0, no Frobenius renorm). A
      per-row SHAPE change, so it survives the radius pin.

      anisotropic per-coordinate decay  p *= 1 - lr*CWD*mask  with mask = 1[update*p > 0]
      (only the coords the optimizer step already shrinks; never fights coords it wants to
      grow), applied AFTER rescale_to_radius so the per-coord SHAPE change is not
      re-normalized away.

The clean PR #321 base retains:
  - SOAP preconditioning on all hidden matrices (all_hidden, freq=1)
  - u/w-floor hyperball constraint (TARGET_UW=0.3825); here per-row (RowFloor)
  - radial scaling + rescale-to-radius (the radius pin)
  - EMA-Nesterov outer wrapper
  - PowerCool LR schedule, mu schedule, val schedule
  - aux-Adam beta2 split, depth-scaled mlp.fc init

All hyperparameters are hardcoded; only --seed is a command-line argument. Set CWD=0.0,
ROWFLOOR=False and TAILEMA_TAU=0 below to recover the clean PR #321 base.
"""

def _mtt3_snap_dense_triple(spec: str):
    """Parse MTT3_SNAP_DENSE 'lo:hi:every' -> (lo, hi, every) or None when unset."""
    if not spec:
        return None
    lo, hi, every = (int(x) for x in spec.split(":"))
    assert every > 0 and hi >= lo, f"bad MTT3_SNAP_DENSE: {spec!r}"
    return lo, hi, every


_MTT3_SNAP_DENSE_TRIPLE = _mtt3_snap_dense_triple(MTT3_SNAP_DENSE)


def _mtt3_snap_hit(step: int) -> bool:
    """Snapshot cadence: the regular (MTT3_SNAP_START, MTT3_SNAP_EVERY) rule UNION
    the optional MTT3_SNAP_DENSE window. Only consulted when MTT3_SNAP_EVERY > 0."""
    if step >= MTT3_SNAP_START and step % MTT3_SNAP_EVERY == 0:
        return True
    if _MTT3_SNAP_DENSE_TRIPLE is not None:
        lo, hi, every = _MTT3_SNAP_DENSE_TRIPLE
        if lo <= step <= hi and (step - lo) % every == 0:
            return True
    return False

# >>> mtt3-gcr-pure >>>  (verbatim H3 ports from analysis/eval_readouts.py; pure torch,
#     no module-global state. Any change here must be mirrored in the evaluator and
#     re-pass the tests/test n parity gate.)
def _mtt3_gcr_tailema_weights(ksteps, tau: float, tail_start: int, tail_end: int):
    """Port of eval_readouts.tailema_weights: per-snapshot causal Tail-EMA weights
    a_k (lazy init at the first in-window snapshot, gap-corrected alpha; in-window
    weights sum to 1; snapshots outside [tail_start, tail_end) get weight 0)."""
    w = torch.zeros(len(ksteps), dtype=torch.float64)
    prev = None
    for i, k in enumerate(ksteps):
        if k < tail_start or k >= tail_end:
            continue
        if prev is None:
            w[i] = 1.0
        else:
            a = 1.0 - (1.0 - 1.0 / tau) ** (k - prev)
            w *= (1.0 - a)
            w[i] = a
        prev = k
    return w


def _mtt3_gcr_coordwise_median(stack: Tensor) -> Tensor:
    """Port of eval_readouts.coordwise_median (order-statistics convention: average
    of the two central values for even n)."""
    n = stack.size(0)
    s = stack.sort(dim=0).values
    m = n // 2
    if n % 2 == 1:
        return s[m].clone()
    return (s[m - 1] + s[m]) * 0.5


def _mtt3_gcr_sphere_log(u: Tensor, anchor: Tensor, eps: float = 1e-6):
    """Port of eval_readouts.sphere_log (guards: phi->0 Taylor, phi->pi cos clamp)."""
    cosphi = (u * anchor.unsqueeze(0)).sum(-1).clamp(-1.0 + eps, 1.0)
    phi = torch.arccos(cosphi)
    factor = torch.where(phi < 1e-6, torch.ones_like(phi), phi / torch.sin(phi).clamp_min(1e-12))
    z = factor.unsqueeze(-1) * (u - cosphi.unsqueeze(-1) * anchor.unsqueeze(0))
    return z, phi


def _mtt3_gcr_sphere_exp(z: Tensor, anchor: Tensor, eps: float = 1e-12) -> Tensor:
    """Port of eval_readouts.sphere_exp (|z|->0 returns the anchor; unit renorm)."""
    zn = z.norm(dim=-1, keepdim=True)
    dirn = z / zn.clamp_min(eps)
    out = torch.cos(zn) * anchor + torch.sin(zn) * dirn
    out = torch.where(zn < eps, anchor, out)
    return out / out.norm(dim=-1, keepdim=True).clamp_min(eps)


def _mtt3_gcr_slerp_rows(u_live: Tensor, u_hat: Tensor, lam) -> Tensor:
    """Port of eval_readouts.slerp_rows (near-parallel rows fall back to lerp+renorm)."""
    if not torch.is_tensor(lam):
        lam = torch.full((u_live.size(0), 1), float(lam),
                         device=u_live.device, dtype=u_live.dtype)
    lam = lam.view(-1, 1)
    cosw = (u_live * u_hat).sum(-1, keepdim=True).clamp(-1.0 + 1e-6, 1.0)
    w = torch.arccos(cosw)
    small = w < 1e-4
    sinw = torch.sin(w).clamp_min(1e-12)
    out = (torch.sin((1.0 - lam) * w) * u_live + torch.sin(lam * w) * u_hat) / sinw
    lerp = (1.0 - lam) * u_live + lam * u_hat
    out = torch.where(small, lerp, out)
    return out / out.norm(dim=-1, keepdim=True).clamp_min(1e-12)


def _mtt3_gcr_weighted_huber_rows(z: Tensor, a: Tensor, iters: int = 3, c: float = 1.345,
                                  eps: float = 1e-12) -> Tensor:
    """Port of eval_readouts.weighted_huber_rows: weighted Huber location per row
    (coordwise-median init, delta = c * per-row MAD of residual norms, `iters`
    IRLS updates). z (K,R,D), base weights a (K,) -> (R,D)."""
    mu = _mtt3_gcr_coordwise_median(z)
    rn = (z - mu.unsqueeze(0)).norm(dim=-1)                      # (K,R)
    delta = (c * rn.median(dim=0).values).clamp_min(eps)         # (R,)
    for _ in range(iters):
        rn = (z - mu.unsqueeze(0)).norm(dim=-1)
        h = (delta.unsqueeze(0) / rn.clamp_min(eps)).clamp(max=1.0)
        wh = a.view(-1, 1) * h                                   # (K,R)
        mu = (wh.unsqueeze(-1) * z).sum(0) / wh.sum(0).clamp_min(eps).unsqueeze(-1)
    return mu


def _mtt3_gcr_weighted_huber_scalar_rows(x: Tensor, a: Tensor, iters: int = 3,
                                         c: float = 1.345, eps: float = 1e-12) -> Tensor:
    """Port of eval_readouts.weighted_huber_scalar_rows: x (K,R), a (K,) -> (R,)."""
    aw = a.view(-1, 1)
    mu = _mtt3_gcr_coordwise_median(x)
    r = (x - mu.unsqueeze(0)).abs()
    delta = (c * r.median(dim=0).values).clamp_min(eps)
    for _ in range(iters):
        r = (x - mu.unsqueeze(0)).abs()
        h = (delta.unsqueeze(0) / r.clamp_min(eps)).clamp(max=1.0)
        wh = aw * h
        mu = (wh * x).sum(0) / wh.sum(0).clamp_min(eps)
    return mu


def _mtt3_gcr_rowsphere_decompose(stack: Tensor, a: Tensor, eps: float = 1e-12):
    """Port of eval_readouts.rowsphere_decompose: row polar decomposition of one
    hidden matrix's causal snapshot stack (K,R,D) with Tail-EMA weights a (K,);
    anchor = normalized weighted mean direction; uniform-weight fallback when no
    snapshot carries EMA weight. Returns (rho, u, anchor, z, a_eff)."""
    a_eff = a.to(device=stack.device, dtype=stack.dtype)
    if float(a_eff.sum()) <= 0:
        a_eff = torch.full_like(a_eff, 1.0 / max(len(a_eff), 1))
    rho = stack.norm(dim=-1)                                     # (K,R)
    u = stack / rho.clamp_min(eps).unsqueeze(-1)                 # (K,R,D)
    ubar = (a_eff.view(-1, 1, 1) * u).sum(0)
    anchor = ubar / ubar.norm(dim=-1, keepdim=True).clamp_min(eps)
    z, _ = _mtt3_gcr_sphere_log(u, anchor)
    return rho, u, anchor, z, a_eff


def _mtt3_gcr_h3_estimate(stack: Tensor, ksteps, live: Tensor, lam: float,
                          tau: float, tail_start: int, tail_end: int) -> Tensor:
    """One hidden matrix's H3 readout: the exact op sequence of the offline
    evaluator's compute_head_estimates H3 branch + its `assemble` at a fixed
    lambda. stack (K,R,D) fp32 causal snapshots (k <= the current val step by ring
    construction), live (R,D) fp32 current weights. Returns the final eval-ready
    (R,D) fp32 tensor (blend already applied)."""
    a = _mtt3_gcr_tailema_weights(ksteps, tau, tail_start, tail_end)
    rho, u, anchor, z, a_eff = _mtt3_gcr_rowsphere_decompose(stack, a)
    rho_live = live.norm(dim=-1)                                 # (R,)
    u_live = live / rho_live.clamp_min(1e-12).unsqueeze(-1)      # (R,D)
    z3 = _mtt3_gcr_weighted_huber_rows(z, a_eff)
    u3 = _mtt3_gcr_sphere_exp(z3, anchor)
    rho3 = _mtt3_gcr_weighted_huber_scalar_rows(rho, a_eff)
    lam_t = torch.full_like(rho_live, float(lam)).view(-1)
    u_ev = _mtt3_gcr_slerp_rows(u_live, u3, lam_t)
    rho_ev = (1.0 - lam_t) * rho_live + lam_t * rho3
    return rho_ev.unsqueeze(-1) * u_ev

def mtt3_pc2_t1(p, base_lr=0.0375, power_c=3.3169534699576625e-06,
                base_power=1.2, end_step=2900):
    """C2 round-4 (K7 Q1): flat-top end t1(p) of the area-conserving capped-power family.

    Derivation (hardcoded; defaults are the REAL-run base-schedule constants):
      base PowerCool: eta(t) = min(base_lr, power_c * (T - t)^1.2), T = end_step = 2900.
      Its exact flat/power crossing is t1* = T - (base_lr/power_c)^(1/1.2) = 513.60
      (the base schedule IS the p=1.2 member of the family with t1 = t1*).
      Baseline LR area (analytic): A0 = base_lr*t1* + base_lr*(T - t1*)/(1.2 + 1) = 59.9373.
      Area conservation for tail power p: base_lr*t1 + base_lr*(T - t1)/(p+1) = A0
        =>  t1(p) = ((p+1)*A0/base_lr - T) / p.
      Anchors (K7 table): t1(1.2)=513.6, t1(1.5)=730.5, t1(1.72)=841.5, t1(2.0)=947.5.
    """
    t1_base = end_step - (base_lr / power_c) ** (1.0 / base_power)
    a0 = base_lr * t1_base + base_lr * (end_step - t1_base) / (base_power + 1.0)
    return ((p + 1.0) * a0 / base_lr - end_step) / p

def mtt3_lr_override_eta(step, mode,
                         eta_start=0.007876, eta_floor=0.001573,
                         fork_step=2250, end_step=2900,
                         base_lr=0.0375, power_c=3.3169534699576625e-06, power=1.2,
                         a2=-0.42085, pc2_power=1.72):
    """C2 fork-family Muon matrix-group LR laws -- a pure function of step (no state).

    Defaults are the REAL-run constants (deliberately literal, so MTT3_TEST_MODE's
    constant mutations can never leak into the branch math):
      eta_start = PowerCool eta at the fork: min(0.0375, 3.3169...e-6 * 650^1.2) = 0.0078747
                  (spec anchor 0.007876; rel diff 1.6e-4, inside the 1e-3 tolerance band)
      eta_floor = equal-LR-budget floor over [2250, 2900] (integral of geo == integral of
                  PowerCool to 1.3e-4 relative)
      a2        = alpha_hat / 2 with alpha_hat = -0.8417 (calorimeter zeta(eta) power-law
                  fit, round K2): geodesic of constant thermodynamic speed under
                  zeta ~ eta^alpha has eta^(alpha/2) linear in t.

    Modes (caller applies only for step >= fork_step; s is clamped for safety):
      geo   : eta(t) = (A + s*(B-A))^(1/a2), s=(t-fork)/(end-fork), A=eta_start^a2, B=eta_floor^a2
      floor : eta(t) = max(powercool(t), eta_floor)
      p15   : eta(t) = eta_start * ((end-t)/(end-fork))^1.5   (arbitrary-slower-tail shape control)
      pc2   : K7 Q1 area-conserving capped-power family (from-scratch, fork_step=0):
              eta(t) = base_lr for t <= t1(p); base_lr * ((end-t)/(end-t1(p)))^p after, with
              p = pc2_power and t1(p) = mtt3_pc2_t1(p). Uses ONLY the base-schedule constants
              (base_lr/power_c/power/end_step); eta_start/eta_floor/fork_step are ignored BY
              DESIGN. p = 1.2 reproduces the base PowerCool schedule exactly.
    """
    span = float(end_step - fork_step)
    if mode == "geo":
        s = min(max((step - fork_step) / span, 0.0), 1.0)
        a = eta_start ** a2
        b = eta_floor ** a2
        return (a + s * (b - a)) ** (1.0 / a2)
    if mode == "floor":
        powercool = min(base_lr, power_c * max(0.0, float(end_step - step)) ** power)
        return max(powercool, eta_floor)
    if mode == "p15":
        u = min(max((end_step - step) / span, 0.0), 1.0)
        return eta_start * u ** 1.5
    if mode == "pc2":
        t1 = mtt3_pc2_t1(pc2_power, base_lr=base_lr, power_c=power_c,
                         base_power=power, end_step=end_step)
        if step <= t1:
            return base_lr
        u = max(0.0, float(end_step - step)) / float(end_step - t1)
        return base_lr * u ** pc2_power
    raise ValueError(f"unknown MTT3_LR_MODE: {mode!r}")

def mtt3_powercool_eta(step, base_lr=0.0375, power_c=3.3169534699576625e-06,
                       power=1.2, end_step=2900):

def mtt3_fp_metric_from_sigma(sigma, eps=1e-4):
    """ARM B (MTT3_FUNC_POLAR) pure math: (R, R_inv) from an input second-moment matrix.

    R = V diag(clamp(lam / mean(lam), min=eps))^{1/2} V^T with Sigma = V diag(lam) V^T:
      - trace-normalized (divide by the mean eigenvalue = tr(Sigma)/d) so R ~= I for
        isotropic inputs and the a>0 path degenerates towards the base direction;
      - eigenvalue clamp at eps RELATIVE to the mean keeps R symmetric positive
        definite (and R_inv finite) even for rank-deficient or all-zero Sigma
        (dead activations / probe tokens < d_in);
      - all-zero Sigma -> R = sqrt(eps)*I, whose direction effect cancels in the
        norm-preserving blend downstream (msign is scale-invariant).
    """
    import torch
    sig = sigma.float()
    sig = (sig + sig.T) * 0.5            # guard fp drift; EMA of symmetric stays symmetric
    try:
        evals, evecs = torch.linalg.eigh(sig)
    except RuntimeError:
        evals, evecs = torch.linalg.eigh(sig.double())
        evals, evecs = evals.float(), evecs.float()
    mean = evals.mean().clamp_min(1e-30)  # = tr(Sigma)/d
    lam = (evals / mean).clamp_min(eps)
    r = (evecs * lam.sqrt()) @ evecs.T
    r_inv = (evecs * lam.rsqrt()) @ evecs.T
    return r, r_inv
def mtt3_bimaxwell_params(mode):
    """K12 Q4.2 Bi-Maxwell dual-timescale stress-memory arms -- pure parameter
    resolver (no state; exec-extracted by the test suite).

    Kernel mean age of an EMA with lerp rate (1-beta): nbar(beta) = beta/(1-beta);
    the baseline single momentum mu=0.95 has nbar = 19. Mean-age parity constraint
    (K12 4.2: "do not casually change the average memory age"):

        w*nbar(beta_f) + (1-w)*nbar(beta_s) = 19.

      BM1: beta_f=0.85, beta_s=0.98. Exact solve: nbar_f=17/3, nbar_s=49 ->
           w* = (49-19)/(49-17/3) = 9/13 = 0.692308; PREREGISTERED literal
           w = 0.69 (K12 rounds w*), realized age 19.100 -- the 0.5% rounding
           slack is documented in the prereg (tests/test p verify both numbers).
      BM2: beta_f=0.90, beta_s=0.985. w solved exactly HERE from the constraint:
           nbar_f=9, nbar_s=197/3 -> w = (nbar_s-19)/(nbar_s-nbar_f) = 14/17
           = 0.823529..., realized age == 19 exactly.
      "bf:bs:w": custom triple -- diagnostic/test form only (e.g. the degenerate
           equal-beta parity arm "0.95:0.95:<any w>"); NOT a preregistered branch.

    Returns (beta_f, beta_s, w, realized_mean_age).
    """
    if mode == "BM1":
        bf, bs, w = 0.85, 0.98, 0.69
    elif mode == "BM2":
        bf, bs = 0.90, 0.985
        nf, ns = bf / (1.0 - bf), bs / (1.0 - bs)
        w = (ns - 19.0) / (ns - nf)
    else:
        parts = mode.split(":")
        assert len(parts) == 3, f"unknown MTT3_BIMAXWELL: {mode!r} (BM1 | BM2 | bf:bs:w)"
        bf, bs, w = (float(x) for x in parts)
    assert 0.0 < bf < 1.0 and 0.0 < bs < 1.0 and 0.0 <= w <= 1.0, \
        f"MTT3_BIMAXWELL parameters out of range: {mode!r}"
    nf, ns = bf / (1.0 - bf), bs / (1.0 - bs)
    return bf, bs, w, w * nf + (1.0 - w) * ns
def mtt3_bm_w_at(step, w0, w_end, t0, t1):
    """BM w-anneal rider (bm_thermo_round2 prereg, "anneal" arm) -- pure math,
    exec-extracted by the test suite. Effective convex mixing weight at muon step
    `step`: constant w0 through t0, LINEAR to w_end over (t0, t1), constant w_end
    from t1 on. The plateau branches return the literals, so w(t0) == w0 and
    w(t1) == w_end EXACTLY (no float lerp at the endpoints). Betas never move."""
    if step <= t0:
        return w0
    if step >= t1:
        return w_end
    return w0 + (w_end - w0) * ((step - t0) / (t1 - t0))
def mtt3_fdt_alpha_use(kstar, gamma, alpha_center=0.05, clip_lo=0.02, clip_hi=0.12):
    """K12.1 FDT step-4 pure math (designs/fdt_bucketed_memory_implementation.md;
    exec-extracted by the test suite): per-bucket effective memory rate from the
    online Kalman gain K*.

        alpha_tgt(b) = clip(K*_b, clip_lo, clip_hi)
        log alpha_use(b) = (1-gamma)*log(alpha_center) + gamma*log(alpha_tgt(b))

    Estimator warm-up (K*_b is None or NaN) -> that bucket stays at alpha_center
    EXACTLY (no exp/log float round-trip), so the not-yet-warm splice pins to the
    center buffer's coefficient rows bit-cleanly. Constants are frozen literals
    (spec section 1: grid/clip/center are code constants, not envs)."""
    import math
    out = []
    for k in kstar:
        if k is None or k != k:                  # None / NaN -> warm-up: exact center rate
            out.append(alpha_center)
            continue
        a_tgt = min(max(float(k), clip_lo), clip_hi)
        out.append(math.exp((1.0 - gamma) * math.log(alpha_center) + gamma * math.log(a_tgt)))
    return out


def mtt3_fdt_splice(u, m_list, alpha_use, center_idx=2,
                    alpha_grid=(0.02, 0.035, 0.05, 0.075, 0.12),
                    spec_buckets=(0.0, 0.05, 0.15, 0.35, 0.65, 1.0)):
    """K12.1 FDT steps 5-6 pure math (exec-extracted by the test suite): spectral-bucket
    splice of the multi-timescale memory buffers in the LEFT-coefficient space of the
    basis u, residual routed through the center buffer M_c = m_list[center_idx].

        C_j = u^T M_j              (formed lazily -- only referenced grid points)
        per bucket b (index buckets over the r singular directions, same boundary
        convention as _mtt3_bucket_means): alpha_use[b] -> adjacent grid interval
        [alpha_lo, alpha_hi], log-linear weight t = (log a - log a_lo)/(log a_hi - log a_lo),
        rows i in b: C_eff[i,:] = (1-t)*C_lo[i,:] + t*C_hi[i,:]
        (t <= 0 / >= 1 -> the EXACT grid-point coefficient rows, no float lerp)
        M_eff = u @ C_eff + (M_c - u @ (u^T M_c))

    The second term reclaims the part of M_c outside span(u) (nonzero only for a STALE
    basis on tall matrices, MTT3_FDT_EVERY > 1; for a fresh SVD basis of M_c it is 0 to
    fp rounding -- the square case walks the same formula, spec section 2 step 5).
    Sum-projection identity: all buckets at a grid point j -> C_eff = C_j exactly, and
    at the center point M_eff == M_c up to fp reorganization. All math fp32."""
    import math
    import torch
    uf = u.float()
    mc = m_list[center_idx].float()
    r = uf.shape[1]
    c_cache = {}

    def _c(j):
        c = c_cache.get(j)
        if c is None:
            c = c_cache[j] = uf.T @ m_list[j].float()
        return c

    c_eff = torch.empty_like(_c(center_idx))
    idx = torch.arange(r, dtype=torch.float32) / max(r, 1)
    nb = len(spec_buckets) - 1
    for b in range(nb):
        lo, hi = spec_buckets[b], spec_buckets[b + 1]
        mask = (idx >= lo) & ((idx < hi) if b < nb - 1 else (idx <= hi))
        if not bool(mask.any()):
            continue
        a = alpha_use[b]
        if a is None or a != a:                  # defensive NaN guard (caller already centers)
            a = alpha_grid[center_idx]
        a = min(max(float(a), alpha_grid[0]), alpha_grid[-1])
        j = 0
        while j < len(alpha_grid) - 2 and a > alpha_grid[j + 1]:
            j += 1
        if a <= alpha_grid[j]:
            t = 0.0
        else:
            t = (math.log(a) - math.log(alpha_grid[j])) / \
                (math.log(alpha_grid[j + 1]) - math.log(alpha_grid[j]))
        if t <= 0.0:
            c_eff[mask] = _c(j)[mask]
        elif t >= 1.0:
            c_eff[mask] = _c(j + 1)[mask]
        else:
            c_eff[mask] = torch.lerp(_c(j)[mask], _c(j + 1)[mask], t)
    return uf @ c_eff + (mc - uf @ (uf.T @ mc))


def mtt3_mu_at_step(step, train_steps, mu_max=0.95, cd_end=0.85,
                    mu_min=0.85, warmup_steps=300, cooldown_steps=200):
    """Muon momentum schedule -- pure math, exec-extracted by the test suite
    (round-3 arm A, designs/round3_mu_and_start_preregistration.md).

    The base v49/v15 arithmetic VERBATIM, parametrized: warmup mu_min -> mu_max over
    warmup_steps, plateau mu_max, cooldown mu_max -> cd_end over the last
    cooldown_steps. Defaults are the base constants (the same floats as _MU_MIN /
    _MU_MAX / _MU_WARMUP_STEPS / _MU_COOLDOWN_STEPS), so the trainer's unset path
    (mu_max=_MU_MAX, cd_end=_MU_MIN) evaluates the EXACT original expressions on the
    exact original values -- byte-identical schedule (HARD invariant; decoys in
    tests/test a). MTT3_MU_CD_END touches ONLY cd_end (the cooldown terminal):
    the warmup start stays mu_min regardless."""
    cd_start = train_steps - cooldown_steps
    if step < warmup_steps:
        frac = step / max(warmup_steps, 1)
        return mu_min + frac * (mu_max - mu_min)
    elif step > cd_start:
        frac = (step - cd_start) / max(cooldown_steps, 1)
        return mu_max - frac * (mu_max - cd_end)
    else:
        return mu_max

def _mtt3_cos(a: Tensor, b: Tensor, eps: float = 1e-30):
    af, bf = a.float(), b.float()
    na, nb = af.norm(), bf.norm()
    if float(na) < eps or float(nb) < eps:
        return float("nan")
    return float((af * bf).sum() / (na * nb))


def _mtt3_bucket_means(vals: Tensor) -> list:
    """Aggregate a length-r per-direction vector into the 5 index buckets (mean of vals)."""
    r = vals.numel()
    idx = torch.arange(r, device=vals.device, dtype=torch.float32) / max(r, 1)
    out = []
    for b in range(len(MTT3_SPEC_BUCKETS) - 1):
        lo, hi = MTT3_SPEC_BUCKETS[b], MTT3_SPEC_BUCKETS[b + 1]
        mask = (idx >= lo) & ((idx < hi) if b < len(MTT3_SPEC_BUCKETS) - 2 else (idx <= hi))
        out.append(float(vals[mask].mean()) if bool(mask.any()) else float("nan"))
    return out


def _mtt3_bucket_counts(r: int) -> list:
    """Directions per spectral bucket -- the I-probe's n_eff. s_b is the MEAN of n_eff
    per-direction projections, so Var(measurement noise of s_b) = T_b / n_eff under the
    documented cross-direction independence approximation (mtt3_meta.json iprobe_note).
    Empty buckets clamp to 1 (their bucket means are NaN anyway)."""
    idx = torch.arange(r, dtype=torch.float32) / max(r, 1)
    out = []
    for b in range(len(MTT3_SPEC_BUCKETS) - 1):
        lo, hi = MTT3_SPEC_BUCKETS[b], MTT3_SPEC_BUCKETS[b + 1]
        mask = (idx >= lo) & ((idx < hi) if b < len(MTT3_SPEC_BUCKETS) - 2 else (idx <= hi))
        out.append(max(int(mask.sum()), 1))
    return out


def _mtt3_spec_occupation(s: Tensor):
    """5-bucket spectral-mass occupation + full-spectrum entropy of svdvals s."""
    tot = float(s.sum())
    if tot <= 0:
        return [float("nan")] * (len(MTT3_SPEC_BUCKETS) - 1), float("nan")
    p_full = (s / s.sum()).clamp_min(1e-30)
    entropy = float(-(p_full * p_full.log()).sum())
    r = s.numel()
    idx = torch.arange(r, device=s.device, dtype=torch.float32) / max(r, 1)
    occ = []
    for b in range(len(MTT3_SPEC_BUCKETS) - 1):
        lo, hi = MTT3_SPEC_BUCKETS[b], MTT3_SPEC_BUCKETS[b + 1]
        mask = (idx >= lo) & ((idx < hi) if b < len(MTT3_SPEC_BUCKETS) - 2 else (idx <= hi))
        occ.append(float(p_full[mask].sum()) if bool(mask.any()) else 0.0)
    return occ, entropy


def _mtt3_soap_ro(x: Tensor, state: dict, eps: float = 1e-8) -> Tensor:
    """READ-ONLY replica of soap_precondition_momentum: identical math, no state mutation."""
    x_f = x.float()
    if state["q_row"] is None:
        return x_f
    q_row, q_col = state["q_row"], state["q_col"]
    projected = q_row.T @ x_f @ q_col
    denom = state["exp_avg_sq"].clamp_min(eps * eps).pow(SOAP_DENOM_POWER)
    precond = q_row @ (projected / denom) @ q_col.T
    return precond * (gram_frobenius_norm_estimate(x_f, eps=eps) / gram_frobenius_norm_estimate(precond, eps=eps))


def _mtt3_ns_k(G: Tensor, iters: int) -> Tensor:
    """msign via Newton-Schulz with an EXPLICIT iteration count (I-probe shadow cosines
    use NS-5; K12 1.5: NS-5 suffices for shadow direction comparison). Same
    normalization / transpose / dtype chain as zeropower_via_newtonschulz5 but
    independent of MTT3_NS_ITERS. Read-only."""
    assert G.ndim >= 2
    a, b, c = 2, -1.5, 0.5
    X = G.bfloat16()
    if G.size(-2) > G.size(-1):
        X = X.mT
    X = X / gram_frobenius_norm_estimate(X, keepdim=True, eps=1e-7).to(X.dtype)
    for _ in range(iters):
        A = X @ X.mT
        B = b * A + c * A @ A
        X = a * X + B @ X
    if G.size(-2) > G.size(-1):
        X = X.mT
    return X

def _load_data_shard(file: Path):
    header = torch.from_file(str(file), False, 256, dtype=torch.int32) # header is 256 int32
    assert header[0] == 20240520, "magic number mismatch in the data .bin file"
    assert header[1] == 1, "unsupported version"
    num_tokens = int(header[2]) # number of tokens (claimed)
    with file.open("rb", buffering=0) as f:
        tokens = torch.empty(num_tokens, dtype=torch.uint16, pin_memory=True)
        f.seek(256 * 4)
        nbytes = f.readinto(tokens.numpy()) # avoid bytes->array copy
        assert nbytes == 2 * num_tokens, "number of tokens read does not match header"
    return tokens

def distributed_data_generator(filename_pattern: str, batch_size: int, seq_len=1024):
    world_size = dist.get_world_size()
    rank = dist.get_rank()
    files = sorted(Path.cwd().glob(filename_pattern))
    assert batch_size % world_size == 0
    local_batch_size = batch_size // world_size
    file_iter = iter(files)
    tokens, pos = _load_data_shard(next(file_iter)), 0
    while True:
        if pos + batch_size + 1 >= len(tokens):
            tokens, pos = _load_data_shard(next(file_iter)), 0
        buf = tokens[pos + rank * local_batch_size:][:local_batch_size + 1]
        inputs = buf[:-1].to(device="cuda", dtype=torch.int32, non_blocking=True)
        targets = buf[1:].to(device="cuda", dtype=torch.int64, non_blocking=True)
        pos += batch_size
        yield inputs.view(-1, seq_len), targets.view(-1, seq_len)

def norm(x: Tensor):
    return F.rms_norm(x, (x.size(-1),))

class RMSNorm(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.gains = nn.Parameter(torch.ones(dim))

    def forward(self, x):
        return (norm(x.float()) * self.gains).type_as(x)

class Linear(nn.Linear):
    def __init__(self, in_features, out_features):
        super().__init__(in_features, out_features, bias=True)

    def forward(self, x):
        return F.linear(x, self.weight.type_as(x), self.bias.type_as(x))

class Rotary(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        # half-truncate RoPE (w/ base freq tuning)
        angular_freq = (1 / 1024) ** torch.linspace(0, 1, steps=dim//4, dtype=torch.float32)
        self.register_buffer("angular_freq", torch.cat([angular_freq, angular_freq.new_zeros(dim//4)]))

    def forward(self, x_BTHD: Tensor):
        pos = torch.arange(x_BTHD.size(1), dtype=torch.float32, device=x_BTHD.device)
        theta = torch.outer(pos, self.angular_freq)[None, :, None, :]
        cos, sin = theta.cos(), theta.sin()
        x1, x2 = x_BTHD.to(dtype=torch.float32).chunk(2, dim=-1)
        y1 = x1 * cos + x2 * sin
        y2 = x1 * (-sin) + x2 * cos
        return torch.cat((y1, y2), 3).type_as(x_BTHD)

class CausalSelfAttention(nn.Module):
    def __init__(self, dim: int, head_dim=128):
        super().__init__()
        self.num_heads = dim // head_dim
        self.head_dim = head_dim
        hdim = self.num_heads * self.head_dim
        self.q = Linear(dim, hdim)
        self.k = Linear(dim, hdim)
        self.v = Linear(dim, hdim)
        self.proj = Linear(hdim, dim)
        self.rotary = Rotary(head_dim)

    def forward(self, x: Tensor):
        B, T = x.size(0), x.size(1)
        q = self.q(x).view(B, T, self.num_heads, self.head_dim)
        k = self.k(x).view(B, T, self.num_heads, self.head_dim)
        v = self.v(x).view(B, T, self.num_heads, self.head_dim)
        q, k = norm(q), norm(k)
        q, k = self.rotary(q), self.rotary(k)
        y = F.scaled_dot_product_attention(q.transpose(1, 2), k.transpose(1, 2),
                                           v.transpose(1, 2), scale=0.12, is_causal=True).transpose(1, 2)
        y = y.contiguous().view(B, T, self.num_heads * self.head_dim)
        y = self.proj(y)
        return y

class MLP(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        hdim = 4 * dim
        self.fc = Linear(dim, hdim)
        self.proj = Linear(hdim, dim)

    def forward(self, x: Tensor):
        x = self.fc(x)
        x = x.relu().square()
        x = self.proj(x)
        return x

class Block(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        self.attn = CausalSelfAttention(dim, head_dim=MTT3_HEAD_DIM)  # ===== MTT3: ==128 (base value) unless MTT3_TEST_MODE
        self.mlp = MLP(dim)
        self.norm1 = RMSNorm(dim)
        self.norm2 = RMSNorm(dim)

    def forward(self, x: Tensor):
        x = x + self.attn(self.norm1(x))
        x = x + self.mlp(self.norm2(x))
        return x

class GPT(nn.Module):
    def __init__(self, vocab_size: int, num_layers: int, model_dim: int):
        super().__init__()
        self.embed = nn.Embedding(vocab_size, model_dim).bfloat16()
        self.blocks = nn.ModuleList([Block(model_dim) for _ in range(num_layers)])
        self.proj = Linear(model_dim, vocab_size)
        self.norm1 = RMSNorm(model_dim)
        self.norm2 = RMSNorm(model_dim)

    def forward(self, inputs: Tensor, targets: Tensor):
        x = self.norm1(self.embed(inputs))
        for block in self.blocks:
            x = block(x)
        logits = self.proj(self.norm2(x)).float()
        logits = 15 * logits * (logits.square() + 15**2).rsqrt()
        return F.cross_entropy(logits.view(targets.numel(), -1), targets.view(-1), reduction="sum")

def gram_frobenius_norm_estimate(G: Tensor, keepdim: bool = False, eps: float = 1e-10) -> Tensor:
    X = G.float()
    gram = X.mT @ X if X.size(-2) > X.size(-1) else X @ X.mT
    return gram.norm(dim=(-2, -1), keepdim=keepdim).sqrt().clamp_min(eps)

def _ns_inner(X: Tensor) -> Tensor:
    a, b, c = 2, -1.5, 0.5
    for _ in range(MTT3_NS_ITERS):  # ===== MTT3: base literal 12; env-gated, default unchanged =====
        A = X @ X.mT
        B = b * A + c * A @ A
        X = a * X + B @ X
    return X

def zeropower_via_newtonschulz5(G: Tensor) -> Tensor:
    # Standard Muon Newton-Schulz orthogonalization. (Aurora wide-matrix row-rescale
    # removed -- ablated neutral; this is exactly the AURORA_K=0 path.)
    assert G.ndim >= 2
    X = G.bfloat16()
    if G.size(-2) > G.size(-1):
        X = X.mT
    X = X / gram_frobenius_norm_estimate(X, keepdim=True, eps=1e-7).to(X.dtype)
    X = _ns_inner(X)
    if G.size(-2) > G.size(-1):
        X = X.mT
    return X

def tensor_cosine(a: Tensor, b: Tensor, eps: float = 1e-8) -> Tensor:
    a_f, b_f = a.float(), b.float()
    return (a_f * b_f).sum() / (a_f.norm() * b_f.norm()).clamp_min(eps)

def trust_gate(raw: Tensor, soap: Tensor, grad: Tensor, eps: float = 1e-8) -> Tensor:
    # SOAP is trusted when it still points with raw momentum and is at least as
    # gradient-aligned as raw momentum. This catches stale whitening bases.
    raw_grad = tensor_cosine(raw, grad, eps)
    soap_grad = tensor_cosine(soap, grad, eps)
    soap_raw = tensor_cosine(soap, raw, eps)

    agree_gate = ((soap_raw - ATTN_TRUST_MIN_AGREE) / (1 - ATTN_TRUST_MIN_AGREE)).clamp(0, 1)
    denom = (raw_grad - ATTN_TRUST_MIN_GRAD_ALIGN).clamp_min(eps)
    grad_gate = ((soap_grad - ATTN_TRUST_MIN_GRAD_ALIGN) / denom).clamp(0, 1)
    gate = (agree_gate * grad_gate).clamp(0, 1)
    if ATTN_TRUST_POWER != 1.0:
        gate = gate.pow(ATTN_TRUST_POWER)
    return gate

def early_trust_floor_for_step(step: int) -> float:
    if ATTN_TRUST_FLOOR_FADE_END_STEP <= ATTN_TRUST_FLOOR_END_STEP:
        return 0.0 if step >= ATTN_TRUST_FLOOR_FADE_END_STEP else ATTN_EARLY_TRUST_FLOOR
    if step < ATTN_TRUST_FLOOR_END_STEP:
        return ATTN_EARLY_TRUST_FLOOR
    if step >= ATTN_TRUST_FLOOR_FADE_END_STEP:
        return 0.0
    return ATTN_EARLY_TRUST_FLOOR * (
        ATTN_TRUST_FLOOR_FADE_END_STEP - step
    ) / (ATTN_TRUST_FLOOR_FADE_END_STEP - ATTN_TRUST_FLOOR_END_STEP)

def bounded_trust_gate(gate: Tensor, step: int) -> Tensor:
    floor = early_trust_floor_for_step(step)
    cap = ATTN_EARLY_TRUST_CAP if step < ATTN_TRUST_FLOOR_FADE_END_STEP else 1.0
    return gate.clamp(min=floor, max=cap)

def norm_preserving_blend(raw: Tensor, soap: Tensor, gate: Tensor, eps: float = 1e-8) -> Tensor:
    blended = raw + (soap - raw) * gate.to(raw.dtype)
    raw_norm = gram_frobenius_norm_estimate(raw, eps=eps)
    blended_norm = gram_frobenius_norm_estimate(blended, eps=eps)
    return (blended * (raw_norm / blended_norm).to(blended.dtype)).to(raw.dtype)

def scale_radial_update(update: Tensor, param: Tensor, eps: float = 1e-12) -> Tensor:
    update_f = update.float()
    param_f = param.float()
    denom = (param_f * param_f).sum().clamp_min(eps)
    coeff = (update_f * param_f).sum() / denom
    radial = coeff * param_f
    tangential = update_f - radial
    # p.add_(update, alpha=-lr), so actual movement is -update.
    # Outward movement means (-update) is aligned with p, i.e. coeff < 0.
    radial_scale = torch.where(
        coeff < 0,
        update_f.new_tensor(RADIAL_OUTWARD_SCALE),
        update_f.new_tensor(RADIAL_INWARD_SCALE),
    )
    return (tangential + radial_scale * radial).to(update.dtype)

def target_radius_after_update(param: Tensor, update: Tensor, lr: float, eps: float = 1e-8) -> Tensor:
    param_f = param.float()
    update_f = update.float()
    before_norm = param_f.norm().clamp_min(eps)
    # Use only the radial component's first-order radius change as the intended
    # radius change; the post-step rescale below removes finite tangent drift.
    radial_delta = -lr * (update_f * param_f).sum() / before_norm
    return (before_norm + radial_delta).clamp_min(eps)

def rescale_to_radius(param: Tensor, target_norm: Tensor, eps: float = 1e-8):
    after_norm = param.float().norm().clamp_min(eps)
    param.mul_((target_norm / after_norm).to(param.dtype))

def soap_eigenbasis(mat: Tensor) -> Tensor:
    try:
        _, q = torch.linalg.eigh(mat + 1e-30 * torch.eye(mat.size(0), device=mat.device))
    except RuntimeError:
        _, q = torch.linalg.eigh(mat.double() + 1e-30 * torch.eye(mat.size(0), device=mat.device))
        q = q.float()
    return torch.flip(q, [1])

def soap_basis_qr(row_gg, col_gg, q_row, q_col, exp_avg_sq):
    row_eig = torch.diag(q_row.T @ row_gg @ q_row)
    row_sort = torch.argsort(row_eig, descending=True)
    q_row = q_row[:, row_sort]
    exp_avg_sq = exp_avg_sq.index_select(0, row_sort)
    q_row, _ = torch.linalg.qr(row_gg @ q_row)

    col_eig = torch.diag(q_col.T @ col_gg @ q_col)
    col_sort = torch.argsort(col_eig, descending=True)
    q_col = q_col[:, col_sort]
    exp_avg_sq = exp_avg_sq.index_select(1, col_sort)
    q_col, _ = torch.linalg.qr(col_gg @ q_col)
    return q_row, q_col, exp_avg_sq

def soap_precondition_momentum(update, state, beta2=SOAP_BETA2, eps=1e-8):
    update_f = update.float()
    if state["q_row"] is None:
        return update
    q_row, q_col = state["q_row"], state["q_col"]
    projected = q_row.T @ update_f @ q_col
    state["exp_avg_sq"].mul_(beta2).add_(projected.square(), alpha=1 - beta2)
    denom = state["exp_avg_sq"].clamp_min(eps * eps).pow(SOAP_DENOM_POWER)
    precond = q_row @ (projected / denom) @ q_col.T
    precond.mul_(gram_frobenius_norm_estimate(update_f, eps=eps) / gram_frobenius_norm_estimate(precond, eps=eps))
    return precond.to(update.dtype)

def soap_update_preconditioner(grad, state, shampoo_beta=SOAP_BETA2, precondition_frequency=SOAP_PRECONDITION_FREQUENCY):
    grad_f = grad.float()
    state["row_gg"].lerp_(grad_f @ grad_f.T, 1 - shampoo_beta)
    state["col_gg"].lerp_(grad_f.T @ grad_f, 1 - shampoo_beta)
    if state["q_row"] is None:
        state["q_row"] = soap_eigenbasis(state["row_gg"])
        state["q_col"] = soap_eigenbasis(state["col_gg"])
    elif state["soap_step"] > 0 and state["soap_step"] % precondition_frequency == 0:
        state["q_row"], state["q_col"], state["exp_avg_sq"] = soap_basis_qr(
            state["row_gg"], state["col_gg"], state["q_row"], state["q_col"], state["exp_avg_sq"]
        )
    state["soap_step"] += 1

def muon_update(update):
    # Newton-Schulz orthogonalization + aspect-ratio scale.
    update = zeropower_via_newtonschulz5(update)
    update *= max(1, update.size(-2) / update.size(-1))**0.5
    return update


# ============== MTT3 (part 2a: avg2 polar TRAINING-PATH intervention) ==============
# MTT3_POLAR_MODE=avg2 replaces the Muon update DIRECTION by the norm-preserving average
# of the polar factors of the two grad-accumulation half-batch momenta. Unlike every
# other MTT3_* hook this MUTATES optimizer state (persistent halves "mtt3_ma"/"mtt3_mb"
# live inside Muon state so the generic ckpt state clone saves/loads them for free).
# With the env unset all of it is dead code -> base byte-identical (tests a/i).

def _mtt3_avg2_apply(opt, p, grad, state, base_update, mu, ctx):
    """avg2 update direction (runs EVERY step, per 2-D Muon param, on the training path).

      A  = p.grad after the first n_mb//2 microbatches (captured by _mtt3_halfgrad_hook
           into _M3.avg2_half, all-reduced like p.grad); Ga = 2A, Gb = 2(G-A)  (exact O1
           split: (Ga+Gb)/2 == G).
      Ma <- lerp(Ma, Ga, 1-mu), Mb <- lerp(Mb, Gb, 1-mu)     (same mu-mixing as momentum;
           (Ma+Mb)/2 must track M to fp rounding -- asserted, max deviation logged)
      Ua = lerp(Ga, Ma, mu), Ub = lerp(Gb, Mb, mu)           (same Nesterov mix as base)
      Qa = NS(soap_ro(Ua)),  Qb = NS(soap_ro(Ub))            (read-only SOAP with the
           CURRENT shared state, NO trust gate -- same convention/caveat as the O1 probes)
      update = (Qa+Qb)/2 rescaled to ||base_update||_F EXACTLY.

    Norm-parity choice (documented per dispatch): the base Q is computed anyway (which
    also keeps the SOAP exp_avg_sq trajectory exactly on the base path), and Q-bar is
    renormed to the base post-NS post-aspect-scale update's Frobenius norm -- exact
    parity by construction, simpler than re-deriving a sqrt(r)-consistent scale. The
    aspect-ratio scale/shape handling is inherited through the base norm; RowFloor /
    radial brake / pin / CWD all operate on the averaged update downstream, unchanged.

    Ckpt: Ma/Mb ride along in "muon_state" via the generic per-key clone (lazy .get on
    load). Resuming a ckpt WITHOUT halves lazy-inits Ma=Mb=M at the fork step (halves
    decorrelate over ~1/(1-mu) ~= 20 steps); on the init step the halves take no
    half-lerp (they equal the post-lerp M), so (Ma+Mb)/2 == M holds exactly from step 0.
    Cost: 2 extra NS + ~2x Muon-param fp32 state -- wall-clock free under track-3 scoring.
    """
    a = _M3.avg2_half[p]                     # fp32, SUM-reduced like grad
    gf = grad.float()
    ga = a * 2.0
    gb = (gf - a) * 2.0
    if "mtt3_ma" not in state:
        state["mtt3_ma"] = state["momentum"].clone()
        state["mtt3_mb"] = state["momentum"].clone()
        if not _M3.avg2_announced:
            _M3.avg2_announced = True
            print0(f"mtt3 polar avg2: lazy-init Ma=Mb=M at muon step {opt.step_count}", console=True)
    else:
        state["mtt3_ma"].lerp_(ga.to(state["mtt3_ma"].dtype), 1.0 - mu)
        state["mtt3_mb"].lerp_(gb.to(state["mtt3_mb"].dtype), 1.0 - mu)
    ma, mb = state["mtt3_ma"], state["mtt3_mb"]
    # reconstruction guard: (Ma+Mb)/2 must stay bit-close to the real momentum
    dev = float(((ma.float() + mb.float()) * 0.5 - state["momentum"].float()).abs().max())
    scale = float(state["momentum"].float().abs().max())
    rel = dev / max(scale, 1e-30)
    assert rel < 1e-3, (f"mtt3 avg2: (Ma+Mb)/2 diverged from momentum: rel={rel:.3e} "
                        f"({opt.mtt3_names.get(p, '?')} step {opt.step_count})")
    _M3.avg2_recon_max = max(_M3.avg2_recon_max, rel)
    ua = ga.lerp(ma.float(), mu)
    ub = gb.lerp(mb.float(), mu)
    if p in opt.soap_params:
        ua = _mtt3_soap_ro(ua, state)
        ub = _mtt3_soap_ro(ub, state)
    qa = zeropower_via_newtonschulz5(ua)
    qb = zeropower_via_newtonschulz5(ub)
    # fp64 renorm: parity |  ||update||_F - ||base_update||_F | / ||base|| ~ 1e-15 (test bound 1e-6);
    # elementwise + one reduction per param, negligible next to the two extra NS matmul chains.
    qbar = (qa.double() + qb.double()) * 0.5
    base_fro = base_update.double().norm()
    out = qbar * (base_fro / qbar.norm().clamp_min(1e-30))
    if ctx is not None:                       # diag-only telemetry (jsonl via the muon pivot)
        ctx["polar_qbase_fro"] = float(base_fro)
        ctx["polar_q_fro"] = float(out.norm())            # parity: == polar_qbase_fro to ~1e-15 rel
        ctx["polar_qbar_fro_prescale"] = float(qbar.norm())
        ctx["polar_cos_qa_qb"] = _mtt3_cos(qa, qb)
        ctx["polar_recon_maxdev_rel"] = rel
    return out.to(base_update.dtype)

# ========= MTT3 (part 2b: structural-core TRAINING-PATH interventions) =========
# Round-5 arms (theory/muon_thermo_deep.md v1 P4). Both env-gated; with the knobs at
# their 0 defaults neither function is ever called -> base byte-identical (tests a/j).

def _mtt3_mom_extrap_apply(opt, state, momentum_update):
    """ARM C -- predictive direction (MTT3_MOM_EXTRAP = kappa).

    M_pred = M + kappa*(M - M_slow), with M := the Nesterov momentum mix entering the
    (SOAP) preconditioner and M_slow a slow EMA of M (beta=MTT3_MOM_EXTRAP_BETA;
    persistent buffer "mtt3_mslow" in Muon state, so the generic per-key ckpt clone
    saves/loads it for free). Resuming a ckpt WITHOUT the buffer lazy-inits M_slow=M
    at the fork step, making the first extrap step exactly the base step (no kick).
    The EMA absorbs the current M BEFORE extrapolating, so the trend term equals
    beta*(M - M_slow_old) -- a pure convention; kappa stays the only knob.

    Graveyard caveat (documented per dispatch): the leaderboard's "Muon-history
    forecasting" was removed as NEUTRAL (details unknown). This arm differs -- we
    extrapolate the GRADIENT-SPACE momentum trend before preconditioning, vs their
    PARAM-SPACE forecasting per the #43 lineage. Adjacent mechanism, not a retest of
    a known-dead family, but the neutral prior is informative.
    """
    ms = state.get("mtt3_mslow")
    if ms is None:
        state["mtt3_mslow"] = momentum_update.detach().clone()
        if not _M3.mex_announced:
            _M3.mex_announced = True
            print0(f"mtt3 mom extrap: lazy-init M_slow at muon step {opt.step_count}", console=True)
        return momentum_update
    ms.lerp_(momentum_update, 1.0 - MTT3_MOM_EXTRAP_BETA)
    return momentum_update + MTT3_MOM_EXTRAP * (momentum_update - ms)


def _mtt3_func_polar_apply(opt, p, momentum_update, base_update, ctx):
    """ARM B -- function-space trust region (MTT3_FUNC_POLAR = a; the headline change).

    Q_func = msign(Mtilde @ R) @ R^{-1}, Mtilde = the (SOAP-whitened) momentum entering
    NS, R / R^{-1} from _M3.fp_R / fp_Rinv (Sigma_x EMA -> mtt3_fp_metric_from_sigma,
    refreshed every MTT3_FP_REFRESH steps; metric not built yet -> base unchanged).
    (Q_func R) is exactly orthogonal, i.e. Q_func Sigma_x Q_func^T = I up to the eps
    clamp: the step produces an EQUALIZED output-space change across all singular
    directions -- the generalized polar factor of Mtilde in the input metric, the
    function-space analog of what msign does in weight space.

    Blend (dispatch: norm_preserving_blend-style, renormed to ||Q_base||_F): Q_func is
    pre-scaled to ||Q_base||_F so `a` has half-angle semantics, then
    out = ((1-a)*Q_base + a*Q_func') renormed to ||Q_base||_F in fp64 -- parity
    | ||out|| - ||Q_base|| | / ||Q_base|| ~ 1e-15 (preregistered test bound 1e-6).
    The exact fp64 renorm (avg2 convention) is used INSTEAD of norm_preserving_blend's
    gram-estimate rescale because the estimate only preserves the estimator
    (sum sigma_i^4)^(1/4), not ||.||_F, and cannot meet the 1e-6 parity bound.
    Aspect-ratio scale is inherited through the base norm; downstream radial brake /
    RowFloor / pin / CWD act on the blended update unchanged. Cost: one extra NS + two
    matmuls per scoped param per step (wall-clock free under track-3 scoring).
    """
    key = opt.mtt3_names[p]
    r = _M3.fp_R.get(key)
    r_inv = _M3.fp_Rinv.get(key)
    if r is None:
        return base_update
    # MTT3_FP_DUAL rider: primal (default) feeds Mtilde@R to msign (generalized polar in
    # the input metric); dual feeds Mtilde@R^-1 (exact steepest-descent dual under
    # ||dW.R||_2 <= eta). Same function-space shell, different orientation (knob comment).
    q_func = zeropower_via_newtonschulz5(
        momentum_update.float() @ (r_inv if MTT3_FP_DUAL else r)).float() @ r_inv
    b64 = base_update.double()
    base_fro = b64.norm()
    q64 = q_func.double()
    q64 = q64 * (base_fro / q64.norm().clamp_min(1e-30))
    mix = b64.lerp(q64, MTT3_FUNC_POLAR)
    out = mix * (base_fro / mix.norm().clamp_min(1e-30))
    if ctx is not None:                       # diag-only telemetry (jsonl via the muon pivot)
        ctx["fp_qbase_fro"] = float(base_fro)
        ctx["fp_q_fro"] = float(out.norm())            # parity: == fp_qbase_fro to ~1e-15 rel
        ctx["fp_cos_qbase_qfunc"] = _mtt3_cos(base_update, q_func)
    return out.to(base_update.dtype)
# ======================== end MTT3 (part 2b) ========================


# ========= MTT3 (part 2c: K12 Bi-Maxwell dual-timescale stress memory) =========

def _mtt3_bimaxwell_apply(opt, state, grad, mu):
    """MTT3_BIMAXWELL -- convex two-buffer (Bi-Maxwell) stress memory (K12 Q4.2).

    Base recursion (still executed right before this call; the legacy momentum buffer
    stays maintained as a ckpt/diag PASSENGER but is no longer consumed by the update):
        state["momentum"].lerp_(grad, 1 - mu_t);  update = grad.lerp(momentum, mu_t)
    Bi-Maxwell replaces the single EMA with two fixed-timescale Maxwell buffers:
        M_fast.lerp_(grad, 1 - beta_f);  M_slow.lerp_(grad, 1 - beta_s)   (betas FIXED)
        M_eff = w*M_fast + (1-w)*M_slow                     (CONVEX combination, w in [0,1])
        momentum_update = grad.lerp(M_eff, mu_t)            (SAME Nesterov mix, SAME
                                                             scheduled mu_t as base)
    Exact semantics (documented per dispatch):
      * The mu warmup/cooldown schedule applies ONLY to the Nesterov mix above;
        beta_f/beta_s never move. Degenerate parity: with beta_f == beta_s the two
        buffers stay bit-identical forever (equal init below), so w cancels EXACTLY at
        any mu_t; with beta_f = beta_s = 0.95 and buffers initialized on the mu=0.95
        plateau (e.g. the fork@700 lazy init), the recursion is the base momentum
        recursion BIT-EXACTLY for as long as the schedule stays at 0.95 (steps
        300..2700 -- the whole fork@700 lever window). Guarded by tests/test p.
      * Lazy init (avg2 convention): buffers absent (fresh run OR resume from a ckpt
        without them) -> M_fast = M_slow = state["momentum"] AFTER this step's base
        lerp and NO own-rate lerp this step, so the init-step update is bit-equal base;
        from then on the buffers ride every ckpt via the generic muon_state clone
        (checksum groups muon_bm_mf / muon_bm_ms).
      * Delayed start (MTT3_BM_START, K13 rider): the caller gates this function on
        step_count >= MTT3_BM_START, so a from-scratch run with BM_START = S follows the
        exact base path (no buffers at all) until step S, then hits the SAME lazy init
        above -- reproducing a fork@S resume without needing a checkpoint.
      * w-anneal rider (MTT3_BM_W_END + MTT3_BM_W_RAMP="t0:t1", bm_thermo_round2
        prereg "anneal" arm): the convex weight follows mtt3_bm_w_at(opt.step_count,
        MTT3_BM_W, MTT3_BM_W_END, t0, t1) -- constant MTT3_BM_W through t0, linear to
        MTT3_BM_W_END at t1, constant after; the betas above never move. Unset
        (default) -> w IS the constant MTT3_BM_W float, so the lerp below is the exact
        pre-knob computation (byte-identical invariant, tests/test p).
      * Graveyard distance: convex kernel-shape change of the FIRST-moment stress
        memory only -- NOT mex30's trend extrapolation M + kappa*(M - M_slow) (K10
        negative), NOT SOAP beta2/denominator memory (K8 negative), and msign / the
        update spectrum downstream are untouched (radius-pin survival: directional).
    """
    mf = state.get("mtt3_bm_mf")
    if mf is None:
        state["mtt3_bm_mf"] = state["momentum"].detach().clone()
        state["mtt3_bm_ms"] = state["momentum"].detach().clone()
        if not _M3.bm_announced:
            _M3.bm_announced = True
            print0(f"mtt3 bimaxwell: lazy-init M_fast=M_slow=M at muon step {opt.step_count}",
                   console=True)
    else:
        mf.lerp_(grad, 1.0 - MTT3_BM_BETA_F)
        state["mtt3_bm_ms"].lerp_(grad, 1.0 - MTT3_BM_BETA_S)
    w = (mtt3_bm_w_at(opt.step_count, MTT3_BM_W, MTT3_BM_W_END,
                      MTT3_BM_RAMP_T0, MTT3_BM_RAMP_T1)
         if MTT3_BM_W_SCHED else MTT3_BM_W)
    m_eff = torch.lerp(state["mtt3_bm_ms"], state["mtt3_bm_mf"], w)
    return grad.lerp(m_eff, mu)
# ======================== end MTT3 (part 2c) ========================


# ========= MTT3 (part 2d: K12.1 FDT bucketed Kalman-memory Muon) =========

def _mtt3_fdt_basis(key, mc_f, step):
    """Cached SVD basis of the fp32 center buffer M_c, refreshed every MTT3_FDT_EVERY
    muon steps (_M3-only cache, never in ckpts). On a non-refresh step the previous
    basis is reused (the splice's residual term reclaims any drift of M_c outside the
    stale span). SVD failure keeps the stale basis; with no cache yet -> None (caller
    falls back to the center buffer)."""
    ent = _M3.fdt_basis.get(key)
    if ent is None or step % MTT3_FDT_EVERY == 0:
        try:
            u, _, vh = torch.linalg.svd(mc_f, full_matrices=False)
            ent = _M3.fdt_basis[key] = {"u": u, "vh": vh}
        except RuntimeError:
            ent = _M3.fdt_basis.get(key)
    return ent


def _mtt3_fdt_estimate(p, grad, key, basis):
    """FDT online per-(tensor, bucket) T-hat / D-hat / K* -- the I-probe estimator
    VERBATIM (_mtt3_iprobe_probe recursions: EMA beta MTT3_IPROBE_EMA_BETA, floor
    MTT3_IPROBE_DMIN, n_eff correction via _mtt3_bucket_counts), with ONE documented
    basis difference: measured in the FDT's own ACTION basis, the SVD of the PRE-SOAP
    center buffer M_c (I-probe measures in the post-SOAP M-tilde basis -- the
    intervention run self-measures on the basis it acts in; I-probe only carried the
    "heterogeneity exists" evidence for firing, design spec section 2 step 2).

        s_b = bucket-mean of u_i^T G v_i     (G = full grad, fp32)
        T_b = bucket-mean of (u_i^T N v_i)^2 (N = 2A - G, exact O1 half split)
        That <- EMA[T_b];  Dema <- EMA[(s_b - prev_s_b)^2]
        Dhat = max(Dema - 2*That/n_eff, MTT3_IPROBE_DMIN);  K* = sqrt(Dhat/That)

    Returns the per-bucket K* list (NaN while warming: prev_s needs 1 step, Dema
    needs 2) or None when the half-grad / basis is unavailable. State lives ONLY in
    _M3.fdt_state (resume re-warms; I-probe shadow convention)."""
    a = _M3.iprobe_half.get(p)
    if a is None or basis is None:
        return None
    G = grad.float()
    N = a * 2.0 - G                                  # (Ga-Gb)/2; Var per direction ~ T
    u, vh = basis["u"], basis["vh"]
    proj_sig = ((u.T @ G) * vh).sum(dim=1)           # u_i^T G v_i
    proj_noise = ((u.T @ N) * vh).sum(dim=1)
    s_list = _mtt3_bucket_means(proj_sig)
    T_list = _mtt3_bucket_means(proj_noise.square())
    ist = _M3.fdt_state.get(key)
    if ist is None:
        ist = _M3.fdt_state[key] = {"prev_s": None, "That": None, "Dema": None,
                                    "neff": _mtt3_bucket_counts(proj_sig.numel())}
    bt = MTT3_IPROBE_EMA_BETA
    if ist["That"] is None:
        ist["That"] = list(T_list)
    else:
        ist["That"] = [o if math.isnan(t) else bt * o + (1.0 - bt) * t
                       for o, t in zip(ist["That"], T_list)]
    nb = len(MTT3_SPEC_BUCKETS) - 1
    k_list = [float("nan")] * nb
    prev = ist["prev_s"]
    if prev is not None:
        inc2 = [(s - ps) ** 2 for s, ps in zip(s_list, prev)]
        if ist["Dema"] is None:
            ist["Dema"] = inc2
        else:
            ist["Dema"] = [o if math.isnan(i2) else bt * o + (1.0 - bt) * i2
                           for o, i2 in zip(ist["Dema"], inc2)]
        for b in range(nb):
            dema, that = ist["Dema"][b], ist["That"][b]
            if math.isnan(dema) or math.isnan(that):
                continue
            dhat = max(dema - 2.0 * that / ist["neff"][b], MTT3_IPROBE_DMIN)
            if that > 0.0:
                k_list[b] = math.sqrt(dhat / that)
    ist["prev_s"] = s_list
    return k_list


def _mtt3_fdt_apply(opt, p, state, grad, mu):
    """MTT3_FDT -- per-spectral-bucket Kalman-memory splice of the first-moment stress
    memory (design spec: designs/fdt_bucketed_memory_implementation.md, K12.1 PASS).

    Base recursion (still executed right before this call; the legacy momentum buffer
    stays maintained as a ckpt/diag PASSENGER but is no longer consumed by the update):
        state["momentum"].lerp_(grad, 1 - mu_t);  update = grad.lerp(momentum, mu_t)
    FDT replaces the single EMA with FIVE fixed-timescale buffers spliced per bucket:
        M_j.lerp_(grad, alpha_j), alpha_j in MTT3_FDT_ALPHA_GRID (rates FIXED; j=2 = M_c)
        basis u = svd(M_c.float()) cached/refreshed every MTT3_FDT_EVERY steps
        per-bucket K* from the online estimator (_mtt3_fdt_estimate, I-probe math)
        alpha_use(b) = exp((1-gamma)*log 0.05 + gamma*log clip(K*_b, 0.02, 0.12))
                       (warm-up / NaN buckets -> exactly 0.05, the center)
        M_eff = mtt3_fdt_splice(u, M_0..4, alpha_use)   (left-coefficient splice +
                                                         center-residual, fp32)
        momentum_update = grad.lerp(M_eff.to(grad.dtype), mu_t)   (SAME Nesterov mix,
                                                                   SAME scheduled mu_t)
    Exact semantics (documented per the frozen spec):
      * The mu warmup/cooldown schedule applies ONLY to the Nesterov mix; the buffer
        rates never move. Parity anchor = OFF (unset/"0"), NOT the gamma->0 limit
        (gamma=0 still runs the fixed-rate M_c vs the mu-scheduled base lerp -- they
        differ off the mu=0.95 plateau; BM-style semantics).
      * Lazy init (BM/avg2 convention): buffers absent (fresh run OR resume from a
        base ckpt) -> all five buffers = state["momentum"] AFTER this step's base lerp,
        NO own-rate lerp this step, and the returned update is the EXACT base
        computation grad.lerp(momentum, mu) -> init-step update bit-equal base; the
        estimator still starts warming this step (read-only). Announced once.
      * Delayed start (MTT3_FDT_START = S, BM_START semantics): steps < S are the exact
        base path (no buffers, no capture consumption) -> the from-scratch twin of a
        fork@S resume.
      * Buffers ride every ckpt via the generic muon_state clone (checksum groups
        muon_fdt_m0..m4); estimator state + basis cache live ONLY in _M3 -> a resume
        re-warms them (documented; I-probe shadow convention). Consequence: a mid-FDT
        ckpt roundtrip is bit-exact only when the resume point precedes/coincides with
        the first estimator step; later resumes re-warm (prereg'd, tests/test q).
      * Half-batch capture: shares _M3.iprobe_half (same buffers, no duplicate memory);
        the halfgrad hook + prestep all-reduce fire whenever FDT is on, independent of
        MTT3_DIAG (pure-run component; n_mb >= 2 required, RuntimeError otherwise).
      * Graveyard distance (spec section 6): changes the TIMESCALE SPECTRUM of the
        first-moment memory kernel (directional, pin-safe); msign and the update
        spectrum downstream are untouched -- NOT the K12 equivalence class
        Q = U f(stat) V^T (update-magnitude scaling), NOT SOAP beta2 (K8 negative),
        NOT mex30 trend extrapolation (K10 negative). BM comparison arm: same
        "kernel shape" survival axis, per-bucket single-point vs channel-uniform
        two-point kernel.
    """
    t0 = time.perf_counter()
    step = opt.step_count
    key = opt.mtt3_names[p]
    lazy = state.get("mtt3_fdt_m0") is None
    if lazy:
        for j in range(len(MTT3_FDT_ALPHA_GRID)):
            state[f"mtt3_fdt_m{j}"] = state["momentum"].detach().clone()
        if not _M3.fdt_announced:
            _M3.fdt_announced = True
            print0(f"mtt3 fdt: lazy-init M_0..M_4=momentum at muon step {step}", console=True)
    else:
        for j, a_j in enumerate(MTT3_FDT_ALPHA_GRID):
            state[f"mtt3_fdt_m{j}"].lerp_(grad, a_j)
    mc = state[f"mtt3_fdt_m{MTT3_FDT_CENTER}"]
    basis = _mtt3_fdt_basis(key, mc.float(), step)
    kstar = _mtt3_fdt_estimate(p, grad, key, basis)
    if lazy:
        # init step: M_eff = M_c = momentum by construction -> return the exact base
        # computation (bit-equal base; estimator warm-up already started above).
        out = grad.lerp(state["momentum"], mu)
    elif basis is None:
        # no basis available at all (SVD failed with an empty cache): center fallback
        out = grad.lerp(mc, mu)
    else:
        nb = len(MTT3_SPEC_BUCKETS) - 1
        alpha_use = mtt3_fdt_alpha_use(kstar if kstar is not None else [float("nan")] * nb,
                                       MTT3_FDT_GAMMA,
                                       alpha_center=MTT3_FDT_ALPHA_GRID[MTT3_FDT_CENTER],
                                       clip_lo=MTT3_FDT_CLIP[0], clip_hi=MTT3_FDT_CLIP[1])
        m_list = [state[f"mtt3_fdt_m{j}"] for j in range(len(MTT3_FDT_ALPHA_GRID))]
        m_eff = mtt3_fdt_splice(basis["u"], m_list, alpha_use,
                                center_idx=MTT3_FDT_CENTER,
                                alpha_grid=MTT3_FDT_ALPHA_GRID,
                                spec_buckets=MTT3_SPEC_BUCKETS)
        out = grad.lerp(m_eff.to(grad.dtype), mu)
    _M3.fdt_t += time.perf_counter() - t0
    # overhead self-report (spec section 5: iprobe_s-style timing, console every 100 steps;
    # once per STEP, not per tensor)
    if step % 100 == 0 and _M3.fdt_print_step != step:
        _M3.fdt_print_step = step
        print0(f"mtt3 fdt: step:{step} cumulative fdt_s={_M3.fdt_t:.1f}", console=True)
    return out
# ======================== end MTT3 (part 2d) ========================


# ====================== MTT3 (part 2: Muon-step observers) ======================
# All hooks below are READ-ONLY w.r.t. training state: they never mutate p, p.grad,
# optimizer state, or the update tensor, and consume no global RNG. Each returns
# early (None ctx) when MTT3_DIAG=0.

def _mtt3_muon_pre(opt, p, grad, state):
    """Per 2-D Muon param, before the momentum update. Returns diag ctx or None."""
    if not MTT3_DIAG:
        return None
    t0 = time.perf_counter()
    key = opt.mtt3_names[p]
    step = opt.step_count
    buf = _M3.p0.get(p)
    if buf is None:
        buf = _M3.p0[p] = torch.empty_like(p)
    buf.copy_(p.detach())
    ctx = {"key": key, "step": step}
    pf = p.detach().float()
    ctx["tg"] = float((pf * grad.float()).sum())     # theta . g   (grad = SUM-reduced batch grad)
    ctx["t2"] = float(pf.square().sum())             # theta^2
    if _mtt3_is_o1_step(step) and p in _M3.half_grad and len(state) > 0:
        _M3.mom_old[p] = state["momentum"].detach().clone()   # momentum BEFORE this step's lerp
    # ===== K12 I-probe: EVERY-step pre-lerp momentum capture for the probe pair
    # (preallocated buffer, read-only; iprobe_half is filled by the halfgrad hook) =====
    if MTT3_IPROBE and p in _M3.iprobe_half and len(state) > 0:
        t0i = time.perf_counter()
        mbuf = _M3.iprobe_mom_old.get(p)
        if mbuf is None:
            mbuf = _M3.iprobe_mom_old[p] = torch.empty_like(state["momentum"])
        mbuf.copy_(state["momentum"])
        _M3.iprobe_t += time.perf_counter() - t0i
    _M3.diag_t += time.perf_counter() - t0
    return ctx


def _mtt3_muon_mid(ctx, p, grad, state, momentum_update, update, mu):
    """After muon_update (NS polar): O1 replica probes, O7 M-tilde spectrum, panel cosines."""
    if ctx is None:
        return
    t0 = time.perf_counter()
    step, key = ctx["step"], ctx["key"]
    is_panel = key in MTT3_PANEL_KEYS
    lvl2 = MTT3_DIAG_LEVEL >= 2
    mt = momentum_update.float()                     # M-tilde: whitened momentum entering NS
    ctx["q_fro"] = float(update.float().norm())      # ||Q||_F entering the radial brake (scale audit)
    if lvl2 and ((step % MTT3_O7_FULL_EVERY == 0) or (is_panel and step % MTT3_O7_PANEL_EVERY == 0)):
        try:
            s_mt = torch.linalg.svdvals(mt)
            occ, ent = _mtt3_spec_occupation(s_mt)
            ctx["o7_p_bucket_mtilde"] = occ
            ctx["o7_entropy_mtilde"] = ent
            if step % MTT3_O7_FULL_EVERY == 0:
                _M3.spectra.setdefault(key, {})["mtilde"] = s_mt.detach().cpu()
        except RuntimeError:
            pass
    if _mtt3_is_o1_step(step) and (is_panel or _mtt3_o1_all72(step)) and p in _M3.half_grad:
        # Replica probes from the grad-accumulation split (documented in mtt3_meta.json):
        #   A  = p.grad materialized after the first half of the microbatches (exact by construction)
        #   G  = the full SUM-reduced grad the optimizer consumes
        #   Ga = 2A, Gb = 2(G-A); (Ga+Gb)/2 == G exactly in real arithmetic (<=1 ulp in fp);
        #   the noise/signal probes below use D = 2A-G and S = G directly (exact).
        A = _M3.half_grad.pop(p)
        G = grad.float()
        Ga = A * 2.0
        Gb = (G - A) * 2.0
        mom_old = _M3.mom_old.pop(p, None)
        mom_old = (mom_old if mom_old is not None else state["momentum"].detach()).float()
        # same mu mixing as the real step: m' = lerp(m, g, 1-mu); update = lerp(g, m', mu)
        Ma = torch.lerp(Ga, torch.lerp(mom_old, Ga, 1.0 - mu), mu)
        Mb = torch.lerp(Gb, torch.lerp(mom_old, Gb, 1.0 - mu), mu)
        qa_raw = zeropower_via_newtonschulz5(Ma)
        qb_raw = zeropower_via_newtonschulz5(Mb)
        ctx["o1_theta_raw"] = 1.0 - _mtt3_cos(qa_raw, qb_raw)
        qa = zeropower_via_newtonschulz5(_mtt3_soap_ro(Ma, state))   # CURRENT SOAP state, read-only
        qb = zeropower_via_newtonschulz5(_mtt3_soap_ro(Mb, state))
        ctx["o1_theta_soap"] = 1.0 - _mtt3_cos(qa, qb)
        try:
            U, S, Vh = torch.linalg.svd(mt, full_matrices=False)
            D = A * 2.0 - G                          # (Ga-Gb)/2 == 2A-G exactly
            proj_sig = ((U.T @ G) * Vh).sum(dim=1)   # u_i^T G v_i per singular direction
            proj_noise = ((U.T @ D) * Vh).sum(dim=1)
            ctx["o1_bucket_s"] = _mtt3_bucket_means(proj_sig.square())
            ctx["o1_bucket_T"] = _mtt3_bucket_means(proj_noise.square())
            if is_panel:
                ctx["o1_sigma_max"] = float(S.max())
                ctx["o1_sigma_min"] = float(S.min())
        except RuntimeError:
            pass
        if is_panel:
            ctx["o1_mtilde_fro"] = float(mt.norm())
            raw_mom = grad.lerp(state["momentum"], mu)          # pre-SOAP mix, recomputed read-only
            ctx["o1_cos_raw_soap"] = _mtt3_cos(zeropower_via_newtonschulz5(raw_mom), update)
            pq = _M3.prev_q.get(key)
            if pq is not None:
                ctx["o1_cos_q_prev"] = _mtt3_cos(update, pq)
    if is_panel and lvl2:
        buf = _M3.prev_q.get(key)
        if buf is None:
            _M3.prev_q[key] = update.detach().float().clone()
        else:
            buf.copy_(update.detach().float())
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_iprobe_probe(ctx, opt, p, grad, state, momentum_update, mu):
    """K12 I-probe: EVERY-step FDT/Kalman measurement on one hidden matrix (read-only).

    Measurement basis (K12 1.2): the SVD of the CURRENT SOAP-whitened momentum M-tilde
    (the matrix entering NS) -- the channels msign actually saturates. Per spectral
    bucket b (the existing 5 index buckets over singular directions):
        s_b = mean_{i in b} u_i^T G v_i        (SIGNED mean projection of the full grad)
        T_b = mean_{i in b} (u_i^T N v_i)^2    (N = 2A - G = (Ga-Gb)/2; per-direction
                                                batch-noise variance of G, exact O1 split)
    Online estimator per (tensor, bucket), EMA beta MTT3_IPROBE_EMA_BETA, lazy init:
        That_b <- EMA[T_b]
        Dema_b <- EMA[(s_b(t) - s_b(t-1))^2]                (innovation second moment)
        Dhat_b  = max(Dema_b - 2*That_b/n_eff_b, 1e-12)     (measurement-noise correction)
        K*_b    = sqrt(Dhat_b / That_b),  logK*_b = log K*_b
    n_eff_b = #directions in bucket b (_mtt3_bucket_counts): s_b is the mean of n_eff
    projections, so Var(noise of s_b) = T_b/n_eff and consecutive differencing doubles
    it -- the 2*That/n_eff subtraction. Cross-direction independence is a DOCUMENTED
    approximation (correlated noise / basis rotation bias Dhat upward); K* mixes the
    bucket-mean drift Dhat with the per-direction That (the exact K12 dispatch form),
    so only IQR(log K*) and the cross-window bucket ORDERING are preregistered reads
    (analysis/analyze_iprobe.py; prereg K12 section).

    Probe pair (EVERY step, dual NS -- the O1 recipe at full cadence): Ma/Mb =
    momentum-mixed halves from the pre-lerp momentum, read-only SOAP (current shared
    state, no trust gate -- O1 caveat), NS -> iprobe_theta = 1 - cos(Qa, Qb).

    Shadow multi-timescale memories (K12 1.5): THREE fixed-beta shadow momentum buffers
    beta in MTT3_IPROBE_SHADOW_BETAS fed the real grads. From scratch (muon step 0)
    they start at zeros and lerp every step (exact fixed-beta EMA from t=0); on a
    resume without buffers they lazy-init M_shadow = M (announced; avg2 convention).
    They live in _M3 ONLY -- never in optimizer state, never in ckpts (tests/test o).
    At the shadow cadence, panel tensors only: cos(msign_NS5(M_shadow), msign_NS5(M_real))
    and predicted work <g, msign_NS5(.)> for real + each shadow.

    Wall time accumulates into iprobe_s (jsonl, rank-local) and diag_s.
    """
    if ctx is None:
        return
    a = _M3.iprobe_half.get(p)
    if a is None:
        return
    t0 = time.perf_counter()
    key = opt.mtt3_names[p]
    step = opt.step_count
    G = grad.float()
    N = a * 2.0 - G                                  # (Ga-Gb)/2; Var per direction ~ T
    mt = momentum_update.float()
    try:
        U, S, Vh = torch.linalg.svd(mt, full_matrices=False)
    except RuntimeError:
        U = None
    if U is not None:
        proj_sig = ((U.T @ G) * Vh).sum(dim=1)       # u_i^T G v_i
        proj_noise = ((U.T @ N) * Vh).sum(dim=1)
        s_list = _mtt3_bucket_means(proj_sig)
        T_list = _mtt3_bucket_means(proj_noise.square())
        ist = _M3.iprobe_state.get(key)
        if ist is None:
            ist = _M3.iprobe_state[key] = {"prev_s": None, "That": None, "Dema": None,
                                           "neff": _mtt3_bucket_counts(proj_sig.numel())}
        bt = MTT3_IPROBE_EMA_BETA
        if ist["That"] is None:
            ist["That"] = list(T_list)
        else:
            ist["That"] = [o if math.isnan(t) else bt * o + (1.0 - bt) * t
                           for o, t in zip(ist["That"], T_list)]
        nb = len(MTT3_SPEC_BUCKETS) - 1
        d_list = [float("nan")] * nb
        k_list = [float("nan")] * nb
        logk_list = [float("nan")] * nb
        prev = ist["prev_s"]
        if prev is not None:
            inc2 = [(s - ps) ** 2 for s, ps in zip(s_list, prev)]
            if ist["Dema"] is None:
                ist["Dema"] = inc2
            else:
                ist["Dema"] = [o if math.isnan(i2) else bt * o + (1.0 - bt) * i2
                               for o, i2 in zip(ist["Dema"], inc2)]
            for b in range(nb):
                dema, that = ist["Dema"][b], ist["That"][b]
                if math.isnan(dema) or math.isnan(that):
                    continue
                dhat = max(dema - 2.0 * that / ist["neff"][b], MTT3_IPROBE_DMIN)
                d_list[b] = dhat
                if that > 0.0:
                    k_list[b] = math.sqrt(dhat / that)
                    logk_list[b] = math.log(k_list[b])
        ist["prev_s"] = s_list
        ctx["iprobe_sT"] = {"s": s_list, "T": T_list}
        ctx["iprobe_D"] = d_list
        ctx["iprobe_Kstar"] = k_list
        ctx["iprobe_logK"] = logk_list
    # --- momentum-mixed dual-NS probe pair (O1 recipe, EVERY step) ---
    mom_old = _M3.iprobe_mom_old.get(p)
    if mom_old is not None:
        Ga = a * 2.0
        Gb = (G - a) * 2.0
        mo = mom_old.float()
        Ma = torch.lerp(Ga, torch.lerp(mo, Ga, 1.0 - mu), mu)
        Mb = torch.lerp(Gb, torch.lerp(mo, Gb, 1.0 - mu), mu)
        if p in opt.soap_params:
            Ma = _mtt3_soap_ro(Ma, state)            # CURRENT SOAP state, read-only, no gate
            Mb = _mtt3_soap_ro(Mb, state)
        qa = zeropower_via_newtonschulz5(Ma)
        qb = zeropower_via_newtonschulz5(Mb)
        ctx["iprobe_theta"] = 1.0 - _mtt3_cos(qa, qb)
    # --- shadow multi-timescale memories (diag-only; never in optimizer state) ---
    sh = _M3.iprobe_shadow.get(p)
    lerp_now = True
    if sh is None:
        if step == 0:
            sh = {b: torch.zeros_like(state["momentum"], dtype=torch.float32)
                  for b in MTT3_IPROBE_SHADOW_BETAS}  # exact fixed-beta EMA from t=0
        else:
            sh = {b: state["momentum"].detach().float().clone()
                  for b in MTT3_IPROBE_SHADOW_BETAS}  # resume convention: M_shadow=M, no lerp now
            lerp_now = False
            if not _M3.iprobe_announced:
                _M3.iprobe_announced = True
                print0(f"mtt3 iprobe: shadow lazy-init M_shadow=M at muon step {step}",
                       console=True)
        _M3.iprobe_shadow[p] = sh
    if lerp_now:
        for b in MTT3_IPROBE_SHADOW_BETAS:
            sh[b].lerp_(G, 1.0 - b)
    if step % MTT3_IPROBE_SHADOW_EVERY == 0 and key in MTT3_PANEL_KEYS:
        q_real = _mtt3_ns_k(state["momentum"].detach().float(), MTT3_IPROBE_NS).float()
        cos_d = {}
        work_d = {"real": float((G * q_real).sum())}
        for b in MTT3_IPROBE_SHADOW_BETAS:
            q_sh = _mtt3_ns_k(sh[b], MTT3_IPROBE_NS).float()
            cos_d[f"{b:g}"] = _mtt3_cos(q_sh, q_real)
            work_d[f"{b:g}"] = float((G * q_sh).sum())
        ctx["iprobe_shadow_cos"] = cos_d
        ctx["iprobe_shadow_work"] = work_d
    dt = time.perf_counter() - t0
    _M3.iprobe_t += dt
    _M3.diag_t += dt


def _mtt3_muon_prepin(ctx, p, update, target_radius, lr):
    """Between p.add_ and rescale_to_radius: pre-pin displacement split + pin factor."""
    if ctx is None:
        return
    t0 = time.perf_counter()
    p0 = _M3.p0[p].float()
    w_norm = p0.norm().clamp_min(1e-12)
    uf = update.float()
    d_norm2 = (lr * lr) * uf.square().sum()
    d_par = lr * (uf * p0).sum().abs() / w_norm
    d_perp = (d_norm2 - d_par.square()).clamp_min(0.0).sqrt()
    ctx["o5_w_norm"] = float(w_norm)
    ctx["o5_dw_par_pre"] = float(d_par)
    ctx["o5_dw_perp_pre"] = float(d_perp)
    ctx["o5_omega_pre"] = float(d_perp / w_norm)
    after_norm = p.detach().float().norm().clamp_min(1e-8)
    ctx["o5_rescale_factor"] = float(target_radius / after_norm)  # factor the pin is about to apply
    ctx["_target_radius"] = float(target_radius)
    if MTT3_LR_FAMILY_SCALE:   # per-param lr audit (key absent when the knob is unset)
        ctx["lr_eff"] = float(lr)
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_muon_post(ctx, p, grad, state, cwd_mask):
    """After the pin + post-pin CWD: actual-displacement observables (O2+/O5+/O3/O1-postpin/O7-dW)."""
    if ctx is None:
        return
    t0 = time.perf_counter()
    step, key = ctx["step"], ctx["key"]
    lvl2 = MTT3_DIAG_LEVEL >= 2
    p0 = _M3.p0[p].float()
    dw = p.detach().float() - p0
    dw_norm = dw.norm()
    ctx["v2"] = float(dw.square().sum())             # ||dW||^2, actual applied displacement
    gf = grad.float()
    ctx["o2p_work"] = float((gf * dw).sum())         # g . dW
    ctx["o2p_cos"] = _mtt3_cos(gf, dw)
    w_norm = p0.norm().clamp_min(1e-12)
    d_par = (dw * p0).sum().abs() / w_norm
    d_perp = (dw.square().sum() - d_par.square()).clamp_min(0.0).sqrt()
    ctx["o5_dw_par_post"] = float(d_par)
    ctx["o5_dw_perp_post"] = float(d_perp)
    ctx["o5_omega_post"] = float(d_perp / w_norm)
    tr = ctx.pop("_target_radius", None)
    if tr:
        ctx["o5_cwd_radius_drift"] = float(p.detach().float().norm() / tr - 1.0)
    if cwd_mask is not None:
        ctx["o5_cwd_mask_frac"] = float(cwd_mask.float().mean())
    if lvl2 and float(dw_norm) > 1e-30:
        dw_dir = dw / dw_norm
        if _mtt3_is_o1_step(step) and (key in MTT3_PANEL_KEYS or _mtt3_o1_all72(step)):
            pd = _M3.prev_dw.get(key)
            if pd is not None:
                ctx["o1_theta_postpin"] = 1.0 - _mtt3_cos(dw_dir, pd)   # consecutive applied dW dirs
        if step % MTT3_O3_EVERY == 0:
            for h in MTT3_O3_HORIZONS:
                ema = _M3.o3_ema[h].get(key)
                if ema is not None:
                    ctx[f"o3_cos_h{h}"] = _mtt3_cos(dw_dir, ema)
                    alpha = MTT3_O3_EVERY / h
                    ema.mul_(1.0 - alpha).add_(dw_dir, alpha=alpha)
                else:
                    _M3.o3_ema[h][key] = dw_dir.clone()
        buf = _M3.prev_dw.get(key)
        if buf is None:
            _M3.prev_dw[key] = dw_dir.clone()
        else:
            buf.copy_(dw_dir)
    if lvl2 and ((step % MTT3_O7_FULL_EVERY == 0) or (key in MTT3_PANEL_KEYS and step % MTT3_O7_PANEL_EVERY == 0)):
        try:
            s_dw = torch.linalg.svdvals(dw)
            occ, ent = _mtt3_spec_occupation(s_dw)
            ctx["o7_p_bucket_dw"] = occ
            ctx["o7_entropy_dw"] = ent
            if step % MTT3_O7_FULL_EVERY == 0:
                _M3.spectra.setdefault(key, {})["dw"] = s_dw.detach().cpu()
        except RuntimeError:
            pass
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_muon_o6_pre(ctx, state, use_soap):
    """Clone the SOAP eigenbasis before this step's refresh (O6 ticks only)."""
    if ctx is None or not use_soap or MTT3_DIAG_LEVEL < 2 or ctx["step"] % MTT3_O6_EVERY != 0:
        return None
    if state.get("q_row") is None:
        return None
    t0 = time.perf_counter()
    out = (state["q_row"].detach().clone(), state["q_col"].detach().clone())
    _M3.diag_t += time.perf_counter() - t0
    return out


def _mtt3_muon_o6_post(ctx, state, qold):
    """Basis drift across the refresh + Gram eigengap (small side only; cost guard)."""
    if ctx is None or qold is None:
        return
    t0 = time.perf_counter()
    q_row_old, q_col_old = qold
    for side, q_old, q_new, gg in (("row", q_row_old, state["q_row"], state["row_gg"]),
                                   ("col", q_col_old, state["q_col"], state["col_gg"])):
        if q_new is not None and q_old.shape == q_new.shape:
            eye = torch.eye(q_new.size(1), device=q_new.device, dtype=torch.float32)
            ctx[f"o6_drift_{side}"] = float((eye - q_old.T @ q_new).norm())
        if gg.size(0) <= 1024:   # guard: eigvalsh only on the <=768 Gram side (fc row side is 3072)
            try:
                ev = torch.linalg.eigvalsh(gg)
                ctx[f"o6_eigengap_{side}"] = float((ev[-1] - ev[-2]) / ev[-1].abs().clamp_min(1e-30))
                ctx[f"o6_lam1_{side}"] = float(ev[-1])
            except RuntimeError:
                pass
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_muon_collect(ctx):
    """Move the finished per-tensor ctx into this step's collection."""
    if ctx is None:
        return
    ctx.pop("step", None)
    key = ctx.pop("key")
    _M3.muon[key] = ctx

class Muon(torch.optim.Optimizer):
    def __init__(self, named_params, lr=0.02, weight_decay=0, mu=0.95):
        assert isinstance(named_params, list) and len(named_params) >= 1
        self.soap_params = {p for n, p in named_params if should_soap_param(n)}
        self.attn_soap_params = {p for n, p in named_params if should_soap_param(n) and is_attn_param(n)}
        self.attn_proj_soap_params = {p for n, p in named_params if should_soap_param(n) and is_attn_proj_param(n)}
        self.step_count = 0
        self.mtt3_names = {p: _mtt3_short(n) for n, p in named_params}  # ===== MTT3: name map for observers (no behavior change)
        # ===== MTT3: per-param family LR scale (all 1.0 when MTT3_LR_FAMILY_SCALE unset) =====
        self.mtt3_family_scale = {p: _mtt3_family_scale_for(k) for p, k in self.mtt3_names.items()}
        # ===== MTT3 (ARM B): function-space polar scope set (never read while MTT3_FUNC_POLAR=0) =====
        self.mtt3_fp_params = {p for n, p in named_params if _mtt3_fp_scope_param(n)}
        params = sorted([p for _, p in named_params], key=lambda x: x.size(), reverse=True)
        defaults = dict(lr=lr, weight_decay=weight_decay, mu=mu)
        super().__init__(params, defaults)

    @torch.no_grad()
    def step(self):
        world_size = dist.get_world_size()
        rank = dist.get_rank()
        for group in self.param_groups:
            params = group["params"]
            params_pad = params + [torch.empty_like(params[-1])] * (world_size - len(params) % world_size)
            for base_i in range(0, len(params), world_size):
                if base_i + rank < len(params):
                    p = params[base_i + rank]
                    state = self.state[p]
                    if len(state) == 0:
                        state["momentum"] = torch.zeros_like(p)
                        if p in self.soap_params:
                            state["exp_avg_sq"] = torch.zeros_like(p, dtype=torch.float32)
                            state["row_gg"] = torch.zeros(p.size(0), p.size(0), dtype=torch.float32, device=p.device)
                            state["col_gg"] = torch.zeros(p.size(1), p.size(1), dtype=torch.float32, device=p.device)
                            state["q_row"] = None
                            state["q_col"] = None
                            state["soap_step"] = 0
                    grad = p.grad
                    # ===== MTT3: per-param family LR scale (unset -> _m3_fs = 1.0 for every
                    # param and x*1.0 is bit-exact, so the base run stays byte-identical) =====
                    _m3_fs = self.mtt3_family_scale[p]
                    _m3_lr = group["lr"] * _m3_fs
                    # ===== MTT3: read-only pre-update capture (None when MTT3_DIAG=0) =====
                    _m3 = _mtt3_muon_pre(self, p, grad, state)
                    state["momentum"].lerp_(grad, 1 - group["mu"])
                    momentum_update = grad.lerp(state["momentum"], group["mu"])
                    # ===== MTT3 (K12 Bi-Maxwell): replace the single-EMA stress memory the
                    # update consumes with the convex two-buffer form (fixed beta_f/beta_s
                    # buffers ride the ckpt; the legacy momentum buffer above stays maintained
                    # as a passenger; env unset -> dead code, exact base path). K13 rider:
                    # steps < MTT3_BM_START stay on the exact base path too (default 0 =
                    # from step 0); the first gated step lazy-inits M_fast=M_slow=momentum,
                    # the from-scratch twin of a fork@BM_START resume. =====
                    if MTT3_BIMAXWELL and self.step_count >= MTT3_BM_START:
                        momentum_update = _mtt3_bimaxwell_apply(self, state, grad, group["mu"])
                    # ===== MTT3 (K12.1 FDT): replace the single-EMA stress memory the update
                    # consumes with the per-spectral-bucket Kalman-memory splice (five fixed-
                    # rate buffers ride the ckpt; the legacy momentum buffer above stays
                    # maintained as a passenger; env unset/"0" -> dead code, exact base path).
                    # Steps < MTT3_FDT_START stay on the exact base path (from-scratch twin of
                    # a fork@START resume); the first gated step lazy-inits all five buffers =
                    # momentum and returns the bit-equal base update. Acts BEFORE the SOAP
                    # precondition (intervention point = memory formation; spec section 2). =====
                    if MTT3_FDT_GAMMA > 0.0 and self.step_count >= MTT3_FDT_START:
                        momentum_update = _mtt3_fdt_apply(self, p, state, grad, group["mu"])
                    # ===== MTT3 (ARM C mom-extrap): predictive direction, applied BEFORE the
                    # SOAP precondition (M_slow buffer rides the ckpt lazily; kappa=0 -> dead
                    # code, exact base path). =====
                    if MTT3_MOM_EXTRAP != 0.0:
                        momentum_update = _mtt3_mom_extrap_apply(self, state, momentum_update)
                    is_attn_soap = p in self.attn_soap_params
                    use_soap = p in self.soap_params
                    if use_soap:
                        if is_attn_soap:
                            soap_update = soap_precondition_momentum(momentum_update, state)
                            if p in self.attn_proj_soap_params:
                                gate = bounded_trust_gate(
                                    trust_gate(momentum_update, soap_update, grad),
                                    self.step_count
                                )
                            else:
                                gate = torch.ones((), dtype=torch.float32, device=p.device)
                            momentum_update = norm_preserving_blend(momentum_update, soap_update, gate)
                        else:
                            momentum_update = soap_precondition_momentum(momentum_update, state)
                    update = muon_update(momentum_update)
                    # ===== MTT3 (avg2): replace the update DIRECTION by the norm-preserving
                    # average of the two half-batch polar factors (norm parity with the base
                    # update by construction; downstream brake/RowFloor/pin/CWD unchanged;
                    # env unset -> dead code). =====
                    if MTT3_AVG2:
                        update = _mtt3_avg2_apply(self, p, grad, state, update, group["mu"], _m3)
                    # ===== MTT3 (ARM B func-polar): blend in the input-metric polar direction
                    # Q_func = msign(Mtilde@R)@R^-1 (norm parity with the base update by fp64
                    # renorm; downstream brake/RowFloor/pin/CWD unchanged; a=0 -> dead code). =====
                    if MTT3_FUNC_POLAR > 0.0 and p in self.mtt3_fp_params:
                        update = _mtt3_func_polar_apply(self, p, momentum_update, update, _m3)
                    # ===== MTT3: O1 replica probes / O7 M-tilde spectrum / panel cosines =====
                    _mtt3_muon_mid(_m3, p, grad, state, momentum_update, update, group["mu"])
                    # ===== MTT3 (K12 I-probe): every-step FDT/Kalman measurement on this
                    # tensor (read-only; env unset -> dead code). =====
                    if MTT3_IPROBE:
                        _mtt3_iprobe_probe(_m3, self, p, grad, state, momentum_update, group["mu"])
                    update = scale_radial_update(update, p)
                    # u/w-floor. SOAP and non-SOAP params can use different floors.
                    p_fro = p.float().norm().clamp_min(1e-8)
                    u_fro = update.float().norm().clamp_min(1e-8)
                    cur_uw = u_fro / p_fro
                    target_uw = SOAP_TARGET_UW if use_soap else NONSOAP_TARGET_UW
                    if ROWFLOOR and p.ndim == 2:
                        # (B) RowFloor: boost each under-updated OUTPUT ROW to its target update/weight
                        # ratio. Per-row SHAPE change; magnitude is re-pinned below, so only the shape
                        # survives the radius pin. RHO=1.0 -> the clamp is the whole floor.
                        r_row = p.float().norm(dim=1, keepdim=True).clamp_min(1e-8)
                        s_row = update.float().norm(dim=1, keepdim=True).clamp_min(1e-8)
                        f_row = torch.clamp(target_uw * r_row / s_row, min=1.0).pow(ROWFLOOR_RHO)
                        update = (update.float() * f_row).to(update.dtype)
                        # ===== MTT3: O5+ RowFloor lift fraction (rows hitting the floor) =====
                        if _m3 is not None:
                            _m3["o5_rowfloor_lift_frac"] = float((f_row > 1.0).float().mean())
                    else:
                        scale = torch.where(cur_uw < target_uw, target_uw * p_fro / u_fro, torch.ones_like(p_fro))
                        update = update * scale.to(update.dtype)
                    target_radius = target_radius_after_update(p, update, _m3_lr)
                    # WD set to 0 — u/w target replaces wd's role (smaller updates as p grows).
                    if CWD > 0.0 and p.ndim == 2:
                        # (C) Cautious Weight Decay: mask the coords where -lr*update already shrinks |p|
                        # (update*p>0); decay only those (never fight a coord the optimizer wants to grow).
                        cwd_mask = (update.float() * p.float() > 0).to(p.dtype)
                    p.add_(update, alpha=-_m3_lr)
                    # ===== MTT3: O5+ pre-pin displacement split + pin rescale factor =====
                    _mtt3_muon_prepin(_m3, p, update, target_radius, _m3_lr)
                    rescale_to_radius(p, target_radius)
                    if CWD > 0.0 and p.ndim == 2:
                        # apply AFTER the radius pin (POST) so the per-coord shape change is preserved.
                        # ===== MTT3 (K4): the CWD dose LR may be decoupled from the update LR.
                        # "mtt3_cwd_lr" is set by set_hparams only when MTT3_CWD_LR_MODE is active
                        # and step >= fork; otherwise .get falls back to group["lr"] (the exact
                        # value used today -> byte-identical, with or without MTT3_LR_MODE).
                        # MTT3_LR_FAMILY_SCALE scales the CWD dose too by default (_m3_fs;
                        # 1.0 when unset -> bit-exact). =====
                        p.mul_(1.0 - (group.get("mtt3_cwd_lr", group["lr"]) * _m3_fs * CWD) * cwd_mask)
                    # ===== MTT3: O2+/O5+ actual-displacement observables (post pin + CWD) =====
                    _mtt3_muon_post(_m3, p, grad, state,
                                    cwd_mask if (CWD > 0.0 and p.ndim == 2) else None)
                    if use_soap:
                        # ===== MTT3: O6 basis drift across this refresh (read-only clones) =====
                        _m3_q_old = _mtt3_muon_o6_pre(_m3, state, use_soap)
                        soap_update_preconditioner(grad, state)
                        _mtt3_muon_o6_post(_m3, state, _m3_q_old)
                    _mtt3_muon_collect(_m3)
                dist.all_gather(params_pad[base_i:base_i + world_size], params_pad[base_i + rank])
        # ===== MTT3 (avg2): reconstruction-guard telemetry (rank-local window max) =====
        if MTT3_AVG2 and self.step_count % 100 == 0:
            print0(f"mtt3 polar avg2 step:{self.step_count} recon_window_max_rel:{_M3.avg2_recon_max:.3e}",
                   console=True)
            _M3.avg2_recon_max = 0.0
        self.step_count += 1


class Adam(torch.optim.Optimizer):
    """Bias-correction-free Adam for the aux (<2D) param groups.

    This is exactly the SOTA's CenterShrinkAdam with center_shrink (rho) = 1.0:
    at rho=1.0 the center-shrink collapses to the identity (mixed == adam_dir and
    the rms-rescale ratio is 1), leaving plain Adam with raw (uncorrected) EMAs.
    NOTE: this intentionally does NOT apply the (1-beta^t) bias correction, so it
    is NOT the same as torch.optim.AdamW. It matches the source under
    AUX_CENTER_SHRINK=1.0 bit-for-bit.
    """
    def __init__(self, params, lr=0.01, betas=(0.8, 0.99), eps=1e-10):
        defaults = dict(lr=lr, betas=betas, eps=eps)
        super().__init__(params, defaults)

    @torch.no_grad()
    def step(self):
        for group in self.param_groups:
            beta1, beta2 = group["betas"]
            eps = group["eps"]
            for p in group["params"]:
                grad = p.grad
                if grad is None:
                    continue
                state = self.state[p]
                if len(state) == 0:
                    state["exp_avg"] = torch.zeros_like(p)
                    state["exp_avg_sq"] = torch.zeros_like(p, dtype=torch.float32)
                exp_avg = state["exp_avg"]
                exp_avg_sq = state["exp_avg_sq"]
                exp_avg.lerp_(grad, 1 - beta1)
                exp_avg_sq.lerp_(grad.float().square(), 1 - beta2)
                adam_dir = exp_avg.float() / (exp_avg_sq.sqrt() + eps)
                p.add_(adam_dir.to(p.dtype), alpha=-group["lr"])


class EMA_Nesterov(torch.optim.Optimizer):
    def __init__(self, params, inner_optimizer, lookahead_stepsize=0, use_scheduled_lookahead_stepsize=True, lookahead_ema=0.9, prefill_steps=0, rest_steps=0):
        if lookahead_stepsize < 0.0:
            raise ValueError("Invalid momentum value: {}".format(lookahead_stepsize))

        super(EMA_Nesterov, self).__init__(params, {})
        self.inner_optimizer = inner_optimizer
        self.use_scheduled_lookahead_stepsize = use_scheduled_lookahead_stepsize
        self.lookahead_stepsize = lookahead_stepsize
        self.lookahead_ema = lookahead_ema
        self.prefill_steps = prefill_steps
        self.rest_steps = rest_steps
        self.it = 0
        self.lookahead_status = False
        self.current_lookahead_stepsize = 0
        self.initialize_buffers()

    def __setstate__(self, state):
        super(EMA_Nesterov, self).__setstate__(state)

    @torch.no_grad()
    def initialize_buffers(self):
        for group in self.param_groups:
            for p in group['params']:
                param_state = self.state[p]
                if 'prev_params' not in param_state:
                    param_state['prev_params'] = (p.clone(), self.it)

                if 'lookahead_buffer' not in param_state:
                    param_state['lookahead_buffer'] = (torch.zeros_like(p), -1)

    def get_lr_lambda(self):
        if isinstance(self.inner_optimizer, list):
            lr_lambda = self.inner_optimizer[0].param_groups[0]["lr"] / self.inner_optimizer[0].param_groups[0]["initial_lr"]
        else:
            lr_lambda = self.inner_optimizer.param_groups[0]["lr"] / self.inner_optimizer.param_groups[0]["initial_lr"]
        return lr_lambda

    @torch.no_grad()
    def lookahead_step(self):
        """Performs nesterov's lookahead."""
        if self.use_scheduled_lookahead_stepsize:
            lookahead_stepsize = self.lookahead_stepsize * self.get_lr_lambda()
        else:
            lookahead_stepsize = self.lookahead_stepsize
        self.current_lookahead_stepsize = lookahead_stepsize

        for group in self.param_groups:
            for p in group['params']:

                param_state = self.state[p]

                lookahead = param_state['lookahead_buffer'][0]

                p.add_(lookahead, alpha=lookahead_stepsize)



    @torch.no_grad()
    def accum_lookahead(self):
        """Update nesterov's lookahead direction."""
        lookahead_ema = self.lookahead_ema
        for group in self.param_groups:
            for p in group['params']:
                param_state = self.state[p]
                look = p.add(param_state['prev_params'][0], alpha=-1)

                # update lookahead buffer
                buf = param_state['lookahead_buffer'][0]

                param_state['lookahead_buffer'] = (buf.lerp_(look, 1 - lookahead_ema), self.it) # m^{t+1} = beta * m^t + (1-beta) * look

                # update prev_params buffer
                param_state['prev_params'] = (param_state['prev_params'][0].copy_(p), self.it)


    @torch.no_grad()
    def nesterov_step(self):
        if self.it + 1 > self.prefill_steps and self.it < self.rest_steps and not self.lookahead_status:
            self.lookahead_step()
        else:
            self.current_lookahead_stepsize = 0
        self.lookahead_status = True

    @torch.no_grad()
    def step(self):
        if not self.lookahead_status:
            raise ValueError("optimizer.nesterov_step() should be invoked before model forward pass.")
        if isinstance(self.inner_optimizer, list):
            for opt in self.inner_optimizer:
                opt.step()
        else:
            self.inner_optimizer.step()

        self.accum_lookahead()

        self.lookahead_status = False
        self.it += 1

def _mtt3_synth_generator(seed_offset: int, nseq: int, seq_len: int, vocab: int):

def _mtt3_halfgrad_hook(step: int, i: int, n_mb: int):
    """After microbatch i's backward: snapshot p.grad at the halfway point.
    Four consumers: the O1 replica probes (diag cadence only, _M3.half_grad), the avg2
    polar TRAINING path (every step, all Muon params, _M3.avg2_half), the K12 I-probe
    (every step, all Muon params, _M3.iprobe_half; preallocated fp32 buffers,
    measurement-only) and the K12.1 FDT TRAINING path (every step, SAME _M3.iprobe_half
    buffers -- shared capture, no duplicate memory; FDT is a pure-run component, so this
    fires independently of MTT3_DIAG)."""
    if MTT3_AVG2 and n_mb < 2:
        raise RuntimeError("MTT3_POLAR_MODE=avg2 needs >=2 grad-accum microbatches per step")
    if MTT3_IPROBE and n_mb < 2:
        raise RuntimeError("MTT3_IPROBE=1 needs >=2 grad-accum microbatches per step")
    if MTT3_FDT_GAMMA > 0.0 and n_mb < 2:
        raise RuntimeError("MTT3_FDT needs >=2 grad-accum microbatches per step")
    if n_mb < 2 or i != n_mb // 2 - 1:
        return
    if MTT3_AVG2:
        with torch.no_grad():
            for p in optimizer2.mtt3_names:
                if p.grad is not None:
                    buf = _M3.avg2_half.get(p)
                    if buf is None:
                        buf = _M3.avg2_half[p] = torch.empty(p.grad.shape, dtype=torch.float32,
                                                             device=p.grad.device)
                    buf.copy_(p.grad.detach())
    # ===== K12 I-probe / K12.1 FDT: EVERY-step half-grad capture on ALL Muon params
    # (read-only; one shared buffer set) =====
    if MTT3_IPROBE or MTT3_FDT_GAMMA > 0.0:
        t0i = time.perf_counter()
        with torch.no_grad():
            for p in optimizer2.mtt3_names:
                if p.grad is not None:
                    buf = _M3.iprobe_half.get(p)
                    if buf is None:
                        buf = _M3.iprobe_half[p] = torch.empty(p.grad.shape, dtype=torch.float32,
                                                               device=p.grad.device)
                    buf.copy_(p.grad.detach())
        dt = time.perf_counter() - t0i
        if MTT3_IPROBE:
            _M3.iprobe_t += dt
            _M3.diag_t += dt
        if MTT3_FDT_GAMMA > 0.0:
            _M3.fdt_t += dt
    if MTT3_DIAG_LEVEL < 2 or not MTT3_DIAG:
        return
    if not _mtt3_is_o1_step(step):
        return
    t0 = time.perf_counter()
    all72 = _mtt3_o1_all72(step)
    with torch.no_grad():
        for p, key in optimizer2.mtt3_names.items():
            if (all72 or key in MTT3_PANEL_KEYS) and p.grad is not None:
                _M3.half_grad[p] = p.grad.detach().float().clone()
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_prestep_capture(step: int):
    """After grad all_reduce, before the optimizer step: O2 dots + pre-step weight copies."""
    t0 = time.perf_counter()
    with torch.no_grad():
        if _M3.half_grad and dist.get_world_size() > 1:
            for p in optimizer2.mtt3_names:          # deterministic order across ranks
                if p in _M3.half_grad:
                    dist.all_reduce(_M3.half_grad[p], op=dist.ReduceOp.SUM)
        for gname, plist in _mtt3_o2_groups():
            tg = 0.0
            t2 = 0.0
            for p in plist:
                pf = p.detach().float()
                tg += float((pf * p.grad.float()).sum())
                t2 += float(pf.square().sum())
                buf = _M3.pre_buf.get(p)
                if buf is None:
                    buf = _M3.pre_buf[p] = torch.empty_like(p)
                buf.copy_(p.detach())
            _M3.o2_partial[gname] = {"tg": tg, "t2": t2}
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_avg2_prestep():
    """avg2: make the half-batch grad snapshot global (SUM over ranks), mirroring the
    p.grad all_reduce in the training loop. World size 1 -> no-op. Kept separate from
    _mtt3_prestep_capture so DIAG+avg2 never double-reduces and DIAG=0 avg2 runs need
    no diag state."""
    if dist.get_world_size() > 1:
        with torch.no_grad():
            for p in optimizer2.mtt3_names:          # deterministic order across ranks
                buf = _M3.avg2_half.get(p)
                if buf is not None:
                    dist.all_reduce(buf, op=dist.ReduceOp.SUM)


def _mtt3_iprobe_prestep():
    """K12 I-probe: make the half-batch grad snapshot global (SUM over ranks) like
    p.grad (avg2 pattern; separate buffer dict so O1/avg2 reductions never double).
    World size 1 -> no-op."""
    if dist.get_world_size() > 1:
        with torch.no_grad():
            for p in optimizer2.mtt3_names:          # deterministic order across ranks
                buf = _M3.iprobe_half.get(p)
                if buf is not None:
                    dist.all_reduce(buf, op=dist.ReduceOp.SUM)


def _mtt3_probe_batch_setup():
    """Fixed probe batch shared by O4 and the ARM-B Sigma_x probe: loaded ONCE at startup
    from a SEPARATE loader instance; the training data iterator is never advanced."""
    if _M3.hvp_batch is not None:
        return
    if MTT3_TEST_MODE:
        xb, yb = next(_mtt3_synth_generator(777, MTT3_TEST_MBS, MTT3_TEST_SEQ, MTT3_TEST_VOCAB))
    else:
        loader = distributed_data_generator("data/fineweb10B/fineweb_train_*.bin", batch_size)
        xb, yb = next(loader)
        xb, yb = xb[:MTT3_O4_SEQS].clone(), yb[:MTT3_O4_SEQS].clone()  # small fixed probe batch (see MTT3_O4_SEQS)
        del loader
    _M3.hvp_batch = (xb, yb)


def _mtt3_o4_setup():
    """O4: fixed cached batch + fixed probe vector, loaded ONCE at startup."""
    _mtt3_probe_batch_setup()
    pgen = torch.Generator().manual_seed(0x04A11CE + SEED)   # independent generator (framework O4)
    probe = [torch.randn(p.shape, generator=pgen, dtype=torch.float32).to(p.device)
             for p in model.parameters()]
    nrm = torch.sqrt(sum((v * v).sum() for v in probe)).clamp_min(1e-30)
    _M3.hvp_probe = [v / nrm for v in probe]

def _mtt3_fp_setup():
    """ARM B setup: probe batch + hook-target module map (short_key -> nn.Module)."""
    _mtt3_probe_batch_setup()
    mods = dict(model.blocks.named_modules())
    _M3.fp_modules = {key: mods[key] for p, key in optimizer2.mtt3_names.items()
                      if p in optimizer2.mtt3_fp_params}


@torch.compiler.disable
def _mtt3_fp_probe_stats(step: int):
    """ARM B Sigma_x capture: one SEPARATE eager forward (model.forward = the O4 route
    that bypasses the compiled graph) on the fixed cached probe batch, with TEMPORARY
    forward-pre-hooks on the scoped Linear layers collecting per-layer inputs x and
    computing cov = x^T x / N. Hooks are registered around this one eager call and
    removed in `finally`, so the compiled training forward never runs with a non-empty
    hook dict (guard state restored; NO data-dependent branches enter the compiled
    graph -- EFC lesson, journal 2026-07-03). Sigma_x estimates therefore come from a
    FIXED batch (documented limitation, same convention as O4); the EMA
    (beta=MTT3_FP_BETA) integrates over probe ticks. Training RNG is asserted identical
    before/after; the training data iterator is never touched (own loader at setup)."""
    rng_cpu = torch.get_rng_state()
    rng_cuda = torch.cuda.get_rng_state_all() if torch.cuda.is_available() else []
    xb, yb = _M3.hvp_batch
    feats = {}
    handles = []

    def _mk(key):
        def _hook(module, args):
            x = args[0]
            with torch.no_grad():
                xf = x.detach().float().reshape(-1, x.size(-1))
                cov = (xf.T @ xf) / xf.size(0)
            prev = feats.get(key)
            feats[key] = cov if prev is None else prev + cov
        return _hook

    for key, mod in _M3.fp_modules.items():
        handles.append(mod.register_forward_pre_hook(_mk(key)))
    try:
        with torch.no_grad():
            model.forward(xb, yb)
    finally:
        for h in handles:
            h.remove()
    if dist.get_world_size() > 1:
        for key in sorted(feats):                    # deterministic order across ranks
            dist.all_reduce(feats[key], op=dist.ReduceOp.SUM)
            feats[key] /= dist.get_world_size()
    fresh = 0
    for key in sorted(feats):
        sig = _M3.fp_sigma.get(key)
        if sig is None:
            _M3.fp_sigma[key] = feats[key]           # lazy init from the first probe estimate
            fresh += 1
        else:
            sig.lerp_(feats[key], 1.0 - MTT3_FP_BETA)
    if fresh and not _M3.fp_announced:
        _M3.fp_announced = True
        print0(f"mtt3 func polar: sigma lazy-init ({fresh} layers) at step {step}", console=True)
    ok = torch.equal(rng_cpu, torch.get_rng_state())
    if ok and torch.cuda.is_available():
        ok = all(torch.equal(a, b) for a, b in zip(rng_cuda, torch.cuda.get_rng_state_all()))
    if not ok:
        raise RuntimeError("MTT3 func-polar: training RNG state was changed by the Sigma_x probe")


def _mtt3_fp_refresh():
    """ARM B metric refresh: (R, R^{-1}) per scoped layer from the Sigma_x EMA via eigh
    (mtt3_fp_metric_from_sigma: trace-normalized, eigenvalues clamped at MTT3_FP_EPS
    relative to the mean -> SPD even for rank-deficient / zero Sigma_x)."""
    for key in sorted(_M3.fp_sigma):
        r, r_inv = mtt3_fp_metric_from_sigma(_M3.fp_sigma[key], eps=MTT3_FP_EPS)
        _M3.fp_R[key] = r
        _M3.fp_Rinv[key] = r_inv


def _mtt3_fp_prestep(step: int):
    """ARM B cadence driver (training path; called right before the optimizer step).
    The `or not ...` arms make a mid-cadence fork/resume deterministic: stats + metric
    exist from the first post-resume Muon step on (fp_state also rides the ckpt, so a
    bit-exact roundtrip never re-probes out of cadence)."""
    if step % MTT3_FP_STAT_EVERY == 0 or not _M3.fp_sigma:
        _mtt3_fp_probe_stats(step)
    if step % MTT3_FP_REFRESH == 0 or not _M3.fp_R:
        _mtt3_fp_refresh()


@torch.compiler.disable
def _mtt3_hvp_once(vs):
    """One Hessian-vector product on the fixed cached batch (eager; dynamo disabled).
    Math SDPA is forced locally: flash/mem-efficient kernels have no double backward
    (the context manager restores the ambient backend selection on exit)."""
    from torch.nn.attention import sdpa_kernel, SDPBackend
    xb, yb = _M3.hvp_batch
    params = list(model.parameters())
    with sdpa_kernel([SDPBackend.MATH]):
        # model.forward = the ORIGINAL eager forward (Module.compile only swaps __call__'s
        # _compiled_call_impl). Bypasses the compiled graph whose donated-buffer backward
        # forbids create_graph=True (torch 2.7 functorch restriction seen at smoke step 0).
        loss = model.forward(xb, yb) / yb.numel()
        grads = torch.autograd.grad(loss, params, create_graph=True)
        gv = sum((g.float() * v).sum() for g, v in zip(grads, vs))
        hv = torch.autograd.grad(gv, params)
    return [h.detach().float() for h in hv]


def _mtt3_o4(step: int):
    """lambda_max(H) via k-step power iteration. Must leave training RNG + data stream untouched."""
    t0 = time.perf_counter()
    rng_cpu = torch.get_rng_state()
    rng_cuda = torch.cuda.get_rng_state_all() if torch.cuda.is_available() else []
    rec = _mtt3_rec()
    try:
        with torch.enable_grad():
            v = [x.clone() for x in _M3.hvp_probe]
            lam = float("nan")
            for _ in range(MTT3_O4_POWER_ITERS):
                hv = _mtt3_hvp_once(v)
                if dist.get_world_size() > 1:
                    for h in hv:
                        dist.all_reduce(h, op=dist.ReduceOp.SUM)
                    hv = [h / dist.get_world_size() for h in hv]
                lam = float(sum((a * b).sum() for a, b in zip(v, hv)))
                nrm = torch.sqrt(sum((h * h).sum() for h in hv)).clamp_min(1e-30)
                v = [h / nrm for h in hv]
        rec["o4_lambda_max"] = lam
        rec["o4_tokens"] = int(_M3.hvp_batch[1].numel())
    except Exception as e:  # keep the run alive; O4 is a diag-only probe
        _M3.o4_dead = True
        rec["o4_error"] = repr(e)
    ok = torch.equal(rng_cpu, torch.get_rng_state())
    if ok and torch.cuda.is_available():
        ok = all(torch.equal(a, b) for a, b in zip(rng_cuda, torch.cuda.get_rng_state_all()))
    if not ok:
        raise RuntimeError("MTT3 O4: training RNG state was changed by the HVP probe")
    rec["o4_rng_ok"] = True
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_poststep(step: int):
    """After the optimizer step + zero_grad: O2 group displacements, then O4 at its cadence."""
    t0 = time.perf_counter()
    with torch.no_grad():
        for gname, plist in _mtt3_o2_groups():
            v2 = 0.0
            for p in plist:
                v2 += float((p.detach().float() - _M3.pre_buf[p].float()).square().sum())
            _M3.o2_partial[gname]["v2"] = v2
    _M3.diag_t += time.perf_counter() - t0
    if MTT3_DIAG_LEVEL >= 2 and step % MTT3_O4_EVERY == 0 and not _M3.o4_dead:
        _mtt3_o4(step)


def _mtt3_multitau_update(step: int):
    """O8+: maintain the 5 EMA heads with the SAME window/update rule as the base Tail-EMA.
    The tau=150 copy must remain bit-equal to the base _tailema head (guarded by test e)."""
    t0 = time.perf_counter()
    if TAILEMA_TAU > 0 and TAILEMA_START <= (step + 1) < TAILEMA_END:
        if _M3.multitau is None:
            _M3.multitau = {tau: [None if p is model.embed.weight else p.detach().float().clone()
                                  for p in model.parameters()] for tau in MTT3_MULTITAU}
        else:
            for tau, bufs in _M3.multitau.items():
                for j, p in enumerate(model.parameters()):
                    if bufs[j] is not None:
                        bufs[j].add_(p.detach().float() - bufs[j], alpha=1.0 / tau)
    # K5 axis A reset heads: same update rule / window end, EMA starts at MTT3_O8_RESET_START.
    if MTT3_O8_RESET_START > 0 and TAILEMA_TAU > 0 and MTT3_O8_RESET_START <= (step + 1) < TAILEMA_END:
        if _M3.multitau_reset is None:
            _M3.multitau_reset = {key: [None if p is model.embed.weight else p.detach().float().clone()
                                        for p in model.parameters()] for key, _ in MTT3_O8_RESET_HEADS}
        else:
            for key, tau in MTT3_O8_RESET_HEADS:
                bufs = _M3.multitau_reset[key]
                for j, p in enumerate(model.parameters()):
                    if bufs[j] is not None:
                        bufs[j].add_(p.detach().float() - bufs[j], alpha=1.0 / tau)
    _M3.diag_t += time.perf_counter() - t0


def _mtt3_val_multitau(step: int, val_loss, val_ema_loss):
    """O8+ readout: evaluate the 4 extra EMA heads with the base blend rule (lambda=0.6).
    The base tau=150 head is NOT re-evaluated: val_ema_loss IS its value."""
    t0 = time.perf_counter()
    rec = _mtt3_rec()
    rec.setdefault("step", step)
    rec["val_loss"] = float(val_loss)
    rec["val_ema_loss"] = float(val_ema_loss)
    vals = {str(int(TAILEMA_TAU)): float(val_ema_loss)}
    if _M3.multitau is not None:
        with torch.no_grad():
            for tau in MTT3_MULTITAU:
                if abs(tau - TAILEMA_TAU) < 1e-9:
                    continue
                bufs = _M3.multitau[tau]
                stash = [p.detach().clone() for p in model.parameters()]
                for j, p in enumerate(model.parameters()):
                    if bufs[j] is not None:
                        p.copy_(((1.0 - TAILEMA_LAMBDA) * p.float() + TAILEMA_LAMBDA * bufs[j]).to(p.dtype))
                v = 0
                for i in range(len(val_inputs) // mbs):
                    v += model(val_inputs[i*mbs:(i+1)*mbs], val_targets[i*mbs:(i+1)*mbs])
                dist.all_reduce(v, op=dist.ReduceOp.SUM)
                v /= val_tokens
                for p, s in zip(model.parameters(), stash):
                    p.copy_(s)
                vals[str(int(tau))] = float(v)
    else:
        for tau in MTT3_MULTITAU:
            if abs(tau - TAILEMA_TAU) >= 1e-9:
                vals[str(int(tau))] = float(val_loss)   # same pre-window default as the base head
    if MTT3_O8_RESET_START > 0:
        # K5 axis A reset heads: base blend rule (lambda=0.6); raw weights until their window starts.
        if _M3.multitau_reset is not None:
            with torch.no_grad():
                for key, _tau in MTT3_O8_RESET_HEADS:
                    bufs = _M3.multitau_reset[key]
                    stash = [p.detach().clone() for p in model.parameters()]
                    for j, p in enumerate(model.parameters()):
                        if bufs[j] is not None:
                            p.copy_(((1.0 - TAILEMA_LAMBDA) * p.float() + TAILEMA_LAMBDA * bufs[j]).to(p.dtype))
                    v = 0
                    for i in range(len(val_inputs) // mbs):
                        v += model(val_inputs[i*mbs:(i+1)*mbs], val_targets[i*mbs:(i+1)*mbs])
                    dist.all_reduce(v, op=dist.ReduceOp.SUM)
                    v /= val_tokens
                    for p, s in zip(model.parameters(), stash):
                        p.copy_(s)
                    vals[key] = float(v)
        else:
            for key, _tau in MTT3_O8_RESET_HEADS:
                vals[key] = float(val_loss)             # same pre-window default as the base head
    rec["val_ema"] = vals
    _M3.val_records.append({"step": step, "val_loss": float(val_loss),
                            "val_ema_loss": float(val_ema_loss), "val_ema": dict(vals)})
    rec["o8_eval_s"] = time.perf_counter() - t0
    print0("mtt3 o8 step:%d " % step
           + " ".join(f"tau{k}:{v:.5f}" for k, v in
                      sorted(vals.items(), key=lambda kv: (kv[0].startswith("r"), int(kv[0].lstrip("r"))))),
           console=True)


def _mtt3_step_end(step: int):
    """Assemble this step's record; rank 0 buffers it; flush every MTT3_FLUSH_EVERY steps."""
    t0 = time.perf_counter()
    muon = _M3.muon
    spectra = _M3.spectra
    if dist.get_world_size() > 1:   # per-tensor shards live on the owning rank
        g1 = [None] * dist.get_world_size()
        dist.all_gather_object(g1, muon)
        muon = {}
        for d in g1:
            muon.update(d)
        g2 = [None] * dist.get_world_size()
        dist.all_gather_object(g2, spectra)
        spectra = {}
        for d in g2:
            spectra.update(d)
    if dist.get_rank() == 0:
        rec = _mtt3_rec()
        rec["step"] = step
        pivot = {}
        for key, ctx in muon.items():
            for f, v in ctx.items():
                pivot.setdefault(f, {})[key] = v
        rename = {"tg": "o2_tensor_tg", "t2": "o2_tensor_t2", "v2": "o2_tensor_v2"}
        for f, d in pivot.items():
            rec[rename.get(f, f)] = d
        o2 = _M3.o2_partial
        if muon:
            o2["muon"] = {"tg": sum(c.get("tg", 0.0) for c in muon.values()),
                          "t2": sum(c.get("t2", 0.0) for c in muon.values()),
                          "v2": sum(c.get("v2", 0.0) for c in muon.values())}
        rec["o2"] = o2
        if spectra:
            import numpy as np
            arrs = {}
            for key, d in spectra.items():
                for kind, s in d.items():
                    arrs[f"{key}|{kind}"] = s.numpy() if torch.is_tensor(s) else s
            fn = _mtt3_diag_dir() / f"spectra_step{step}.npz"
            np.savez(fn, **arrs)
            rec["o7_npz"] = fn.name
        rec["lr_muon"] = float(optimizer2.param_groups[0]["lr"])
        if MTT3_CWD_LR_MODE:   # K4: record the decoupled CWD-dose LR (key absent otherwise)
            rec["lr_muon_cwd"] = float(optimizer2.param_groups[0].get(
                "mtt3_cwd_lr", optimizer2.param_groups[0]["lr"]))
        rec["mu"] = float(optimizer2.param_groups[0]["mu"])
        rec["wall"] = time.time() - _M3.wall0
        rec["diag_s"] = _M3.diag_t + (time.perf_counter() - t0)
        # ===== K12 I-probe: per-step probe wall time (rank-0/rank-local; the prereg
        # probe run is single-GPU) + console timing line at cadence =====
        if MTT3_IPROBE:
            rec["iprobe_s"] = _M3.iprobe_t
            if step % 100 == 0:
                print0(f"mtt3 iprobe step:{step} iprobe_s:{_M3.iprobe_t:.3f}s", console=True)
        _M3.records.append(rec)
    _M3.record = None
    _M3.half_grad.clear()
    _M3.mom_old.clear()
    if step % MTT3_FLUSH_EVERY == 0:
        _mtt3_flush()


def _mtt3_flush(final: bool = False):
    if not MTT3_DIAG:
        return
    if final and _M3.record is not None:      # e.g. the final val-only record
        rec = _M3.record
        rec["wall"] = time.time() - _M3.wall0
        if dist.get_rank() == 0:
            _M3.records.append(rec)
        _M3.record = None
    if dist.get_rank() != 0 or not _M3.records:
        return
    path = Path(MTT3_DIAG_PATH) if MTT3_DIAG_PATH else (_mtt3_diag_dir() / f"mtt3_diag_seed{SEED}.jsonl")
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "a") as f:
        for r in _M3.records:
            f.write(_mtt3_json.dumps(r) + "\n")
    _M3.emitted += len(_M3.records)
    _M3.records = []

# ---------- MTT3 checkpointing (full state + fp64 sqsum checksums) ----------

def _mtt3_sqsum(t) -> float:
    return float(t.detach().double().square().sum()) if torch.is_tensor(t) else 0.0


def _mtt3_opt_state_tensors(opt, params):
    # NOTE: opt.state is a defaultdict; use .get so observation never creates entries.
    out = []
    for p in params:
        st = opt.state.get(p)
        if st:
            out.extend(v for k, v in sorted(st.items()) if torch.is_tensor(v))
    return out


def _mtt3_train_state_groups():
    """Named groups of live TRAINING-state tensors (checksum + test-dump basis)."""
    g = {"model": [p.detach() for p in model.parameters()]}
    g["adamw"] = _mtt3_opt_state_tensors(optimizer1, [p for grp in optimizer1.param_groups for p in grp["params"]])
    for name in ("muon_momentum", "soap_exp_avg_sq", "soap_row_gg", "soap_col_gg", "soap_q_row", "soap_q_col",
                 "muon_avg2_ma", "muon_avg2_mb", "muon_mslow", "muon_bm_mf", "muon_bm_ms",
                 "muon_fdt_m0", "muon_fdt_m1", "muon_fdt_m2", "muon_fdt_m3", "muon_fdt_m4"):
        g[name] = []
    for p in _mtt3_muon_params():
        st = optimizer2.state.get(p)
        if not st:
            continue
        g["muon_momentum"].append(st["momentum"])
        for k, gk in (("exp_avg_sq", "soap_exp_avg_sq"), ("row_gg", "soap_row_gg"),
                      ("col_gg", "soap_col_gg"), ("q_row", "soap_q_row"), ("q_col", "soap_q_col"),
                      ("mtt3_ma", "muon_avg2_ma"), ("mtt3_mb", "muon_avg2_mb"),
                      ("mtt3_mslow", "muon_mslow"),
                      ("mtt3_bm_mf", "muon_bm_mf"), ("mtt3_bm_ms", "muon_bm_ms"),
                      ("mtt3_fdt_m0", "muon_fdt_m0"), ("mtt3_fdt_m1", "muon_fdt_m1"),
                      ("mtt3_fdt_m2", "muon_fdt_m2"), ("mtt3_fdt_m3", "muon_fdt_m3"),
                      ("mtt3_fdt_m4", "muon_fdt_m4")):
            v = st.get(k)
            if torch.is_tensor(v):
                g[gk].append(v)
    g["fp_sigma"] = list(_M3.fp_sigma.values())   # ARM B input-metric EMA (empty when off)
    g["aux_adam"] = _mtt3_opt_state_tensors(optimizer3, [p for grp in optimizer3.param_groups for p in grp["params"]])
    ema = optimizers[0]
    g["ema_prev"] = [ema.state[p]["prev_params"][0] for grp in ema.param_groups for p in grp["params"]]
    g["ema_lookahead"] = [ema.state[p]["lookahead_buffer"][0] for grp in ema.param_groups for p in grp["params"]]
    g["tailema"] = [b for b in (_tailema or []) if b is not None]
    return g


def _mtt3_checksums():
    return {name: float(sum(_mtt3_sqsum(t) for t in ts))
            for name, ts in _mtt3_train_state_groups().items()}

def _mtt3_load_ckpt(path: str) -> int:
    global _tailema, training_time
    rank = dist.get_rank()
    pth = Path(path)
    if dist.get_world_size() > 1 and "_rank0" in pth.name:
        pth = pth.with_name(pth.name.replace("_rank0", f"_rank{rank}"))
    ck = torch.load(pth, weights_only=False)
    assert ck["world_size"] == dist.get_world_size(), "mtt3 resume: world-size mismatch"
    ema = optimizers[0]
    with torch.no_grad():
        for n, p in model.named_parameters():
            src = ck["model"][n]
            assert src.dtype == p.dtype and src.shape == p.shape, n
            p.copy_(src.to(p.device))
        i = 0
        for grp in optimizer1.param_groups:
            for p in grp["params"]:
                st = ck["adamw_state"][i]; i += 1
                if st:
                    optimizer1.state[p] = st
        mp = _mtt3_muon_params()
        assert len(mp) == len(ck["muon_state"])
        for p, st in zip(mp, ck["muon_state"]):
            if st:
                optimizer2.state[p] = st
        optimizer2.step_count = ck["muon_step_count"]
        i = 0
        for grp in optimizer3.param_groups:
            for p in grp["params"]:
                st = ck["aux_state"][i]; i += 1
                if st:
                    optimizer3.state[p] = st
        i = 0
        for grp in ema.param_groups:
            for p in grp["params"]:
                st = ck["ema_state"][i]; i += 1
                ema.state[p]["prev_params"] = (st["prev_params"][0], st["prev_params"][1])
                ema.state[p]["lookahead_buffer"] = (st["lookahead_buffer"][0], st["lookahead_buffer"][1])

        ema.lookahead_status = ck["ema_lookahead_status"]
        ema.current_lookahead_stepsize = ck["ema_current_lookahead_stepsize"]
        _tailema = ck["tailema"]
        if ck.get("multitau") is not None:
            _M3.multitau = ck["multitau"]
        if ck.get("multitau_reset") is not None:   # absent in pre-axis-A ckpts: lazy-init later
            _M3.multitau_reset = ck["multitau_reset"]
        if ck.get("fp_state"):                     # absent in pre-ARM-B ckpts: lazy re-probe at the fork
            _M3.fp_sigma = dict(ck["fp_state"]["sigma"])
            _M3.fp_R = dict(ck["fp_state"]["r"])
            _M3.fp_Rinv = dict(ck["fp_state"]["rinv"])
        ds = ck.get("diag_state") or {}
        o3 = {h: {} for h in MTT3_O3_HORIZONS}
        o3.update(ds.get("o3_ema", {}))
        _M3.o3_ema = o3
        _M3.prev_dw = dict(ds.get("prev_dw", {}))
        _M3.prev_q = dict(ds.get("prev_q", {}))
    torch.set_rng_state(ck["rng_cpu"])

def _mtt3_save_snapshot(step: int):
    """snap_step{N}.pt: non-embedding params as MTT3_SNAP_DTYPE (bf16 default) CPU
    tensors. Synchronous by design (~0.25 GB at bf16/fp16, ~0.5 GB at fp32:
    device-side cast -> one PCIe copy -> torch.save); the per-save wall time is
    printed so the <2 s budget is auditable from the log."""
    if dist.get_rank() != 0:
        return
    t0s = time.perf_counter()
    snapdir = Path(MTT3_SNAP_DIR)
    snapdir.mkdir(parents=True, exist_ok=True)
    snap_dt = {"bf16": torch.bfloat16, "fp16": torch.float16,
               "fp32": torch.float32}[MTT3_SNAP_DTYPE]
    with torch.no_grad():
        payload = {
            "mtt3_snap_version": 1, "kind": "step", "step": step, "seed": SEED,
            "snap_every": MTT3_SNAP_EVERY, "snap_start": MTT3_SNAP_START,
            "snap_dtype": MTT3_SNAP_DTYPE, "snap_dense": MTT3_SNAP_DENSE,
            "params": {n: p.detach().to(snap_dt).to("cpu")
                       for n, p in _mtt3_snap_named_params()},
        }
    path = snapdir / f"snap_step{step}.pt"
    torch.save(payload, path)
    print0(f"mtt3 snap saved step:{step} -> {path} ({time.perf_counter()-t0s:.2f}s)",
           console=True)

_MTT3_GCR_RING = {}       # step -> {name: MTT3_SNAP_DTYPE CPU tensor} (hidden 2-D matrices only)
_MTT3_GCR_RECORDS = []    # rank-0 per-val-point summaries {step, val_gcr_loss, gcr_eval_s, n_snaps}
_MTT3_GCR_LAST = None     # test-mode-only: the last composite non-embed weights (fp32 CPU)


def _mtt3_gcr_hidden_named_params():
    """The GCR hidden-matrix set: named parameters under blocks.* with ndim == 2 --
    exactly eval_readouts.is_hidden_matrix AND exactly the trainer's Muon param set
    (model.blocks.named_parameters() with ndim >= 2; 72 matrices at base config).
    Everything else non-embedding keeps the baseline Tail-EMA readout
    (eval_readouts.py section 7.3 split)."""
    return [(n, p) for n, p in model.named_parameters()
            if n.startswith("blocks.") and p.ndim == 2]


def _mtt3_gcr_capture(step: int):
    """In-RAM ring capture (every rank keeps its own identical ring -- params are
    replicated and the composite eval below is a collective). Same cast chain as
    _mtt3_save_snapshot (device-side cast -> one CPU copy), read-only w.r.t.
    training state; ring tensors bit-equal the disk snapshots' hidden entries
    (guarded by tests/test n). From-scratch runs only ride this ring; a mid-tail
    ckpt resume would rebuild it from the resume step on (documented limitation)."""
    snap_dt = {"bf16": torch.bfloat16, "fp16": torch.float16,
               "fp32": torch.float32}[MTT3_SNAP_DTYPE]
    with torch.no_grad():
        # copy=True: .to() is a no-op alias when dtype AND device already match
        # (fp32 CPU test mode) -- the ring must FREEZE the weights, never track them.
        _MTT3_GCR_RING[step] = {n: p.detach().to(snap_dt).to("cpu", copy=True)
                                for n, p in _mtt3_gcr_hidden_named_params()}
    if len(_MTT3_GCR_RING) == 1:
        print0(f"mtt3 gcr ring: first capture at step {step} "
               f"(dtype={MTT3_SNAP_DTYPE}, every={MTT3_GCR_EVERY}, "
               f"{len(_MTT3_GCR_RING[step])} hidden matrices)", console=True)


def _mtt3_gcr_val(step: int):
    """val_gcr_loss at one val point (>= MTT3_GCR_VAL_START): the H3 composite
    readout eval. Hidden 2-D matrices <- the online H3 estimate from the causal
    ring (all ring steps <= `step` by construction; fp32 compute on the params'
    device, streaming one matrix stack at a time); every other non-embedding
    param <- the baseline Tail-EMA blend (the val_ema code path's exact rule with
    the LIVE fp32 buffer); the token embedding stays live. stash -> blend -> eval
    -> restore, mirroring the base Tail-EMA block; no RNG, no data-iterator use.
    Returns (val_gcr_loss, wall_s, n_snaps)."""
    global _MTT3_GCR_LAST
    t0g = time.perf_counter()
    ksteps = sorted(k for k in _MTT3_GCR_RING if k <= step)   # causal by construction
    est = {}
    with torch.no_grad():
        for n, p in _mtt3_gcr_hidden_named_params():
            stack = torch.stack([_MTT3_GCR_RING[k][n] for k in ksteps]).to(p.device).float()
            est[n] = _mtt3_gcr_h3_estimate(stack, ksteps, p.detach().float(),
                                           MTT3_GCR_LAMBDA, TAILEMA_TAU,
                                           TAILEMA_START, TAILEMA_END)
            del stack
        stash = [p.detach().clone() for p in model.parameters()]
        for j, (n, p) in enumerate(model.named_parameters()):
            if n in est:
                p.copy_(est[n].to(p.dtype))
            elif _tailema is not None and _tailema[j] is not None:
                p.copy_(((1.0 - TAILEMA_LAMBDA) * p.float() + TAILEMA_LAMBDA * _tailema[j]).to(p.dtype))
        if MTT3_TEST_MODE:   # parity-test hook: the composite non-embed weights as applied
            _MTT3_GCR_LAST = {n: p.detach().float().to("cpu").clone()
                              for n, p in model.named_parameters()
                              if p is not model.embed.weight}
        val_gcr = 0
        for i in range(len(val_inputs) // mbs):
            val_gcr += model(val_inputs[i*mbs:(i+1)*mbs], val_targets[i*mbs:(i+1)*mbs])
        dist.all_reduce(val_gcr, op=dist.ReduceOp.SUM)
        val_gcr /= val_tokens
        for p, s in zip(model.parameters(), stash):
            p.copy_(s)
    wall = time.perf_counter() - t0g
    if dist.get_rank() == 0:
        _MTT3_GCR_RECORDS.append({"step": step, "val_gcr_loss": float(val_gcr),
                                  "gcr_eval_s": round(wall, 2), "n_snaps": len(ksteps)})
    return float(val_gcr), wall, len(ksteps)

def headmean_antithetic_pair(shape, *, device, layer_idx: int, seed: int, head_dim: int = HEAD_DIM):
    width = shape[-1]
    if width % head_dim != 0:
        s0 = (torch.randint(0, 2, shape, device=device, dtype=torch.float32) * 2 - 1)
        return s0, -s0
    gen = torch.Generator(device=device)
    gen.manual_seed(0xA9170000 + seed * 1009 + layer_idx * 9176)
    heads0 = []
    heads1 = []
    for _ in range(width // head_dim):
        h0 = (torch.randint(0, 2, (head_dim,), device=device, generator=gen, dtype=torch.int64) * 2 - 1).float()
        plus0 = int((h0 > 0).sum().item())
        plus0 = int(round(head_dim / 2 + _CGI_HEAD_MEAN_SHRINK * (plus0 - head_dim / 2)))
        plus0 = max(0, min(head_dim, plus0))
        h0 = torch.cat([
            torch.ones(plus0, device=device, dtype=torch.float32),
            -torch.ones(head_dim - plus0, device=device, dtype=torch.float32),
        ])
        h0 = h0[torch.randperm(head_dim, device=device, generator=gen)]
        plus1 = head_dim - plus0
        h1 = torch.cat([
            torch.ones(plus1, device=device, dtype=torch.float32),
            -torch.ones(head_dim - plus1, device=device, dtype=torch.float32),
        ])
        h1 = h1[torch.randperm(head_dim, device=device, generator=gen)]
        heads0.append(h0)
        heads1.append(h1)
    return torch.cat(heads0).reshape(shape), torch.cat(heads1).reshape(shape)

# learning rate schedule: stable then decay
def _lr(step, initial_lr, power_c, power=1.0):
    t_end = FINAL_SCHEDULE_STEPS
    flat_lr = initial_lr
    downward_lr = power_c * max(0.0, t_end - step) ** power
    return min(flat_lr, downward_lr)
